     1→// P0-24 Sub-Wave 2b — Outbox Publisher Worker
     2→//
     3→// Cron-triggered publisher with DB-backed lease/claim + retry state.
     4→//
     5→// This is NOT a "continuously running worker that polls." It is a
     6→// CRON-TRIGGERED process that:
     7→//   1. Claims PENDING events (atomic UPDATE ... WHERE status='PENDING' → CLAIMED)
     8→//   2. Publishes each claimed event via Socket.io
     9→//   3. Marks as PUBLISHED (success) or increments attempts (failure)
    10→//   4. After max retries (5), marks as FAILED + alerts
    11→//   5. Recovers stale CLAIMED events (lease expired → back to PENDING)
    12→//
    13→// Crash safety: if the process dies between claim + publish, the lease
    14→// expires and a future invocation re-claims the event.
    15→//
    16→// Run via: Vercel Cron (1-minute interval) OR manual trigger
    17→// Port: 3009 (for health check endpoint)
    18→

   117→// ----------------------------------------------------------------------------
   118→// Wave-4 Sub-Wave 4c Phase 2 — PAYMENT_CAPTURE_REQUESTED handler
   119→// ----------------------------------------------------------------------------
   120→// Command-event handler: consumes a PAYMENT_CAPTURE_REQUESTED outbox row and
   121→// performs the actual Razorpay capture (safely OUTSIDE any transaction body).
   122→//
   123→// Safety properties (the whole point of Wave-4 4c):
   124→//   1. captureRazorpayPayment() is called OUTSIDE any DB transaction. If the
   125→//      success-path txn retries (Prisma P2034 conflict), the capture call is
   126→//      NOT re-executed — preventing double-charge at the gateway.
   127→//   2. Idempotency: if Payment.status is already CAPTURED (e.g., a webhook
   128→//      raced ahead), the handler marks the outbox event PUBLISHED and exits.
   129→//   3. Race-safe Payment update: conditional updateMany (WHERE status='CAPTURE_PENDING')
   130→//      prevents overwriting CAPTURED/FAILED status set by a concurrent path.
   131→//   4. Atomic success commit: Payment.status=CAPTURED + capturedAt + AuditLog +
   132→//      Outbox.status=PUBLISHED all commit in the SAME txn — no half-states.
   133→//
   134→// On capture failure: Payment.retryCount is incremented + failureReason set,
   135→// status is left as CAPTURE_PENDING, and the handler throws — the publisher's
   136→// existing retry/backoff/FAILED logic then drives the outbox event's lifecycle.
   137→// ----------------------------------------------------------------------------
   138→
   139→interface CaptureRequestedPayload {
   140→  paymentId: string
   141→  orderId: string
   142→  gatewayPaymentId: string
   143→  amount: number
   144→}
   145→
   146→async function processPaymentCaptureRequested(event: {
   147→  id: string
   148→  eventId: string
   149→  aggregateId: string
   150→  payload: string
   151→  attempts: number
   152→}): Promise<void> {
   153→  // Parse payload (written by capture route's enqueueOutboxEvent call)
   154→  let payload: CaptureRequestedPayload
   155→  try {
   156→    payload = JSON.parse(event.payload) as CaptureRequestedPayload
   157→  } catch {
   158→    // Malformed payload — non-retryable; mark outbox PUBLISHED to stop retries.
   159→    // (This should never happen since enqueueOutboxEvent JSON.stringify's the payload.)
   160→    await db.outbox.update({
   161→      where: { id: event.id },
   162→      data: {
   163→        status: 'PUBLISHED',
   164→        publishedAt: new Date(),
   165→        claimedAt: null,
   166→        claimUntil: null,
   167→        workerId: null,
   168→        lastError: 'malformed-payload-marked-published',
   169→      },
   170→    })
   171→    await log({
   172→      level: 'error',
   173→      message: 'capture-payload-malformed',
   174→      eventId: event.eventId,
   175→      eventType: 'PAYMENT_CAPTURE_REQUESTED',
   176→    })
   177→    return
   178→  }
   179→
   180→  // 1. Read the Payment (aggregateId = paymentId per the capture route's
   181→  //    enqueueOutboxEvent({ aggregateType: 'Payment', aggregateId: payment.id, ... }))
   182→  const payment = await db.payment.findUnique({
   183→    where: { id: event.aggregateId },
   184→    select: {
   185→      id: true,
   186→      status: true,
   187→      amount: true,
   188→      currency: true,
   189→      gatewayPaymentId: true,
   190→      retryCount: true,
   191→      version: true,
   192→    },
   193→  })
   194→
   195→  if (!payment) {
   196→    // Payment row missing — non-retryable; mark outbox PUBLISHED to stop
   197→    // retrying. (By design this is impossible: the capture route writes Payment
   198→    // + Outbox in the SAME txn, so an outbox row can't exist without a Payment.)
   199→    await db.outbox.update({
   200→      where: { id: event.id },
   201→      data: {
   202→        status: 'PUBLISHED',
   203→        publishedAt: new Date(),
   204→        claimedAt: null,
   205→        claimUntil: null,
   206→        workerId: null,
   207→        lastError: `payment-not-found:aggregateId=${event.aggregateId}`,
   208→      },
   209→    })
   210→    await log({
   211→      level: 'error',
   212→      message: 'capture-payment-not-found',
   213→      eventId: event.eventId,
   214→      eventType: 'PAYMENT_CAPTURE_REQUESTED',
   215→      paymentId: event.aggregateId,
   216→    })
   217→    return
   218→  }
   219→
   220→  // 2. Idempotency: if already CAPTURED (e.g., a webhook raced ahead and
   221→  //    captured first, or a prior publisher invocation captured but failed to
   222→  //    mark the outbox PUBLISHED), the capture command has already succeeded.
   223→  //    Mark the outbox event PUBLISHED and exit.
   224→  if (payment.status === 'CAPTURED') {
   225→    await db.outbox.update({
   226→      where: { id: event.id },
   227→      data: {
   228→        status: 'PUBLISHED',
   229→        publishedAt: new Date(),
   230→        claimedAt: null,
   231→        claimUntil: null,
   232→        workerId: null,
   233→      },
   234→    })
   235→    await log({
   236→      level: 'info',
   237→      message: 'capture-already-captured-idempotent',
   238→      eventId: event.eventId,
   239→      eventType: 'PAYMENT_CAPTURE_REQUESTED',
   240→      paymentId: payment.id,
   241→    })
   242→    return
   243→  }
   244→
   245→  // 3. If Payment is in a non-capture-pending state (FAILED, FROZEN, etc.),
   246→  //    do NOT attempt capture. Throw so the publisher's existing retry path
   247→  //    handles it; if it exhausts retries, the outbox event becomes FAILED +
   248→  //    alerts fire (manual intervention required).
   249→  if (payment.status !== 'CAPTURE_PENDING') {
   250→    throw new Error(
   251→      `Payment ${payment.id} status is ${payment.status} (expected CAPTURE_PENDING) — capture skipped`,
   252→    )
   253→  }
   254→
   255→  // 4. Call captureRazorpayPayment() OUTSIDE any transaction body.
   256→  //    This is the Wave-4 4c safety improvement: if the success-path txn
   257→  //    retries (P2034 conflict), this call is NOT re-executed.
   258→  //    In demo mode (realPayments=false), this returns mock success immediately.
   259→  const gatewayPaymentId = payment.gatewayPaymentId ?? payload.gatewayPaymentId
   260→  let captureResult
   261→  try {
   262→    captureResult = await captureRazorpayPayment(
   263→      gatewayPaymentId,
   264→      payment.amount,
   265→      payment.currency,
   266→    )
   267→  } catch (captureError) {
   268→    // Capture call failed (network error, gateway 5xx, etc.) — record the
   269→    // failure on the Payment row, then rethrow so the publisher's existing
   270→    // retry/backoff/FAILED logic drives the outbox event's lifecycle.
   271→    // Payment.status is left as CAPTURE_PENDING (capture may succeed on retry).
   272→    const errorMsg = (captureError as Error).message || 'unknown-capture-error'
   273→    await db.payment.update({
   274→      where: { id: payment.id },
   275→      data: {
   276→        retryCount: { increment: 1 },
   277→        failureReason: `Capture failed: ${errorMsg}`,
   278→      },
   279→    })
   280→    await log({
   281→      level: 'warn',
   282→      message: 'capture-call-failed',
   283→      eventId: event.eventId,
   284→      eventType: 'PAYMENT_CAPTURE_REQUESTED',
   285→      paymentId: payment.id,
   286→      attempt: event.attempts + 1,
   287→      error: errorMsg,
   288→    })
   289→    throw captureError
   290→  }
   291→
   292→  // If the gateway returned captured=false (decline), treat as failure.
   293→  if (!captureResult.captured) {
   294→    await db.payment.update({
   295→      where: { id: payment.id },
   296→      data: {
   297→        retryCount: { increment: 1 },
   298→        failureReason: 'Gateway declined capture (captured=false)',
   299→      },
   300→    })
   301→    await log({
   302→      level: 'warn',
   303→      message: 'capture-declined-by-gateway',
   304→      eventId: event.eventId,
   305→      eventType: 'PAYMENT_CAPTURE_REQUESTED',
   306→      paymentId: payment.id,
   307→      attempt: event.attempts + 1,
   308→    })
   309→    throw new Error('Gateway declined capture (captured=false)')
   310→  }
   311→
   312→  // 5. Capture succeeded — open a NEW transaction and atomically:
   313→  //    (a) Update Payment status CAPTURE_PENDING → CAPTURED + capturedAt (race-safe)
   314→  //    (b) AuditLog (PAYMENT_CAPTURED)
   315→  //    (c) Mark outbox event PUBLISHED
   316→  //    All three commit in the SAME txn — no half-states.
   317→  //
   318→  //    NOTE: this txn does NOT include captureRazorpayPayment() — that call
   319→  //    already happened above (outside the txn). This is the Wave-4 4c safety
   320→  //    improvement: a P2034 retry would re-run only the DB writes, NOT the
   321→  //    capture HTTP call (no double-charge risk).
   322→  await db.$transaction(async (tx) => {
   323→    // Race-safe: only update if status is still CAPTURE_PENDING.
   324→    // If a concurrent path (e.g., webhook) already captured, count === 0 —
   325→    // we still mark the outbox PUBLISHED (capture command effectively done)
   326→    // but skip writing a duplicate AuditLog.
   327→    const updated = await tx.payment.updateMany({
   328→      where: {
   329→        id: payment.id,
   330→        status: 'CAPTURE_PENDING',
   331→      },
   332→      data: {
   333→        status: 'CAPTURED',
   334→        capturedAt: new Date(),
   335→        version: { increment: 1 },
   336→      },
   337→    })
   338→
   339→    if (updated.count > 0) {
   340→      // AuditLog — PAYMENT_CAPTURED (publisher-driven).
   341→      // Distinct from WEBHOOK_PAYMENT_CAPTURED (webhook-driven) so audit
   342→      // consumers can distinguish capture-confirmation paths.
   343→      await tx.auditLog.create({
   344→        data: {
   345→          actorId: null,
   346→          actorRole: 'SYSTEM',
   347→          action: 'PAYMENT_CAPTURED',
   348→          metadata: JSON.stringify({
   349→            paymentId: payment.id,
   350→            orderId: payload.orderId,
   351→            gatewayPaymentId: captureResult.gatewayPaymentId,
   352→            amount: payment.amount,
   353→            source: 'outbox-publisher',
   354→            outboxEventId: event.eventId,
   355→          }),
   356→        },
   357→      })
   358→    }
   359→
   360→    // Mark outbox event PUBLISHED (whether or not we updated Payment —
   361→    // the capture command's business effect has been achieved either way).
   362→    await tx.outbox.update({
   363→      where: { id: event.id },
   364→      data: {
   365→        status: 'PUBLISHED',
   366→        publishedAt: new Date(),
   367→        claimedAt: null,
   368→        claimUntil: null,
   369→        workerId: null,
   370→      },
   371→    })
   372→  })
   373→
   374→  await log({
   375→    level: 'info',
   376→    message: 'capture-completed',
   377→    eventId: event.eventId,
   378→    eventType: 'PAYMENT_CAPTURE_REQUESTED',
   379→    paymentId: payment.id,
   380→    orderId: payload.orderId,
   381→  })
   382→}
   383→
   384→// ----------------------------------------------------------------------------
   385→// Wave-5 Sub-Wave 5a — PAYMENT_REFUND_REQUESTED handler
   386→// ----------------------------------------------------------------------------
   387→// Command-event handler: consumes a PAYMENT_REFUND_REQUESTED outbox row and
   388→// performs the actual Razorpay refund (safely OUTSIDE any transaction body).
   389→//
   390→// Mirrors `processPaymentCaptureRequested()` (Wave-4 4c Phase 2) exactly:
   391→//   1. Parse payload (written by refund route's enqueueOutboxEvent call).
   392→//      aggregateId = refund.id (per the refund route's enqueueOutboxEvent
   393→//      ({ aggregateType: 'Refund', aggregateId: refund.id, ... })).
   394→//   2. Read Refund row by id.
   395→//   3. Idempotency: if Refund.status === 'REFUNDED', mark outbox PUBLISHED
   396→//      and exit (no duplicate external refund).
   397→//   4. If Refund.status === 'FAILED', mark outbox PUBLISHED (terminal state).
   398→//   5. If Refund.status !== 'REFUND_PENDING', throw (outbox retry/backoff drives).
   399→//   6. Call refundRazorpayPayment() OUTSIDE any txn (Wave-4 4c safety property).
   400→//   7. On refund-call failure / declined: increment Refund.version + set
   401→//      failureReason, leave status REFUND_PENDING, throw (publisher catch
   402→//      handles retry/backoff/FAILED lifecycle).
   403→//   8. On success: NEW db.$transaction() atomically commits:
   404→//        (a) Refund REFUND_PENDING → REFUNDED + refundedAt + gatewayRefundId
   405→//            (race-safe conditional updateMany WHERE status='REFUND_PENDING').
   406→//        (b) If full refund AND Payment.status==='CAPTURED': transition Payment
   407→//            CAPTURED → REFUNDED (race-safe conditional updateMany).
   408→//        (c) AuditLog (PAYMENT_REFUNDED) if either Refund or Payment was updated.
   409→//        (d) Outbox.status=PUBLISHED (always — refund command effect achieved).
   410→//
   411→// The success txn does NOT include refundRazorpayPayment() — that call already
   412→// happened above (outside the txn). This is the Wave-4 4c safety improvement:
   413→// a P2034 retry would re-run only the DB writes, NOT the refund HTTP call
   414→// (no double-refund risk).
   415→// ----------------------------------------------------------------------------
   416→
   417→interface RefundRequestedPayload {
   418→  refundId: string
   419→  paymentId: string
   420→  orderId: string
   421→  gatewayPaymentId: string | null
   422→  amount: number
   423→  currency: string
   424→  fullRefund: boolean
   425→}
   426→
   427→async function processPaymentRefundRequested(event: {
   428→  id: string
   429→  eventId: string
   430→  aggregateId: string
   431→  payload: string
   432→  attempts: number
   433→}): Promise<void> {
   434→  // Parse payload (written by refund route's enqueueOutboxEvent call)
   435→  let payload: RefundRequestedPayload
   436→  try {
   437→    payload = JSON.parse(event.payload) as RefundRequestedPayload
   438→  } catch {
   439→    // Malformed payload — non-retryable; mark outbox PUBLISHED to stop retries.
   440→    await db.outbox.update({
   441→      where: { id: event.id },
   442→      data: {
   443→        status: 'PUBLISHED',
   444→        publishedAt: new Date(),
   445→        claimedAt: null,
   446→        claimUntil: null,
   447→        workerId: null,
   448→        lastError: 'malformed-payload-marked-published',
   449→      },
   450→    })
   451→    await log({
   452→      level: 'error',
   453→      message: 'refund-payload-malformed',
   454→      eventId: event.eventId,
   455→      eventType: 'PAYMENT_REFUND_REQUESTED',
   456→    })
   457→    return
   458→  }
   459→
   460→  // 1. Read the Refund (aggregateId = refundId per the refund route's
   461→  //    enqueueOutboxEvent({ aggregateType: 'Refund', aggregateId: refund.id, ... }))
   462→  const refund = await db.refund.findUnique({
   463→    where: { id: event.aggregateId },
   464→    select: {
   465→      id: true,
   466→      paymentId: true,
   467→      amount: true,
   468→      currency: true,
   469→      status: true,
   470→      version: true,
   471→    },
   472→  })
   473→
   474→  if (!refund) {
   475→    // Refund row missing — non-retryable; mark outbox PUBLISHED to stop
   476→    // retrying. (By design this is impossible: the refund route writes Refund
   477→    // + Outbox in the SAME txn, so an outbox row can't exist without a Refund.)
   478→    await db.outbox.update({
   479→      where: { id: event.id },
   480→      data: {
   481→        status: 'PUBLISHED',
   482→        publishedAt: new Date(),
   483→        claimedAt: null,
   484→        claimUntil: null,
   485→        workerId: null,
   486→        lastError: `refund-not-found:aggregateId=${event.aggregateId}`,
   487→      },
   488→    })
   489→    await log({
   490→      level: 'error',
   491→      message: 'refund-not-found',
   492→      eventId: event.eventId,
   493→      eventType: 'PAYMENT_REFUND_REQUESTED',
   494→      refundId: event.aggregateId,
   495→    })
   496→    return
   497→  }
   498→
   499→  // 2. Idempotency: if already REFUNDED (e.g., a webhook raced ahead and
   500→  //    processed the refund first, or a prior publisher invocation refunded
   501→  //    but failed to mark the outbox PUBLISHED), the refund command has
   502→  //    already succeeded. Mark the outbox event PUBLISHED and exit.
   503→  if (refund.status === 'REFUNDED') {
   504→    await db.outbox.update({
   505→      where: { id: event.id },
   506→      data: {
   507→        status: 'PUBLISHED',
   508→        publishedAt: new Date(),
   509→        claimedAt: null,
   510→        claimUntil: null,
   511→        workerId: null,
   512→      },
   513→    })
   514→    await log({
   515→      level: 'info',
   516→      message: 'refund-already-refunded-idempotent',
   517→      eventId: event.eventId,
   518→      eventType: 'PAYMENT_REFUND_REQUESTED',
   519→      refundId: refund.id,
   520→      paymentId: refund.paymentId,
   521→    })
   522→    return
   523→  }
   524→
   525→  // 3. Terminal FAILED state: refund exhausted retries earlier and was marked
   526→  //    FAILED. Mark outbox PUBLISHED to stop further retry attempts (manual
   527→  //    intervention would have created a NEW Refund with a new idempotency key).
   528→  if (refund.status === 'FAILED') {
   529→    await db.outbox.update({
   530→      where: { id: event.id },
   531→      data: {
   532→        status: 'PUBLISHED',
   533→        publishedAt: new Date(),
   534→        claimedAt: null,
   535→        claimUntil: null,
   536→        workerId: null,
   537→        lastError: 'refund-terminal-failed-marked-published',
   538→      },
   539→    })
   540→    await log({
   541→      level: 'warn',
   542→      message: 'refund-terminal-failed',
   543→      eventId: event.eventId,
   544→      eventType: 'PAYMENT_REFUND_REQUESTED',
   545→      refundId: refund.id,
   546→      paymentId: refund.paymentId,
   547→    })
   548→    return
   549→  }
   550→
   551→  // 4. If Refund is in a non-pending state (shouldn't happen since REFUNDED
   552→  //    and FAILED are handled above, but defensive), do NOT attempt refund.
   553→  //    Throw so the publisher's existing retry path handles it.
   554→  if (refund.status !== 'REFUND_PENDING') {
   555→    throw new Error(
   556→      `Refund ${refund.id} status is ${refund.status} (expected REFUND_PENDING) — refund skipped`,
   557→    )
   558→  }
   559→
   560→  // 5. Resolve the Razorpay payment ID. Prefer the payload's gatewayPaymentId
   561→  //    (written by the refund route from Payment.gatewayPaymentId). Fall back
   562→  //    to reading Payment.gatewayPaymentId. If neither is set (impossible for a
   563→  //    CAPTURED payment, but defensive), throw.
   564→  let gatewayPaymentId = payload.gatewayPaymentId
   565→  if (!gatewayPaymentId) {
   566→    const payment = await db.payment.findUnique({
   567→      where: { id: refund.paymentId },
   568→      select: { gatewayPaymentId: true, status: true },
   569→    })
   570→    if (!payment) {
   571→      throw new Error(`Payment ${refund.paymentId} not found for refund ${refund.id}`)
   572→    }
   573→    gatewayPaymentId = payment.gatewayPaymentId
   574→  }
   575→  if (!gatewayPaymentId) {
   576→    throw new Error(
   577→      `Cannot refund: Payment ${refund.paymentId} has no gatewayPaymentId (refund ${refund.id})`,
   578→    )
   579→  }
   580→
   581→  // 6. Call refundRazorpayPayment() OUTSIDE any transaction body.
   582→  //    This is the Wave-4 4c / Wave-5 5a safety improvement: if the
   583→  //    success-path txn retries (P2034 conflict), this call is NOT re-executed.
   584→  //    In demo mode (realPayments=false), this returns mock success immediately.
   585→  let refundResult
   586→  try {
   587→    refundResult = await refundRazorpayPayment(
   588→      gatewayPaymentId,
   589→      refund.amount,
   590→      refund.currency,
   591→    )
   592→  } catch (refundError) {
   593→    // Refund call failed (network error, gateway 5xx, etc.) — record the
   594→    // failure on the Refund row, then rethrow so the publisher's existing
   595→    // retry/backoff/FAILED logic drives the outbox event's lifecycle.
   596→    // Refund.status is left as REFUND_PENDING (refund may succeed on retry).
   597→    const errorMsg = (refundError as Error).message || 'unknown-refund-error'
   598→    await db.refund.update({
   599→      where: { id: refund.id },
   600→      data: {
   601→        version: { increment: 1 },
   602→        failureReason: `Refund failed: ${errorMsg}`,
   603→      },
   604→    })
   605→    await log({
   606→      level: 'warn',
   607→      message: 'refund-call-failed',
   608→      eventId: event.eventId,
   609→      eventType: 'PAYMENT_REFUND_REQUESTED',
   610→      refundId: refund.id,
   611→      paymentId: refund.paymentId,
   612→      attempt: event.attempts + 1,
   613→      error: errorMsg,
   614→    })
   615→    throw refundError
   616→  }
   617→
   618→  // If the gateway returned refunded=false (decline), treat as failure.
   619→  if (!refundResult.refunded) {
   620→    await db.refund.update({
   621→      where: { id: refund.id },
   622→      data: {
   623→        version: { increment: 1 },
   624→        failureReason: 'Gateway declined refund (refunded=false)',
   625→      },
   626→    })
   627→    await log({
   628→      level: 'warn',
   629→      message: 'refund-declined-by-gateway',
   630→      eventId: event.eventId,
   631→      eventType: 'PAYMENT_REFUND_REQUESTED',
   632→      refundId: refund.id,
   633→      paymentId: refund.paymentId,
   634→      attempt: event.attempts + 1,
   635→    })
   636→    throw new Error('Gateway declined refund (refunded=false)')
   637→  }
   638→
   639→  // 7. Refund succeeded — open a NEW transaction and atomically:
   640→  //    (a) Update Refund status REFUND_PENDING → REFUNDED + refundedAt +
   641→  //        gatewayRefundId (race-safe conditional updateMany).
   642→  //    (b) If full refund AND Payment.status==='CAPTURED': transition Payment
   643→  //        CAPTURED → REFUNDED (race-safe conditional updateMany).
   644→  //    (c) AuditLog (PAYMENT_REFUNDED) if either Refund or Payment was updated.
   645→  //    (d) Outbox.status=PUBLISHED (always — refund command effect achieved).
   646→  //    All commit in the SAME txn — no half-states.
   647→  //
   648→  //    NOTE: this txn does NOT include refundRazorpayPayment() — that call
   649→  //    already happened above (outside the txn). This is the Wave-4 4c safety
   650→  //    improvement: a P2034 retry would re-run only the DB writes, NOT the
   651→  //    refund HTTP call (no double-refund risk).
   652→  await db.$transaction(async (tx) => {
   653→    // Race-safe: only update if Refund.status is still REFUND_PENDING.
   654→    // If a concurrent path (e.g., webhook) already marked REFUNDED, count===0
   655→    // — we still mark the outbox PUBLISHED (refund command effectively done)
   656→    // but skip writing a duplicate AuditLog.
   657→    const refundUpdated = await tx.refund.updateMany({
   658→      where: {
   659→        id: refund.id,
   660→        status: 'REFUND_PENDING',
   661→      },
   662→      data: {
   663→        status: 'REFUNDED',
   664→        refundedAt: new Date(),
   665→        gatewayRefundId: refundResult.gatewayRefundId,
   666→        version: { increment: 1 },
   667→      },
   668→    })
   669→
   670→    // Only transition Payment → REFUNDED for a FULL refund (partial refunds
   671→    // leave Payment as CAPTURED — the payment is partially refunded but
   672→    // still considered "captured" for fulfillment purposes). Race-safe
   673→    // conditional updateMany WHERE status='CAPTURED'.
   674→    let paymentUpdated = { count: 0 }
   675→    if (payload.fullRefund) {
   676→      paymentUpdated = await tx.payment.updateMany({
   677→        where: {
   678→          id: refund.paymentId,
   679→          status: 'CAPTURED',
   680→        },
   681→        data: {
   682→          status: 'REFUNDED',
   683→          version: { increment: 1 },
   684→        },
   685→      })
   686→    }
   687→
   688→    if (refundUpdated.count > 0 || paymentUpdated.count > 0) {
   689→      // AuditLog — PAYMENT_REFUNDED (publisher-driven).
   690→      await tx.auditLog.create({
   691→        data: {
   692→          actorId: null,
   693→          actorRole: 'SYSTEM',
   694→          action: 'PAYMENT_REFUNDED',
   695→          metadata: JSON.stringify({
   696→            refundId: refund.id,
   697→            paymentId: refund.paymentId,
   698→            orderId: payload.orderId,
   699→            gatewayRefundId: refundResult.gatewayRefundId,
   700→            amount: refund.amount,
   701→            fullRefund: payload.fullRefund,
   702→            source: 'outbox-publisher',
   703→            outboxEventId: event.eventId,
   704→          }),
   705→        },
   706→      })
   707→    }
   708→
   709→    // Mark outbox event PUBLISHED (whether or not we updated Refund/Payment —
   710→    // the refund command's business effect has been achieved either way).
   711→    await tx.outbox.update({
   712→      where: { id: event.id },
   713→      data: {
   714→        status: 'PUBLISHED',
   715→        publishedAt: new Date(),
   716→        claimedAt: null,
   717→        claimUntil: null,
   718→        workerId: null,
   719→      },
   720→    })
   721→  })
   722→
   723→  await log({
   724→    level: 'info',
   725→    message: 'refund-completed',
   726→    eventId: event.eventId,
   727→    eventType: 'PAYMENT_REFUND_REQUESTED',
   728→    refundId: refund.id,
   729→    paymentId: refund.paymentId,
   730→    orderId: payload.orderId,
   731→  })
   732→}

   741→/**
   742→ * Main publisher loop — claims + publishes + marks events.
   743→ * Called by Vercel Cron (or manual trigger).
   744→ */
   745→async function publishPendingEvents(): Promise<{
   746→  claimed: number
   747→  published: number
   748→  failed: number
   749→  recovered: number
   750→  errors: number
   751→}> {
   752→  const workerId = `worker-${process.pid}-${Date.now()}`
   753→  const result = { claimed: 0, published: 0, failed: 0, recovered: 0, errors: 0 }
   754→
   755→  // Step 1: Recover stale CLAIMED events (lease expired)
   756→  const now = new Date()
   757→  const staleRecovery = await db.outbox.updateMany({
   758→    where: {
   759→      status: 'CLAIMED',
   760→      claimUntil: { lt: now },
   761→    },
   762→    data: {
   763→      status: 'PENDING',
   764→      claimedAt: null,
   765→      claimUntil: null,
   766→      workerId: null,
   767→    },
   768→  })
   769→  result.recovered = staleRecovery.count
   770→  if (result.recovered > 0) {
   771→    await log({ level: 'warn', message: 'recovered-stale-claimed-events', workerId, count: result.recovered })
   772→  }
   773→
   774→  // Step 2: Claim PENDING events (atomic — WHERE status='PENDING')
   775→  const claimResult = await db.outbox.updateMany({
   776→    where: {
   777→      status: 'PENDING',
   778→    },
   779→    data: {
   780→      status: 'CLAIMED',
   781→      claimedAt: now,
   782→      claimUntil: new Date(now.getTime() + LEASE_DURATION_MS),
   783→      workerId,
   784→    },
   785→    // Note: Prisma updateMany doesn't support LIMIT directly, but we
   786→    // fetch only BATCH_SIZE events in the next query
   787→  })
   788→  result.claimed = claimResult.count
   789→
   790→  if (result.claimed === 0) {
   791→    return result // nothing to do
   792→  }
   793→
   794→  // Step 3: Fetch claimed events for this worker
   795→  const claimedEvents = await db.outbox.findMany({
   796→    where: {
   797→      status: 'CLAIMED',
   798→      workerId,
   799→    },
   800→    orderBy: { createdAt: 'asc' },
   801→    take: BATCH_SIZE,
   802→  })
   803→
   804→  // Step 4: Publish each event
   805→  // PUBLISHED means: successful transport handoff to the consumer.
   806→  // If transport fails, the event goes to retry (PENDING with incremented attempts)
   807→  // or FAILED (max retries). PUBLISHED is NOT set on failure.
   808→  for (const event of claimedEvents) {
   809→    try {
   810→      // Wave-4 4c Phase 2 / Wave-5 5a: command events trigger a business
   811→      // operation (capture / refund payment) rather than a transport handoff.
   812→      // The handler fully owns its own outbox state transitions (PUBLISHED on
   813→      // success, PENDING+attempts on failure via thrown error → publisher's
   814→      // existing catch block handles retry/backoff). It must NOT fall through
   815→      // to the EVENT_TYPE_TO_SOCKET lookup below (which would throw "Unknown
   816→      // event type" since command events are intentionally not in that map).
   817→      if (COMMAND_EVENT_TYPES.has(event.eventType)) {
   818→        if (event.eventType === 'PAYMENT_CAPTURE_REQUESTED') {
   819→          await processPaymentCaptureRequested(event)
   820→        } else if (event.eventType === 'PAYMENT_REFUND_REQUESTED') {
   821→          await processPaymentRefundRequested(event)
   822→        } else {
   823→          // Defensive: should be unreachable (COMMAND_EVENT_TYPES membership is
   824→          // checked above). Throw so the catch block schedules a retry / marks
   825→          // FAILED if the condition persists.
   826→          throw new Error(`No handler registered for command event type: ${event.eventType}`)
   827→        }
   828→        result.published++
   829→        continue
   830→      }
   831→
   832→      const socketEventName = EVENT_TYPE_TO_SOCKET[event.eventType]
   833→      if (!socketEventName) {
   834→        throw new Error(`Unknown event type: ${event.eventType}`)
   835→      }
   836→
   837→      const payload = JSON.parse(event.payload)
   838→
   839→      // Transport: deliver the event via HTTP or Socket.io
   840→      if (TRANSPORT_MODE === 'http') {
   841→        // HTTP mode: POST to consumer endpoint
   842→        if (!CONSUMER_URL) {
   843→          throw new Error('CONSUMER_URL not set for HTTP transport mode')
   844→        }
   845→
   846→        const consumerEndpoint = `${CONSUMER_URL}/api/test/consume-event`
   847→        const response = await fetch(consumerEndpoint, {
   848→          method: 'POST',
   849→          headers: { 'Content-Type': 'application/json' },
   850→          body: JSON.stringify({
   851→            eventId: event.eventId,
   852→            eventType: event.eventType,
   853→            payload,
   854→          }),
   855→          signal: AbortSignal.timeout(5000),
   856→        })
   857→
   858→        if (!response.ok) {
   859→          const errBody = await response.text().catch(() => 'unknown')
   860→          throw new Error(`Consumer returned HTTP ${response.status}: ${errBody}`)
   861→        }
   862→
   863→        const consumerResult = await response.json()
   864→        await log({ level: 'info', message: 'event-delivered-via-http', eventId: event.eventId, eventType: event.eventType, workerId, consumerProcessed: consumerResult.processed })
   865→      } else {
   866→        // Socket.io mode: emit via realtime service
   867→        const sock = getRealtimeSocket()
   868→        if (!sock || !sock.connected) {
   869→          throw new Error('Realtime service not connected — transport failed')
   870→        }
   871→        sock.emit(socketEventName, payload)
   872→        await log({ level: 'info', message: 'event-published-via-socketio', eventId: event.eventId, eventType: event.eventType, workerId })
   873→      }
   874→
   875→      // Mark as PUBLISHED ONLY after successful transport
   876→      await db.outbox.update({
   877→        where: { id: event.id },
   878→        data: {
   879→          status: 'PUBLISHED',
   880→          publishedAt: new Date(),
   881→          claimedAt: null,
   882→          claimUntil: null,
   883→          workerId: null,
   884→        },
   885→      })
   886→      result.published++
   887→    } catch (error) {
   888→      const errorMsg = (error as Error).message
   889→      result.errors++
   890→
   891→      const newAttempts = event.attempts + 1
   892→
   893→      if (newAttempts >= MAX_RETRIES) {
   894→        // Max retries exhausted → FAILED
   895→        await db.outbox.update({
   896→          where: { id: event.id },
   897→          data: {
   898→            status: 'FAILED',
   899→            attempts: newAttempts,
   900→            lastError: errorMsg,
   901→            claimedAt: null,
   902→            claimUntil: null,
   903→            workerId: null,
   904→          },
   905→        })
   906→        result.failed++
   907→        await log({ level: 'error', message: 'event-failed-max-retries', eventId: event.eventId, eventType: event.eventType, workerId, attempt: newAttempts, error: errorMsg })
   908→      } else {
   909→        // Retry: back to PENDING (will be re-claimed on next invocation with backoff)
   910→        // We use claimUntil to enforce backoff: set it to now + backoff
   911→        const backoffMs = BACKOFF_SCHEDULE_MS[Math.min(newAttempts - 1, BACKOFF_SCHEDULE_MS.length - 1)]
   912→        await db.outbox.update({
   913→          where: { id: event.id },
   914→          data: {
   915→            status: 'PENDING',
   916→            attempts: newAttempts,
   917→            lastError: errorMsg,
   918→            claimedAt: null,
   919→            claimUntil: new Date(Date.now() + backoffMs), // enforce backoff
   920→            workerId: null,
   921→          },
   922→        })
   923→        await log({ level: 'warn', message: 'event-retry-scheduled', eventId: event.eventId, eventType: event.eventType, workerId, attempt: newAttempts, error: errorMsg })
   924→      }
   925→    }
   926→  }
   927→
   928→  return result
   929→}

   931→// Health check + manual trigger endpoint
   932→const server = Bun.serve({
   933→  port: PORT,
   934→  async fetch(req) {
   935→    const url = new URL(req.url)
   936→
   937→    if (url.pathname === '/') {
   938→      return Response.json({ status: 'ok', service: 'outbox-publisher', port: PORT })
   939→    }
   940→
   941→    if (url.pathname === '/trigger') {
   942→      // Manual trigger — runs publishPendingEvents once
   943→      try {
   944→        const result = await publishPendingEvents()
   945→        return Response.json({ ok: true, result })
   946→      } catch (error) {
   947→        return Response.json({ ok: false, error: (error as Error).message }, { status: 500 })
   948→      }
   949→    }
   950→
   951→    if (url.pathname === '/lag') {
   952→      // Check outbox lag (age of oldest PENDING event)
   953→      const oldestPending = await db.outbox.findFirst({
   954→        where: { status: 'PENDING' },
   955→        orderBy: { createdAt: 'asc' },
   956→        select: { createdAt: true, eventId: true, eventType: true },
   957→      })
   958→
   959→      if (!oldestPending) {
   960→        return Response.json({ lagSeconds: 0, oldestEventId: null })
   961→      }
   962→
   963→      const lagSeconds = Math.floor((Date.now() - oldestPending.createdAt.getTime()) / 1000)
   964→      return Response.json({
   965→        lagSeconds,
   966→        oldestEventId: oldestPending.eventId,
   967→        oldestEventType: oldestPending.eventType,
   968→      })
   969→    }
   970→
   971→    return new Response('Not found', { status: 404 })
   972→  },
   973→})
