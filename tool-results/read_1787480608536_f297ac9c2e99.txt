     5→// FriendsScreen — Friends sub-tab content for the Social screen (Wave 6 Task 6B).
     6→//
     7→// Per blueprint §18 SOCIAL GRAPH + DESIGN_SYSTEM.md §5.2.7 (Friends list):
     8→//  - Search bar (debounced 250ms) → GET /api/social/search?q=
     9→//  - Pending requests section:
    10→//      • Incoming (status === 'PENDING_RECEIVED') — Accept / Reject buttons
    11→//      • Outgoing (status === 'PENDING_SENT') — "Pending" label
    12→//  - Current friends list (status === 'ACCEPTED'):
    13→//      avatar + name + phone + "Message" (placeholder toast) + "Unfriend" button
    14→//  - Search results: each row shows "Add friend" button (calls sendRequest).
    15→//    Already-connected users are excluded client-side.
    16→//
    17→// State sourcing:
    18→//  - Reads `connections` + mutation methods from `useSocial` (Task 1C owns the
    19→//    store — we CALL its methods, never modify it).
    20→//  - Search results are fetched locally (the social-store has no `search`
    21→//    method by design — search is a transient, non-cached query).
    22→//
    23→// Governance (Task 6B):
    24→//  - Does NOT touch any API route (Task 6A owns /api/social/**).
    25→//  - Does NOT touch social-store.ts (Task 1C owns).
    26→//  - Does NOT touch consumer-view.tsx (Task 3A owns the tab routing).
    27→//  - Resilient to missing endpoints — surfaces a soft "Couldn't load" toast.

    93→function initials(name: string): string {
    94→  return name
    95→    .split(' ')
    96→    .map((p) => p[0])
    97→    .filter(Boolean)
    98→    .slice(0, 2)
    99→    .join('')
   100→    .toUpperCase()
   101→}

   103→// ─────────────────────────────────────────────────────────────────────────────
   104→// FriendsScreen
   105→// ─────────────────────────────────────────────────────────────────────────────
   106→
   107→export function FriendsScreen({ className }: FriendsScreenProps) {
   108→  const prefersReduced = useReducedMotion()
   109→  const { toast } = useToast()
   110→
   111→  // ── social-store hooks (Task 1C) ─────────────────────────────────────────
   112→  const connections = useSocial((s) => s.connections)
   113→  const isLoading = useSocial((s) => s.isLoading)
   114→  const refresh = useSocial((s) => s.refresh)
   115→  const sendRequest = useSocial((s) => s.sendRequest)
   116→  const acceptRequest = useSocial((s) => s.acceptRequest)
   117→  const declineRequest = useSocial((s) => s.declineRequest)
   118→  const unfollow = useSocial((s) => s.unfollow)
   119→
   120→  // ── Derived slices ─────────────────────────────────────────────────────────
   121→  const pendingIn = React.useMemo(
   122→    // S1 Reconstruction: server returns 'PENDING_RECEIVED' (they sent to me)
   123→    () => connections.filter((c) => c.status === 'PENDING_RECEIVED'),
   124→    [connections],
   125→  )
   126→  const pendingOut = React.useMemo(
   127→    // S1 Reconstruction: server returns 'PENDING_SENT' (I sent to them)
   128→    () => connections.filter((c) => c.status === 'PENDING_SENT'),
   129→    [connections],
   130→  )
   131→  const friends = React.useMemo(
   132→    () =>
   133→      connections
   134→        .filter((c) => c.status === 'ACCEPTED')
   135→        // newest accepted first (acceptedAt ?? createdAt fallback)
   136→        .sort((a, b) => {
   137→          const at = (t?: string | null) => (t ? new Date(t).getTime() : 0)
   138→          return at(b.acceptedAt) - at(a.acceptedAt)
   139→        }),
   140→    [connections],
   141→  )
   142→
   143→  // Set of friend IDs (any status) for client-side exclusion of search results.
   144→  const connectedIds = React.useMemo(
   145→    () => new Set(connections.map((c) => c.userId)),
   146→    [connections],
   147→  )
   148→
   149→  // ── Search state (local — not in social-store by design) ───────────────────
   150→  const [query, setQuery] = React.useState('')
   151→  const [debounced, setDebounced] = React.useState('')
   152→  const [results, setResults] = React.useState<SocialSearchUser[]>([])
   153→  const [searching, setSearching] = React.useState(false)
   154→  const [searchError, setSearchError] = React.useState<string | null>(null)
   155→  // Track outgoing requests that succeeded locally so we can flip the row's
   156→  // button to "Pending" without waiting for a refresh.
   157→  const [pendingSentIds, setPendingSentIds] = React.useState<Set<string>>(new Set())
   158→
   159→  // 250ms debounce — only triggers a fetch when query length ≥ 2.
   160→  React.useEffect(() => {
   161→    const q = query.trim()
   162→    if (q === debounced) return
   163→    const t = setTimeout(() => setDebounced(q), 250)
   164→    return () => clearTimeout(t)
   165→  }, [query, debounced])
   166→
   167→  // Fetch search results whenever the debounced query changes.
   168→  React.useEffect(() => {
   169→    const q = debounced.trim()
   170→    if (q.length < 2) {
   171→      setResults([])
   172→      setSearchError(null)
   173→      setSearching(false)
   174→      return
   175→    }
   176→    let cancelled = false
   177→    setSearching(true)
   178→    setSearchError(null)
   179→    fetch(`/api/social/search?q=${encodeURIComponent(q)}`, {
   180→      headers: { 'Content-Type': 'application/json' },
   181→    })
   182→      .then(async (res) => {
   183→        if (!res.ok) {
   184→          const body = await res.json().catch(() => ({}))
   185→          throw new Error(body?.error || `Search failed (${res.status})`)
   186→        }
   187→        const data = (await res.json()) as { users?: SocialSearchUser[] }
   188→        if (!cancelled) {
   189→          setResults(data.users ?? [])
   190→        }
   191→      })
   192→      .catch((err) => {
   193→        if (!cancelled) {
   194→          setSearchError(err instanceof Error ? err.message : 'Search failed')
   195→          setResults([])
   196→        }
   197→      })
   198→      .finally(() => {
   199→        if (!cancelled) setSearching(false)
   200→      })
   201→    return () => {
   202→      cancelled = true
   203→    }
   204→  }, [debounced])
   205→
   206→  // ── Action handlers ─────────────────────────────────────────────────────────
   207→  async function handleAccept(conn: SocialConnection) {
   208→    try {
   209→      await acceptRequest(conn.id)
   210→      toast({
   211→        title: 'Friend request accepted',
   212→        description: `${conn.name} is now your friend.`,
   213→      })
   214→    } catch (e) {
   215→      toast({
   216→        title: 'Could not accept request',
   217→        description: (e as Error).message,
   218→        variant: 'destructive',
   219→      })
   220→    }
   221→  }
   222→
   223→  async function handleReject(conn: SocialConnection) {
   224→    try {
   225→      await declineRequest(conn.id)
   226→      toast({
   227→        title: 'Request declined',
   228→        description: `${conn.name}'s request was dismissed.`,
   229→      })
   230→    } catch (e) {
   231→      toast({
   232→        title: 'Could not decline request',
   233→        description: (e as Error).message,
   234→        variant: 'destructive',
   235→      })
   236→    }
   237→  }
   238→
   239→  async function handleUnfriend(conn: SocialConnection) {
   240→    // Optimistic confirmation — friend removal is destructive.
   241→    const ok = window.confirm(
   242→      `Unfriend ${conn.name}? They won't be notified, but you'll need to send a new request to reconnect.`,
   243→    )
   244→    if (!ok) return
   245→    try {
   246→      await unfollow(conn.userId)
   247→      toast({
   248→        title: 'Unfriended',
   249→        description: `${conn.name} is no longer your friend.`,
   250→      })
   251→    } catch (e) {
   252→      toast({
   253→        title: 'Could not unfriend',
   254→        description: (e as Error).message,
   255→        variant: 'destructive',
   256→      })
   257→    }
   258→  }
   259→
   260→  async function handleSendMessage(conn: SocialConnection) {
   261→    // Placeholder — messaging is on the Wave 8+ roadmap (blueprint §22).
   262→    toast({
   263→      title: 'Messaging coming soon',
   264→      description: `You'll be able to message ${conn.name} in a future release.`,
   265→    })
   266→  }
   267→
   268→  async function handleAddFriend(user: SocialSearchUser) {
   269→    try {
   270→      await sendRequest(user.id)
   271→      setPendingSentIds((s) => new Set(s).add(user.id))
   272→      toast({
   273→        title: 'Friend request sent',
   274→        description: `Waiting for ${user.name} to accept.`,
   275→      })
   276→    } catch (e) {
   277→      toast({
   278→        title: 'Could not send request',
   279→        description: (e as Error).message,
   280→        variant: 'destructive',
   281→      })
   282→    }
   283→  }
   284→
   285→  function handleRefresh() {
   286→    refresh().catch(() => {
   287→      /* best-effort — store surfaces its own error */
   288→    })
   289→  }
   290→
   291→  const hasSearchQuery = debounced.trim().length >= 2
   292→  const filteredResults = React.useMemo(
   293→    () => results.filter((u) => !connectedIds.has(u.id) && !pendingSentIds.has(u.id)),
   294→    [results, connectedIds, pendingSentIds],
   295→  )
   296→
   297→  // ── Loading skeleton ─────────────────────────────────────────────────────────
   298→  if (isLoading && connections.length === 0) {
   299→    return (
   300→      <div className="space-y-3" role="status" aria-label="Loading friends">
   301→        <span className="sr-only">Loading your friends…</span>
   302→        {Array.from({ length: 4 }).map((_, i) => (
   303→          <SocialFeedSkeleton key={i} />
   304→        ))}
   305→      </div>
   306→    )
   307→  }
   308→
   309→  return (
   310→    <div className={cn('space-y-5', className)}>
   311→      {/* ── Search bar ─────────────────────────────────────────────────────── */}
   312→      <div className="relative">
   313→        <Search
   314→          className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   315→          aria-hidden="true"
   316→        />
   317→        <Input
   318→          type="search"
   319→          value={query}
   320→          onChange={(e) => setQuery(e.target.value)}
   321→          placeholder="Search by name or phone…"
   322→          aria-label="Search for friends"
   323→          className="h-11 rounded-xl pl-9 pr-9"
   324→        />
   325→        {query && (
   326→          <button
   327→            type="button"
   328→            onClick={() => setQuery('')}
   329→            aria-label="Clear search"
   330→            className="snak-focus-ring absolute right-2.5 top-1/2 -translate-y-1/2 rounded-full p-1 text-muted-foreground transition hover:bg-muted"
   331→          >
   332→            <X className="h-4 w-4" aria-hidden="true" />
   333→          </button>
   334→        )}
   335→      </div>
   336→
   337→      {/* ── SEARCH RESULTS mode ─────────────────────────────────────────────── */}
   338→      {hasSearchQuery && (
   339→        <section aria-label="Search results" className="space-y-2">
   340→          <header className="flex items-center justify-between gap-2">
   341→            <h2 className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   342→              {searching
   343→                ? 'Searching…'
   344→                : `${filteredResults.length} ${filteredResults.length === 1 ? 'result' : 'results'}`}
   345→            </h2>
   346→          </header>
   347→
   348→          {searching ? (
   349→            <div className="space-y-2">
   350→              {Array.from({ length: 3 }).map((_, i) => (
   351→                <SocialFeedSkeleton key={i} />
   352→              ))}
   353→            </div>
   354→          ) : searchError ? (
   355→            <Card>
   356→              <CardContent className="flex items-center gap-2 p-4 text-sm text-muted-foreground">
   357→                <AlertCircle className="h-4 w-4 text-danger-600" aria-hidden="true" />
   358→                <span>{searchError}</span>
   359→              </CardContent>
   360→            </Card>
   361→          ) : filteredResults.length === 0 ? (
   362→            <EmptyState
   363→              variant="no-friends"
   364→              title="No users found"
   365→              description={
   366→                results.length === 0
   367→                  ? `No one matching "${debounced}". Try a different name or phone.`
   368→                  : 'Everyone matching your search is already your friend or has a pending request.'
   369→              }
   370→              className="py-8"
   371→            />
   372→          ) : (
   373→            <motion.ul
   374→              variants={LIST_CONTAINER}
   375→              initial={prefersReduced ? false : 'hidden'}
   376→              animate="show"
   377→              className="space-y-2"
   378→            >
   379→              <AnimatePresence>
   380→                {filteredResults.map((u) => (
   381→                  <motion.li key={u.id} variants={LIST_ITEM} exit={{ opacity: 0 }}>
   382→                    <SearchResultRow
   383→                      user={u}
   384→                      pending={pendingSentIds.has(u.id)}
   385→                      onAdd={() => handleAddFriend(u)}
   386→                    />
   387→                  </motion.li>
   388→                ))}
   389→              </AnimatePresence>
   390→            </motion.ul>
   391→          )}
   392→        </section>
   393→      )}
   394→
   395→      {/* ── DEFAULT mode (no active search) ────────────────────────────────── */}
   396→      {!hasSearchQuery && (
   397→        <motion.div
   398→          variants={LIST_CONTAINER}
   399→          initial={prefersReduced ? false : 'hidden'}
   400→          animate="show"
   401→          className="space-y-5"
   402→        >
   403→          {/* Pending incoming requests */}
   404→          <section aria-labelledby="friends-pending-in">
   405→            <header className="mb-2 flex items-center justify-between gap-2">
   406→              <h2
   407→                id="friends-pending-in"
   408→                className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   409→              >
   410→                <span className="h-1.5 w-1.5 rounded-full bg-violet-500" aria-hidden="true" />
   411→                Incoming Requests
   412→                {pendingIn.length > 0 && (
   413→                  <span className="rounded-full bg-violet-100 px-1.5 py-0.5 text-[10px] font-bold text-violet-700 dark:bg-violet-950/60 dark:text-violet-300">
   414→                    {pendingIn.length}
   415→                  </span>
   416→                )}
   417→              </h2>
   418→              <button
   419→                type="button"
   420→                onClick={handleRefresh}
   421→                className="text-xs font-medium text-muted-foreground transition hover:text-foreground"
   422→                aria-label="Refresh friends list"
   423→              >
   424→                Refresh
   425→              </button>
   426→            </header>
   427→
   428→            {pendingIn.length === 0 ? (
   429→              <p className="rounded-lg bg-muted/40 px-3 py-3 text-xs text-muted-foreground">
   430→                No pending requests.
   431→              </p>
   432→            ) : (
   433→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   434→                {pendingIn.map((c) => (
   435→                  <motion.li key={c.id} variants={LIST_ITEM}>
   436→                    <PendingRequestRow
   437→                      conn={c}
   438→                      direction="in"
   439→                      onAccept={() => handleAccept(c)}
   440→                      onReject={() => handleReject(c)}
   441→                    />
   442→                  </motion.li>
   443→                ))}
   444→              </motion.ul>
   445→            )}
   446→          </section>
   447→
   448→          {/* Pending outgoing requests */}
   449→          {pendingOut.length > 0 && (
   450→            <section aria-labelledby="friends-pending-out">
   451→              <h2
   452→                id="friends-pending-out"
   453→                className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   454→              >
   455→                <Clock className="h-3 w-3" aria-hidden="true" />
   456→                Sent Requests
   457→                <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-bold text-muted-foreground">
   458→                  {pendingOut.length}
   459→                </span>
   460→              </h2>
   461→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   462→                {pendingOut.map((c) => (
   463→                  <motion.li key={c.id} variants={LIST_ITEM}>
   464→                    <PendingRequestRow conn={c} direction="out" />
   465→                  </motion.li>
   466→                ))}
   467→              </motion.ul>
   468→            </section>
   469→          )}
   470→
   471→          {/* Current friends list */}
   472→          <section aria-labelledby="friends-list">
   473→            <h2
   474→              id="friends-list"
   475→              className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   476→            >
   477→              <span className="h-1.5 w-1.5 rounded-full bg-teal-500" aria-hidden="true" />
   478→              Your Friends
   479→              {friends.length > 0 && (
   480→                <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-bold text-muted-foreground">
   481→                  {friends.length}
   482→                </span>
   483→              )}
   484→            </h2>
   485→
   486→            {friends.length === 0 ? (
   487→              <EmptyState
   488→                variant="no-friends"
   489→                title="No friends yet"
   490→                description="Search by name or phone above to find people you know on SnakZap."
   491→                className="py-6"
   492→              />
   493→            ) : (
   494→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   495→                {friends.map((c) => (
   496→                  <motion.li key={c.id} variants={LIST_ITEM}>
   497→                    <FriendRow
   498→                      conn={c}
   499→                      onMessage={() => handleSendMessage(c)}
   500→                      onUnfriend={() => handleUnfriend(c)}
   501→                    />
   502→                  </motion.li>
   503→                ))}
   504→              </motion.ul>
   505→            )}
   506→          </section>
   507→        </motion.div>
   508→      )}
   509→    </div>
   510→  )
   511→}

   516→
   517→interface PendingRequestRowProps {
   518→  conn: SocialConnection
   519→  direction: 'in' | 'out'
   520→  onAccept?: () => void
   521→  onReject?: () => void
   522→}
   523→
   524→function PendingRequestRow({ conn, direction, onAccept, onReject }: PendingRequestRowProps) {
   525→  const [busy, setBusy] = React.useState<'accept' | 'reject' | null>(null)
   526→
   527→  async function run(action: 'accept' | 'reject') {
   528→    setBusy(action)
   529→    try {
   530→      if (action === 'accept') await onAccept?.()
   531→      else await onReject?.()
   532→    } finally {
   533→      setBusy(null)
   534→    }
   535→  }
   536→
   537→  return (
   538→    <Card className="overflow-hidden">
   539→      <CardContent className="flex items-center gap-3 p-3">
   540→        <Avatar className="h-10 w-10 ring-2 ring-violet-300 ring-offset-2 ring-offset-background">
   541→          <AvatarFallback className="bg-gradient-to-br from-violet-400 to-violet-600 text-xs font-bold text-white">
   542→            {initials(conn.name) || '?'}
   543→          </AvatarFallback>
   544→        </Avatar>
   545→        <div className="min-w-0 flex-1">
   546→          <p className="truncate text-sm font-semibold text-foreground">{conn.name}</p>
   547→          <p className="truncate text-xs text-muted-foreground">
   548→            {conn.phone ?? 'SnakZap user'}
   549→          </p>
   550→        </div>
   551→        {direction === 'in' ? (
   552→          <div className="flex shrink-0 items-center gap-1.5">
   553→            <Button
   554→              type="button"
   555→              size="sm"
   556→              onClick={() => run('accept')}
   557→              disabled={busy !== null}
   558→              className="h-8 bg-violet-600 px-3 text-xs hover:bg-violet-700"
   559→              aria-label={`Accept ${conn.name}'s friend request`}
   560→            >
   561→              {busy === 'accept' ? (
   562→                <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   563→              ) : (
   564→                <Check className="h-3.5 w-3.5" aria-hidden="true" />
   565→              )}
   566→              Accept
   567→            </Button>
   568→            <Button
   569→              type="button"
   570→              size="sm"
   571→              variant="outline"
   572→              onClick={() => run('reject')}
   573→              disabled={busy !== null}
   574→              className="h-8 px-3 text-xs"
