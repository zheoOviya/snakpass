   307→// P0-17 — Idempotency key store
   308→// ----------------------------------------
   309→// Stores the idempotency key + cached response for critical writes (orders,
   310→// payments, refunds, status updates). Retries with the same key return the
   311→// cached response instead of creating a duplicate resource.
   312→//
   313→// Usage: client sends `Idempotency-Key` header on POST /api/orders (and
   314→// future payment routes). The server checks this table inside the same
   315→// transaction as the business write — if the key exists, return the cached
   316→// response; else execute the write + store the key + response.
   317→model IdempotencyKey {
   318→  id              String   @id @default(cuid())
   319→  key             String   @unique // the client-provided idempotency key
   320→  resourceType    String   // e.g. "Order", "Payment" — which resource this key created
   321→  resourceId      String   // the id of the created resource (for lookup)
   322→  // Cached response payload (status + body) so retries return the exact same response.
   323→  responseStatus  Int      // HTTP status code
   324→  responseBody    String   // JSON-serialized response body
   325→  // Sub-Wave 3c: Request body hash (SHA-256 hex) for materially-different-request detection.
   326→  // Nullable for backward compatibility — pre-3c records have null (hash check skipped).
   327→  // When requestHashEnforcement flag is ON and this is non-null, a mismatch with the
   328→  // incoming request hash throws IdempotencyKeyReuseError (HTTP 422).
   329→  requestHash     String?
   330→  createdAt       DateTime @default(now())
   331→  expiresAt       DateTime // TTL (default 24h); old keys are safe to delete
   332→
   333→  @@index([resourceType, resourceId])
   334→}

   337→// P0-28 — Unknown-exception handling (exception queue + freeze state)
   338→// ----------------------------------------
   339→// When an invariant violation is detected (e.g., order in unknown state,
   340→// payment amount mismatch, audit hash-chain break), the system:
   341→//   1. Preserves evidence (full state snapshot + trace + invariant name)
   342→//   2. Creates an ExceptionQueue entry
   343→//   3. Applies the smallest sufficient freeze (Level 1/2/3)
   344→//   4. Alerts
   345→//
   346→// Freeze levels:
   347→//   Level 1 (TRANSACTION): freeze a single order/payment (set frozen=true)

   350→model ExceptionQueue {

   354→  // The resource that triggered the exception
   355→  entityType      String   // "Order" | "Payment" | "User" | "Restaurant" | "System"
   356→  entityId        String   // the id of the affected resource
   357→  // Freeze level applied: 1 (transaction) | 2 (entity) | 3 (system)
   358→  freezeLevel     Int      @default(1)
   359→  // Full state snapshot at time of detection (JSON: the resource state + context)

   361→  // Trace ID for correlation with logs
   362→  traceId         String?
   363→  // Human-readable description of what went wrong
   364→  description     String
   365→  // Resolution tracking
   366→  resolvedAt      DateTime?
   367→  resolvedBy      String?  // admin user id who resolved it
   368→  resolutionNote  String?  // what was done to resolve
   369→  createdAt       DateTime @default(now())
   370→
   371→  @@index([entityType, entityId])
   372→  @@index([freezeLevel, resolvedAt])

   375→// ----------------------------------------
   376→// P0-24 — Transactional outbox (cross-entity atomic writes)
   377→// ----------------------------------------
   378→// The Outbox table is the persistent event queue committed INSIDE the same
   379→// transaction as the business write. This ensures the event is NOT lost even
   380→// if the process crashes after commit — the outbox publisher worker will
   381→// pick it up and deliver it later (at-least-once delivery).
   382→//
   383→// Key principle (Architectural Law 2 — Idempotent Business Effect):
   384→//   "We do NOT chase technical 'exactly-once delivery.' We chase idempotent
   385→//    business effect." — the consumer side deduplicates via the eventId.
   386→//
   387→// Usage:
   388→//   await withTransaction(async (tx) => {
   389→//     const order = await tx.order.create({ ... })      // business mutation
   390→//     await enqueueOutboxEvent(tx, {                    // outbox INSERT (same txn)
   391→//       eventType: 'ORDER_CREATED',
   392→//       aggregateType: 'Order',
   393→//       aggregateId: order.id,
   394→//       payload: { orderId: order.id, status: order.status, ... },
   395→//     })
   396→//   })
   397→//   // Publisher (Sub-Wave 2b) later reads PENDING rows + delivers via Socket.io
   398→model Outbox {
   399→  id              String   @id @default(cuid())
   400→  // Unique event ID for consumer-side idempotency (Q17 decision: event id = idempotency key)
   401→  eventId         String   @unique
   402→  // Event type for routing (e.g., "ORDER_CREATED", "ORDER_STATUS_CHANGED", "KILL_SWITCH_TOGGLED")
   403→  eventType       String
   404→  // The aggregate root type (e.g., "Order", "KillSwitch")
   405→  aggregateType   String
   406→  // The aggregate root ID (e.g., order.id, kill-switch key)
   407→  aggregateId     String
   408→  // JSON payload of the event (the data the consumer needs)
   409→  payload         String   // JSON
   410→  // PENDING = not yet published; PUBLISHED = delivered by publisher; FAILED = max retries exhausted
   411→  status          String   @default("PENDING") // PENDING | CLAIMED | PUBLISHED | FAILED
   412→  // Number of publish attempts (for retry tracking + backoff)
   413→  attempts        Int      @default(0)
   414→  // Last error message (if publishing failed)
   415→  lastError       String?
   416→  // When the event was created (inside the business transaction)
   417→  createdAt       DateTime @default(now())
   418→  // When the publisher successfully delivered the event
   419→  publishedAt     DateTime?
   420→  // P0-24 2b-2: Claim/lease fields for crash-safe publisher
   421→  // When a worker claims this event (sets status=CLAIMED)
   422→  claimedAt       DateTime?
   423→  // When the lease expires (if publisher crashes, another worker can re-claim after this time)
   424→  claimUntil      DateTime?
   425→  // ID of the worker that claimed this event (for debugging + lease management)
   426→  workerId        String?
   427→
   428→  @@index([status, createdAt]) // publisher polls WHERE status='PENDING' ORDER BY createdAt
   429→  @@index([aggregateType, aggregateId]) // lookup by aggregate
   430→}

   432→// ----------------------------------------
   433→// P0-24 Sub-Wave 2b — Consumer-side idempotency (ProcessedEvent)
   434→// ----------------------------------------
   435→// When the outbox publisher delivers an event, the consumer checks this
   436→// table to determine if the event has already been processed. If the
   437→// eventId exists, the business effect is NOT re-applied (exactly-once
   438→// business effect, even if delivery happens multiple times).
   439→//
   440→// The dedup record and the protected business effect must be in the same
   441→// transaction (atomic dedup).
   442→model ProcessedEvent {
   443→  // Matches Outbox.eventId — the consumer-side idempotency key
   444→  eventId     String   @id
   445→  // Event type (for debugging + audit)
   446→  eventType   String
   447→  // Which consumer processed it (for multi-consumer future)
   448→  consumerId  String   @default("default")
   449→  // When the event was processed
   450→  processedAt DateTime @default(now())
   451→  // Optional: payload hash for verification
   452→  payloadHash String?
   453→
   454→  @@index([consumerId, processedAt])
   455→}

   550→// P0-04 Wave-5 Sub-Wave 5a — Refund (Razorpay refund lifecycle)
   551→// ----------------------------------------
   552→// A Refund is created when a CAPTURED payment is to be (partially or fully)
   553→// refunded. The refund API call to Razorpay is performed by the outbox publisher
   554→// (OUTSIDE any DB transaction) — the same Wave-4 4c pattern that protects the
   555→// capture flow's TRANSACTION_RETRY_INVARIANT. The refund route only writes the
   556→// Refund + reversal LedgerEntry pair + AuditLog + Outbox event inside a single
   557→// transaction; the publisher then calls refundRazorpayPayment() and on success
   558→// transitions Refund.status REFUND_PENDING → REFUNDED + Payment.status

   561→// Lifecycle:
   562→//   REFUND_PENDING (created by refund route, inside txn)
   563→//     → REFUNDED   (publisher confirms Razorpay refund success, inside new txn)

   567→// Payment are supported (1:N). Each Refund produces its own Dr/Cr reversal

   575→  // refund, is less than Payment.amount. Must be > 0.
   576→  amount          Int
   577→  currency        String   @default("INR")
   578→  // REFUND_PENDING → REFUNDED | FAILED
   579→  status          String   @default("REFUND_PENDING")
   580→  // Razorpay refund ID (rpf_*) — set by the publisher after a successful refund.
   581→  gatewayRefundId String?
   582→  // P0-17: Idempotency-Key linkage (refund double-click dedup). Unique per
   583→  // refund attempt — same key on retry returns the cached Refund.
   584→  idempotencyKey  String?  @unique
   585→  // Failure tracking (set by publisher on retry/decline).
   586→  failureReason   String?
   587→  // Set by publisher when Razorpay confirms the refund.
   588→  refundedAt      DateTime?
   589→  // P0-25: Optimistic-locking version field (race-safe conditional update).
   590→  version         Int      @default(0)
   591→  createdAt       DateTime @default(now())
   592→  updatedAt       DateTime @updatedAt
   593→

   595→  @@index([status, createdAt])

   985→// Gift context (blueprint §19)
   986→// ---------------------------------------------------------------------------
   987→// Food gifting lifecycle:
   988→//   CREATED → PAID → AVAILABLE → REDEEMED | EXPIRED | CANCELLED | REFUNDED
   989→//
   990→// Implementation uses the "ghost Order" pattern (plan Decision #6):
   991→//   - Sender pays for a normal Order via the EXISTING /api/payments route
   992→//     (governance: DO NOT modify payment routes). Order.note encodes
   993→//     `GIFT:${giftId}:for:${recipientId}` so the gift backend can correlate.
   994→//   - The ghost Order is filtered from "My Orders" UI via note-prefix filter.
   995→//   - Recipient's redemption creates a NEW zero-amount Order with note
   996→//     `GIFT_FROM:${senderId}:${giftId}`. ZERO modifications to Order model.
   997→//
   998→// Payment linkage (paymentId) is a soft FK — no Prisma relation to Payment
   999→// (governance: Payment is a money-state table, DO NOT TOUCH its relations).
  1000→// ---------------------------------------------------------------------------
  1001→model Gift {
  1002→  id             String   @id @default(cuid())
  1003→  senderId       String
  1004→  recipientId    String
  1005→  menuItemId     String
  1006→  menuItem       MenuItem @relation(fields: [menuItemId], references: [id])
  1007→  // Snapshot at creation time (price + name — protects against later menu edits)
  1008→  menuItemName   String
  1009→  menuItemPrice  Int      // paise
  1010→  message        String?
  1011→  // CREATED → PAID → AVAILABLE → REDEEMED | EXPIRED | CANCELLED | REFUNDED
  1012→  status         String   @default("CREATED")
  1013→  // Single-use code the recipient uses to redeem
  1014→  redemptionCode String   @unique
  1015→  // 30-day default expiry (set when status → AVAILABLE)
  1016→  expiresAt      DateTime
  1017→  // Soft FK to Payment.id — sender's payment (no Prisma relation; Payment is governance-protected)
  1018→  paymentId      String?
  1019→  // Recipient's order (set when recipient redeems) — soft FK to Order.id
  1020→  recipientOrderId String?
  1021→  // Lifecycle timestamps
  1022→  paidAt         DateTime?
  1023→  availableAt    DateTime?
  1024→  redeemedAt     DateTime?
  1025→  cancelledAt    DateTime?
  1026→  refundedAt     DateTime?
  1027→  createdAt      DateTime @default(now())
  1028→  updatedAt      DateTime @updatedAt
  1029→
  1030→  @@index([senderId, createdAt])
  1031→  @@index([recipientId, status])
  1032→  @@index([status, expiresAt])
  1033→}
