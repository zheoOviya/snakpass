   208→   208→// P0-17 — Idempotency key store
   209→   209→// ----------------------------------------
   210→   210→// Stores the idempotency key + cached response for critical writes (orders,
   211→   211→// payments, refunds, status updates). Retries with the same key return the
   212→   212→// cached response instead of creating a duplicate resource.
   213→   213→//
   214→   214→// Usage: client sends `Idempotency-Key` header on POST /api/orders (and
   215→   215→// future payment routes). The server checks this table inside the same
   216→   216→// transaction as the business write — if the key exists, return the cached
   217→   217→// response; else execute the write + store the key + response.
   218→   218→model IdempotencyKey {
   219→   219→  id              String   @id @default(cuid())

   221→   221→  resourceType    String   // e.g. "Order", "Payment" — which resource this key created
   222→   222→  resourceId      String   // the id of the created resource (for lookup)
   223→   223→  // Cached response payload (status + body) so retries return the exact same response.
   224→   224→  responseStatus  Int      // HTTP status code
   225→   225→  responseBody    String   // JSON-serialized response body
   226→   226→  // Sub-Wave 3c: Request body hash (SHA-256 hex) for materially-different-request detection.
   227→   227→  // Nullable for backward compatibility — pre-3c records have null (hash check skipped).
   228→   228→  // When requestHashEnforcement flag is ON and this is non-null, a mismatch with the
   229→   229→  // incoming request hash throws IdempotencyKeyReuseError (HTTP 422).
   230→   230→  requestHash     String?
   231→   231→  createdAt       DateTime @default(now())
   232→   232→  expiresAt       DateTime // TTL (default 24h); old keys are safe to delete
   233→   233→
   234→   234→  @@index([resourceType, resourceId])

   238→   238→// P0-28 — Unknown-exception handling (exception queue + freeze state)
   239→   239→// ----------------------------------------
   240→   240→// When an invariant violation is detected (e.g., order in unknown state,
   241→   241→// payment amount mismatch, audit hash-chain break), the system:
   242→   242→//   1. Preserves evidence (full state snapshot + trace + invariant name)
   243→   243→//   2. Creates an ExceptionQueue entry
   244→   244→//   3. Applies the smallest sufficient freeze (Level 1/2/3)
   245→   245→//   4. Alerts
   246→   246→//
   247→   247→// Freeze levels:
   248→   248→//   Level 1 (TRANSACTION): freeze a single order/payment (set frozen=true)
   249→   249→//   Level 2 (ENTITY): quarantine a whole Restaurant/User (set quarantined=true)
   250→   250→//   Level 3 (SYSTEM): activate system kill switch (links to P0-23 KillSwitch)
   251→   251→model ExceptionQueue {
   252→   252→  id              String   @id @default(cuid())
   253→   253→  // What was violated (e.g., "I-02 Order Integrity", "UNKNOWN_STATE")
   254→   254→  invariant       String
   255→   255→  // The resource that triggered the exception
   256→   256→  entityType      String   // "Order" | "Payment" | "User" | "Restaurant" | "System"
   257→   257→  entityId        String   // the id of the affected resource
   258→   258→  // Freeze level applied: 1 (transaction) | 2 (entity) | 3 (system)
   259→   259→  freezeLevel     Int      @default(1)
   260→   260→  // Full state snapshot at time of detection (JSON: the resource state + context)
   261→   261→  stateSnapshot   String   // JSON
   262→   262→  // Trace ID for correlation with logs
   263→   263→  traceId         String?
   264→   264→  // Human-readable description of what went wrong
   265→   265→  description     String
   266→   266→  // Resolution tracking
   267→   267→  resolvedAt      DateTime?
   268→   268→  resolvedBy      String?  // admin user id who resolved it
   269→   269→  resolutionNote  String?  // what was done to resolve
   270→   270→  createdAt       DateTime @default(now())
   271→   271→
   272→   272→  @@index([entityType, entityId])
   273→   273→  @@index([freezeLevel, resolvedAt])
   274→   274→}

   276→   276→// ----------------------------------------
   277→   277→// P0-24 — Transactional outbox (cross-entity atomic writes)
   278→   278→// ----------------------------------------
   279→   279→// The Outbox table is the persistent event queue committed INSIDE the same
   280→   280→// transaction as the business write. This ensures the event is NOT lost even
   281→   281→// if the process crashes after commit — the outbox publisher worker will
   282→   282→// pick it up and deliver it later (at-least-once delivery).
   283→   283→//
   284→   284→// Key principle (Architectural Law 2 — Idempotent Business Effect):
   285→   285→//   "We do NOT chase technical 'exactly-once delivery.' We chase idempotent
   286→   286→//    business effect." — the consumer side deduplicates via the eventId.
   287→   287→//
   288→   288→// Usage:
   289→   289→//   await withTransaction(async (tx) => {
   290→   290→//     const order = await tx.order.create({ ... })      // business mutation
   291→   291→//     await enqueueOutboxEvent(tx, {                    // outbox INSERT (same txn)
   292→   292→//       eventType: 'ORDER_CREATED',
   293→   293→//       aggregateType: 'Order',
   294→   294→//       aggregateId: order.id,
   295→   295→//       payload: { orderId: order.id, status: order.status, ... },
   296→   296→//     })
   297→   297→//   })
   298→   298→//   // Publisher (Sub-Wave 2b) later reads PENDING rows + delivers via Socket.io
   299→   299→model Outbox {
   300→   300→  id              String   @id @default(cuid())
   301→   301→  // Unique event ID for consumer-side idempotency (Q17 decision: event id = idempotency key)
   302→   302→  eventId         String   @unique
   303→   303→  // Event type for routing (e.g., "ORDER_CREATED", "ORDER_STATUS_CHANGED", "KILL_SWITCH_TOGGLED")
   304→   304→  eventType       String
   305→   305→  // The aggregate root type (e.g., "Order", "KillSwitch")
   306→   306→  aggregateType   String
   307→   307→  // The aggregate root ID (e.g., order.id, kill-switch key)
   308→   308→  aggregateId     String
   309→   309→  // JSON payload of the event (the data the consumer needs)
   310→   310→  payload         String   // JSON
   311→   311→  // PENDING = not yet published; PUBLISHED = delivered by publisher; FAILED = max retries exhausted
   312→   312→  status          String   @default("PENDING") // PENDING | CLAIMED | PUBLISHED | FAILED
   313→   313→  // Number of publish attempts (for retry tracking + backoff)
   314→   314→  attempts        Int      @default(0)
   315→   315→  // Last error message (if publishing failed)
   316→   316→  lastError       String?
   317→   317→  // When the event was created (inside the business transaction)
   318→   318→  createdAt       DateTime @default(now())
   319→   319→  // When the publisher successfully delivered the event
   320→   320→  publishedAt     DateTime?
   321→   321→  // P0-24 2b-2: Claim/lease fields for crash-safe publisher
   322→   322→  // When a worker claims this event (sets status=CLAIMED)
   323→   323→  claimedAt       DateTime?
   324→   324→  // When the lease expires (if publisher crashes, another worker can re-claim after this time)
   325→   325→  claimUntil      DateTime?
   326→   326→  // ID of the worker that claimed this event (for debugging + lease management)
   327→   327→  workerId        String?
   328→   328→
   329→   329→  @@index([status, createdAt]) // publisher polls WHERE status='PENDING' ORDER BY createdAt
   330→   330→  @@index([aggregateType, aggregateId]) // lookup by aggregate
   331→   331→}

   334→   334→// P0-24 Sub-Wave 2b — Consumer-side idempotency (ProcessedEvent)
   335→   335→// ----------------------------------------
   336→   336→// When the outbox publisher delivers an event, the consumer checks this
   337→   337→// table to determine if the event has already been processed. If the
   338→   338→// eventId exists, the business effect is NOT re-applied (exactly-once
   339→   339→// business effect, even if delivery happens multiple times).

   343→   343→model ProcessedEvent {

   345→   345→  eventId     String   @id
   346→   346→  // Event type (for debugging + audit)
   347→   347→  eventType   String
   348→   348→  // Which consumer processed it (for multi-consumer future)
   349→   349→  consumerId  String   @default("default")
   350→   350→  // When the event was processed
   351→   351→  processedAt DateTime @default(now())
   352→   352→  // Optional: payload hash for verification
   353→   353→  payloadHash String?
   354→   354→
   355→   355→  @@index([consumerId, processedAt])
   356→   356→}

   451→   451→// P0-04 Wave-5 Sub-Wave 5a — Refund (Razorpay refund lifecycle)
   452→   452→// ----------------------------------------
   453→   453→// A Refund is created when a CAPTURED payment is to be (partially or fully)
   454→   454→// refunded. The refund API call to Razorpay is performed by the outbox publisher
   455→   455→// (OUTSIDE any DB transaction) — the same Wave-4 4c pattern that protects the
   456→   456→// capture flow's TRANSACTION_RETRY_INVARIANT. The refund route only writes the
   457→   457→// Refund + reversal LedgerEntry pair + AuditLog + Outbox event inside a single
   458→   458→// transaction; the publisher then calls refundRazorpayPayment() and on success
   459→   459→// transitions Refund.status REFUND_PENDING → REFUNDED + Payment.status
   460→   460→// CAPTURED → REFUNDED (full refund) atomically.
   461→   461→//
   462→   462→// Lifecycle:
   463→   463→//   REFUND_PENDING (created by refund route, inside txn)
   464→   464→//     → REFUNDED   (publisher confirms Razorpay refund success, inside new txn)
   465→   465→//     → FAILED     (publisher exhausts retries / gateway declined)
   466→   466→//
   467→   467→// Partial refunds: amount < Payment.amount is allowed. Multiple Refunds per
   468→   468→// Payment are supported (1:N). Each Refund produces its own Dr/Cr reversal
   469→   469→// LedgerEntry pair tied to the same paymentId (entries are append-only — no
   470→   470→// updates to existing rows).
   471→   471→model Refund {
   472→   472→  id              String   @id @default(cuid())
   473→   473→  paymentId       String
   474→   474→  payment         Payment  @relation(fields: [paymentId], references: [id])
   475→   475→  // Amount (paise). For a full refund, equals Payment.amount. For a partial
   476→   476→  // refund, is less than Payment.amount. Must be > 0.
   477→   477→  amount          Int
   478→   478→  currency        String   @default("INR")
   479→   479→  // REFUND_PENDING → REFUNDED | FAILED
   480→   480→  status          String   @default("REFUND_PENDING")
   481→   481→  // Razorpay refund ID (rpf_*) — set by the publisher after a successful refund.
   482→   482→  gatewayRefundId String?
   483→   483→  // P0-17: Idempotency-Key linkage (refund double-click dedup). Unique per
   484→   484→  // refund attempt — same key on retry returns the cached Refund.
   485→   485→  idempotencyKey  String?  @unique
   486→   486→  // Failure tracking (set by publisher on retry/decline).
   487→   487→  failureReason   String?
   488→   488→  // Set by publisher when Razorpay confirms the refund.
   489→   489→  refundedAt      DateTime?
   490→   490→  // P0-25: Optimistic-locking version field (race-safe conditional update).
   491→   491→  version         Int      @default(0)
   492→   492→  createdAt       DateTime @default(now())
   493→   493→  updatedAt       DateTime @updatedAt
   494→   494→
   495→   495→  @@index([paymentId, status])
   496→   496→  @@index([status, createdAt])
   497→   497→}

   500→   500→// P0-03 Wave-5 Sub-Wave 5b — Reconciliation (detection-only)
   501→   501→// ----------------------------------------
   502→   502→// Two Class-2 ADDITIVE models that store the audit trail of reconciliation
   503→   503→// runs + their findings. The reconciliation job is READ-ONLY with respect to
   504→   504→// all money-state tables (Payment, Refund, LedgerEntry, Outbox, WebhookEvent,
   505→   505→// IdempotencyKey, AuditLog). Its only writes are to these two tables + to
   506→   506→// ExceptionQueue (via the existing reportInvariantViolation() P0-28 path).
   507→   507→//
   508→   508→// ReconciliationRun: one row per reconciliation cycle (cron / manual / evidence).
   509→   509→// Tracks when it started, completed, what it checked, and what it found.
   510→   510→//
   511→   511→// ReconciliationFinding: one row per detected mismatch (deduped by
   512→   512→// (mismatchClass, entityId) while unresolved — a second run for the same
   513→   513→// mismatch updates lastSeenAt, NOT a new row). High-severity findings are
   514→   514→// routed to ExceptionQueue via reportInvariantViolation() (exceptionId FK).
   515→   515→//
   516→   516→// SAFETY CONTRACT (Orchestrator hard boundary):
   517→   517→//   - Reconciliation NEVER writes to Payment, Refund, LedgerEntry, Outbox,
   518→   518→//     WebhookEvent, IdempotencyKey, or AuditLog.
   519→   519→//   - Reconciliation NEVER makes external Razorpay API calls.
   520→   520→//   - Reconciliation NEVER triggers capture / refund / outbox enqueue.
   521→   521→//   - Reconciliation NEVER performs automatic financial correction.
   522→   522→//   - Its job is: detect → classify → record → report. Repair is a separate
   523→   523→//     authorization boundary (5C or dedicated Orchestrator directive).
   524→   524→model ReconciliationRun {
   525→   525→  id              String   @id @default(cuid())
   526→   526→  // What triggered this run: 'cron' | 'manual' | 'evidence'
   527→   527→  // (renamed from `trigger` — that's a reserved keyword in SQLite/Prisma)
   528→   528→  triggerType     String   @default("cron")
   529→   529→  // Status of the run itself: RUNNING | COMPLETED | FAILED
   530→   530→  status          String   @default("RUNNING")
   531→   531→  startedAt       DateTime @default(now())
   532→   532→  completedAt     DateTime?
   533→   533→  // Summary counts (populated on completion)
   534→   534→  paymentsChecked Int      @default(0)
   535→   535→  refundsChecked  Int      @default(0)
   536→   536→  outboxChecked    Int      @default(0)
   537→   537→  webhooksChecked  Int      @default(0)
   538→   538→  findingsCount    Int      @default(0)
   539→   539→  // The mismatch_count metric value emitted for this run (activates the
   540→   540→  // existing 'reconciliation-mismatch' alert rule in alerting.ts).
   541→   541→  mismatchCount    Int      @default(0)
   542→   542→  // Error tracking (if the run itself failed)
   543→   543→  lastError        String?
   544→   544→  findings         ReconciliationFinding[]
   545→   545→
   546→   546→  @@index([status, startedAt])
   547→   547→}

   549→   549→model ReconciliationFinding {
   550→   550→  id              String   @id @default(cuid())
   551→   551→  runId           String
   552→   552→  run             ReconciliationRun @relation(fields: [runId], references: [id], onDelete: Cascade)
   553→   553→  // Mismatch class (M1-M17 from WAVE5_5B_GATE_REVIEW.md §2 D4)
   554→   554→  // e.g. "M1_LEDGER_IMBALANCE", "M9_STUCK_CAPTURE_PENDING", "M10_STUCK_REFUND_PENDING"
   555→   555→  mismatchClass   String
   556→   556→  // Severity: CRITICAL | HIGH | MEDIUM
   557→   557→  // CRITICAL/HIGH → routed to ExceptionQueue via reportInvariantViolation()
   558→   558→  severity        String
   559→   559→  // The affected entity type: "Payment" | "Refund" | "Outbox" | "WebhookEvent"
   560→   560→  entityType      String
   561→   561→  // The affected entity id (paymentId / refundId / outbox id / webhookEvent id)
   562→   562→  entityId        String
   563→   563→  // Full state snapshot at detection time (JSON — the evidence for the finding)
   564→   564→  stateSnapshot   String
   565→   565→  // Human-readable description of the mismatch
   566→   566→  description     String
   567→   567→  // Recommended remediation (informational only — NOT auto-applied)
   568→   568→  recommendedRemediation String?
   569→   569→  // ExceptionQueue entry id (set only for CRITICAL/HIGH findings routed via

   572→   572→  exceptionId     String?
   573→   573→  // Detection timestamps (idempotency: a second run for the same mismatch

   579→   579→  resolvedBy      String?
   580→   580→  resolutionNote  String?
   581→   581→  // 5C: Remediation actions taken for this finding (1:N — a finding may be

   583→   583→  remediationActions RemediationAction[]
   584→   584→
   585→   585→  // Idempotency: one unresolved finding per (mismatchClass, entityId).

   598→   598→// P0-03 Wave-5 Sub-Wave 5C — Remediation Action (M16-only, Class-2 additive)
   599→   599→// ----------------------------------------
   600→   600→// Audit trail for remediation/repair actions. Currently ONLY M16 (outbox lag —
   601→   601→// operational, non-financial) remediation is authorized. M3/M9/M10 + CLASS B/D/E
   602→   602→// are NOT authorized — they require separate Orchestrator directives.
   603→   603→//
   604→   604→// SI-6 (Safety Invariant): every repair writes a RemediationAction audit record.
   605→   605→// SI-2 (Repair Idempotency): the unique constraint on (findingId, repairType)
   606→   606→// prevents duplicate repair actions for the same finding + repair type. A second
   607→   607→// attempt to repair the same finding with the same repairType is a no-op (caught
   608→   608→// by the unique constraint → the handler skips).
   609→   609→//
   610→   610→// SAFETY CONTRACT (Orchestrator hard boundary for 5C-M16):
   611→   611→//   - M16 remediation NEVER writes to Payment, Refund, LedgerEntry, Outbox,
   612→   612→//     WebhookEvent, IdempotencyKey, or AuditLog.
   613→   613→//   - M16 remediation NEVER makes external Razorpay API calls.
   614→   614→//   - M16 remediation's only "external" action is an HTTP call to the outbox
   615→   615→//     publisher's /trigger endpoint (operational restart — no financial mutation).
   616→   616→//   - M16 remediation's only DB writes are to this table (RemediationAction) +
   617→   617→//     ReconciliationFinding (resolves the finding).
   618→   618→model RemediationAction {
   619→   619→  id              String   @id @default(cuid())
   620→   620→  // The finding being repaired
   621→   621→  findingId       String
   622→   622→  finding         ReconciliationFinding @relation(fields: [findingId], references: [id], onDelete: Cascade)
   623→   623→  // The repair type (e.g., "M16_PUBLISHER_TRIGGER"). Only M16 is authorized.
   624→   624→  repairType      String
   625→   625→  // The status of the repair action: ATTEMPTED | SUCCEEDED | SKIPPED | FAILED
   626→   626→  status          String   @default("ATTEMPTED")
   627→   627→  // What the repair did (JSON snapshot — e.g., { publisherTriggerCalled: true, lagBefore: 360, lagAfter: 45 })
   628→   628→  actionSnapshot  String
   629→   629→  // When the repair was attempted
   630→   630→  attemptedAt     DateTime @default(now())
   631→   631→  // When the repair completed (set on SUCCEEDED/SKIPPED/FAILED)
   632→   632→  completedAt     DateTime?
   633→   633→  // Error message if the repair FAILED
   634→   634→  error           String?
   635→   635→
   636→   636→  // Idempotency: one repair action per (findingId, repairType). A second attempt
   637→   637→  // to repair the same finding with the same repairType throws P2002 (unique
   638→   638→  // constraint) — the handler catches this and skips (SI-2: Repair Idempotency).
   639→   639→  @@unique([findingId, repairType])
   640→   640→  @@index([status, attemptedAt])
   641→   641→  @@index([repairType])
   642→   642→}

   645→   645→// P0-06 Wave-6 — Fulfilment (parallel state machine — additive-only)
   646→   646→// ----------------------------------------
   647→   647→// 1:1 to Order. Parallel to Order.status (NEXT_STATUS) — does NOT modify
   648→   648→// the existing Order state machine. Fulfilment has its own NEXT_FULFILMENT_STATUS
   649→   649→// machine: PREPARING → ALMOST_READY → READY_FOR_PICKUP → PICKED_UP.
   650→   650→//
   651→   651→// Lifecycle:
   652→   652→//   PREPARING        (default — created lazily on first access)
   653→   653→//     → ALMOST_READY
   654→   654→//     → READY_FOR_PICKUP
   655→   655→//     → PICKED_UP    (terminal)
   656→   656→//
   657→   657→// Lazy-created: when GET/PATCH /api/orders/[id]/fulfilment is called and no
   658→   658→// Fulfilment row exists yet, it is created with status='PREPARING' +
   659→   659→// pickupOtp copied from Order.pickupOtp.
   660→   660→//
   661→   661→// Optimistic locking: every PATCH uses a conditional UPDATE
   662→   662→// (`WHERE id = X AND version = expectedVersion`). A version mismatch returns
   663→   663→// HTTP 409 (same pattern as Order.status PATCH — P0-25 Case B).
   664→   664→//
   665→   665→// P0-07 (pickup attribution): pickupOtp / pickupVerifiedAt / pickupVerifiedBy
   666→   666→// fields are PRESENT but INACTIVE in this wave. No QR+OTP verification is
   667→   667→// performed on the PICKED_UP transition; no RBAC is enforced on the
   668→   668→// PICKED_UP actor. P0-07 activation is a separate Orchestrator directive.
   669→   669→model Fulfilment {
   670→   670→  id                String   @id @default(cuid())
   671→   671→  // 1:1 to Order — @unique ensures at most one Fulfilment per Order.
   672→   672→  orderId           String   @unique
   673→   673→  order             Order    @relation(fields: [orderId], references: [id])
   674→   674→  // Parallel state machine (NEXT_FULFILMENT_STATUS) — does NOT touch Order.status.
   675→   675→  status            String   @default("PREPARING") // PREPARING | ALMOST_READY | READY_FOR_PICKUP | PICKED_UP
   676→   676→  // JSON array of {status, at} — parallel to Order.statusHistory.
   677→   677→  statusHistory     String   @default("[]")
   678→   678→  // P0-25: Optimistic-locking version field. Incremented on every PATCH.
   679→   679→  version           Int      @default(0)
   680→   680→  // P0-07 future: pickup attribution fields (INACTIVE in this wave).
   681→   681→  pickupOtp         String?  // copied from Order.pickupOtp on lazy-create
   682→   682→  pickupVerifiedAt  DateTime? // set when pickup is verified (P0-07 — not yet enforced)
   683→   683→  pickupVerifiedBy  String?  // actor who verified pickup (P0-07 — not yet enforced)
   684→   684→  createdAt         DateTime @default(now())
   685→   685→  updatedAt         DateTime @updatedAt
   686→   686→
   687→   687→  @@index([status])
   688→   688→}
