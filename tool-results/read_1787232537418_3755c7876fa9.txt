   247→// P0-17 — Idempotency key store
   248→// ----------------------------------------
   249→// Stores the idempotency key + cached response for critical writes (orders,
   250→// payments, refunds, status updates). Retries with the same key return the
   251→// cached response instead of creating a duplicate resource.
   252→//
   253→// Usage: client sends `Idempotency-Key` header on POST /api/orders (and
   254→// future payment routes). The server checks this table inside the same
   255→// transaction as the business write — if the key exists, return the cached
   256→// response; else execute the write + store the key + response.
   257→model IdempotencyKey {
   258→  id              String   @id @default(cuid())
   259→  key             String   @unique // the client-provided idempotency key
   260→  resourceType    String   // e.g. "Order", "Payment" — which resource this key created
   261→  resourceId      String   // the id of the created resource (for lookup)
   262→  // Cached response payload (status + body) so retries return the exact same response.
   263→  responseStatus  Int      // HTTP status code
   264→  responseBody    String   // JSON-serialized response body
   265→  // Sub-Wave 3c: Request body hash (SHA-256 hex) for materially-different-request detection.
   266→  // Nullable for backward compatibility — pre-3c records have null (hash check skipped).
   267→  // When requestHashEnforcement flag is ON and this is non-null, a mismatch with the
   268→  // incoming request hash throws IdempotencyKeyReuseError (HTTP 422).
   269→  requestHash     String?
   270→  createdAt       DateTime @default(now())
   271→  expiresAt       DateTime // TTL (default 24h); old keys are safe to delete
   272→
   273→  @@index([resourceType, resourceId])
   274→}

   295→  entityType      String   // "Order" | "Payment" | "User" | "Restaurant" | "System"
   296→  entityId        String   // the id of the affected resource
   297→  // Freeze level applied: 1 (transaction) | 2 (entity) | 3 (system)
   298→  freezeLevel     Int      @default(1)

   311→  @@index([entityType, entityId])
   312→  @@index([freezeLevel, resolvedAt])

   315→// ----------------------------------------
   316→// P0-24 — Transactional outbox (cross-entity atomic writes)
   317→// ----------------------------------------
   318→// The Outbox table is the persistent event queue committed INSIDE the same
   319→// transaction as the business write. This ensures the event is NOT lost even
   320→// if the process crashes after commit — the outbox publisher worker will
   321→// pick it up and deliver it later (at-least-once delivery).
   322→//
   323→// Key principle (Architectural Law 2 — Idempotent Business Effect):
   324→//   "We do NOT chase technical 'exactly-once delivery.' We chase idempotent
   325→//    business effect." — the consumer side deduplicates via the eventId.
   326→//
   327→// Usage:
   328→//   await withTransaction(async (tx) => {
   329→//     const order = await tx.order.create({ ... })      // business mutation
   330→//     await enqueueOutboxEvent(tx, {                    // outbox INSERT (same txn)
   331→//       eventType: 'ORDER_CREATED',
   332→//       aggregateType: 'Order',
   333→//       aggregateId: order.id,
   334→//       payload: { orderId: order.id, status: order.status, ... },
   335→//     })
   336→//   })
   337→//   // Publisher (Sub-Wave 2b) later reads PENDING rows + delivers via Socket.io
   338→model Outbox {
   339→  id              String   @id @default(cuid())
   340→  // Unique event ID for consumer-side idempotency (Q17 decision: event id = idempotency key)
   341→  eventId         String   @unique
   342→  // Event type for routing (e.g., "ORDER_CREATED", "ORDER_STATUS_CHANGED", "KILL_SWITCH_TOGGLED")
   343→  eventType       String
   344→  // The aggregate root type (e.g., "Order", "KillSwitch")
   345→  aggregateType   String
   346→  // The aggregate root ID (e.g., order.id, kill-switch key)
   347→  aggregateId     String
   348→  // JSON payload of the event (the data the consumer needs)
   349→  payload         String   // JSON
   350→  // PENDING = not yet published; PUBLISHED = delivered by publisher; FAILED = max retries exhausted
   351→  status          String   @default("PENDING") // PENDING | CLAIMED | PUBLISHED | FAILED
   352→  // Number of publish attempts (for retry tracking + backoff)
   353→  attempts        Int      @default(0)
   354→  // Last error message (if publishing failed)
   355→  lastError       String?
   356→  // When the event was created (inside the business transaction)
   357→  createdAt       DateTime @default(now())
   358→  // When the publisher successfully delivered the event
   359→  publishedAt     DateTime?
   360→  // P0-24 2b-2: Claim/lease fields for crash-safe publisher
   361→  // When a worker claims this event (sets status=CLAIMED)
   362→  claimedAt       DateTime?
   363→  // When the lease expires (if publisher crashes, another worker can re-claim after this time)
   364→  claimUntil      DateTime?
   365→  // ID of the worker that claimed this event (for debugging + lease management)
   366→  workerId        String?
   367→
   368→  @@index([status, createdAt]) // publisher polls WHERE status='PENDING' ORDER BY createdAt
   369→  @@index([aggregateType, aggregateId]) // lookup by aggregate
   370→}

   391→  // Optional: payload hash for verification

   393→
   394→  @@index([consumerId, processedAt])

   489→// ----------------------------------------
   490→// P0-04 Wave-5 Sub-Wave 5a — Refund (Razorpay refund lifecycle)
   491→// ----------------------------------------
   492→// A Refund is created when a CAPTURED payment is to be (partially or fully)
   493→// refunded. The refund API call to Razorpay is performed by the outbox publisher
   494→// (OUTSIDE any DB transaction) — the same Wave-4 4c pattern that protects the
   495→// capture flow's TRANSACTION_RETRY_INVARIANT. The refund route only writes the
   496→// Refund + reversal LedgerEntry pair + AuditLog + Outbox event inside a single
   497→// transaction; the publisher then calls refundRazorpayPayment() and on success

   539→// P0-03 Wave-5 Sub-Wave 5b — Reconciliation (detection-only)

   541→// Two Class-2 ADDITIVE models that store the audit trail of reconciliation

   547→// ReconciliationRun: one row per reconciliation cycle (cron / manual / evidence).

   638→// ----------------------------------------

   640→// operational, non-financial) remediation is authorized. M3/M9/M10 + CLASS B/D/E

   667→  actionSnapshot  String
