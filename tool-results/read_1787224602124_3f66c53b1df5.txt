   208→// P0-17 — Idempotency key store
   209→// ----------------------------------------
   210→// Stores the idempotency key + cached response for critical writes (orders,
   211→// payments, refunds, status updates). Retries with the same key return the
   212→// cached response instead of creating a duplicate resource.
   213→//
   214→// Usage: client sends `Idempotency-Key` header on POST /api/orders (and
   215→// future payment routes). The server checks this table inside the same
   216→// transaction as the business write — if the key exists, return the cached
   217→// response; else execute the write + store the key + response.
   218→model IdempotencyKey {
   219→  id              String   @id @default(cuid())
   220→  key             String   @unique // the client-provided idempotency key
   221→  resourceType    String   // e.g. "Order", "Payment" — which resource this key created
   222→  resourceId      String   // the id of the created resource (for lookup)
   223→  // Cached response payload (status + body) so retries return the exact same response.
   224→  responseStatus  Int      // HTTP status code
   225→  responseBody    String   // JSON-serialized response body
   226→  // Sub-Wave 3c: Request body hash (SHA-256 hex) for materially-different-request detection.
   227→  // Nullable for backward compatibility — pre-3c records have null (hash check skipped).
   228→  // When requestHashEnforcement flag is ON and this is non-null, a mismatch with the
   229→  // incoming request hash throws IdempotencyKeyReuseError (HTTP 422).
   230→  requestHash     String?
   231→  createdAt       DateTime @default(now())
   232→  expiresAt       DateTime // TTL (default 24h); old keys are safe to delete
   233→
   234→  @@index([resourceType, resourceId])
   235→}

   238→// P0-28 — Unknown-exception handling (exception queue + freeze state)
   239→// ----------------------------------------
   240→// When an invariant violation is detected (e.g., order in unknown state,
   241→// payment amount mismatch, audit hash-chain break), the system:
   242→//   1. Preserves evidence (full state snapshot + trace + invariant name)
   243→//   2. Creates an ExceptionQueue entry
   244→//   3. Applies the smallest sufficient freeze (Level 1/2/3)
   245→//   4. Alerts
   246→//
   247→// Freeze levels:
   248→//   Level 1 (TRANSACTION): freeze a single order/payment (set frozen=true)
   249→//   Level 2 (ENTITY): quarantine a whole Restaurant/User (set quarantined=true)
   250→//   Level 3 (SYSTEM): activate system kill switch (links to P0-23 KillSwitch)
   251→model ExceptionQueue {
   252→  id              String   @id @default(cuid())
   253→  // What was violated (e.g., "I-02 Order Integrity", "UNKNOWN_STATE")
   254→  invariant       String
   255→  // The resource that triggered the exception
   256→  entityType      String   // "Order" | "Payment" | "User" | "Restaurant" | "System"
   257→  entityId        String   // the id of the affected resource
   258→  // Freeze level applied: 1 (transaction) | 2 (entity) | 3 (system)
   259→  freezeLevel     Int      @default(1)
   260→  // Full state snapshot at time of detection (JSON: the resource state + context)
   261→  stateSnapshot   String   // JSON
   262→  // Trace ID for correlation with logs
   263→  traceId         String?
   264→  // Human-readable description of what went wrong
   265→  description     String
   266→  // Resolution tracking
   267→  resolvedAt      DateTime?
   268→  resolvedBy      String?  // admin user id who resolved it
   269→  resolutionNote  String?  // what was done to resolve
   270→  createdAt       DateTime @default(now())
   271→
   272→  @@index([entityType, entityId])
   273→  @@index([freezeLevel, resolvedAt])
   274→}

   276→// ----------------------------------------
   277→// P0-24 — Transactional outbox (cross-entity atomic writes)
   278→// ----------------------------------------
   279→// The Outbox table is the persistent event queue committed INSIDE the same
   280→// transaction as the business write. This ensures the event is NOT lost even
   281→// if the process crashes after commit — the outbox publisher worker will
   282→// pick it up and deliver it later (at-least-once delivery).
   283→//
   284→// Key principle (Architectural Law 2 — Idempotent Business Effect):
   285→//   "We do NOT chase technical 'exactly-once delivery.' We chase idempotent
   286→//    business effect." — the consumer side deduplicates via the eventId.
   287→//
   288→// Usage:
   289→//   await withTransaction(async (tx) => {
   290→//     const order = await tx.order.create({ ... })      // business mutation
   291→//     await enqueueOutboxEvent(tx, {                    // outbox INSERT (same txn)
   292→//       eventType: 'ORDER_CREATED',
   293→//       aggregateType: 'Order',
   294→//       aggregateId: order.id,
   295→//       payload: { orderId: order.id, status: order.status, ... },
   296→//     })
   297→//   })
   298→//   // Publisher (Sub-Wave 2b) later reads PENDING rows + delivers via Socket.io
   299→model Outbox {
   300→  id              String   @id @default(cuid())
   301→  // Unique event ID for consumer-side idempotency (Q17 decision: event id = idempotency key)
   302→  eventId         String   @unique
   303→  // Event type for routing (e.g., "ORDER_CREATED", "ORDER_STATUS_CHANGED", "KILL_SWITCH_TOGGLED")
   304→  eventType       String
   305→  // The aggregate root type (e.g., "Order", "KillSwitch")
   306→  aggregateType   String
   307→  // The aggregate root ID (e.g., order.id, kill-switch key)
   308→  aggregateId     String
   309→  // JSON payload of the event (the data the consumer needs)
   310→  payload         String   // JSON
   311→  // PENDING = not yet published; PUBLISHED = delivered by publisher; FAILED = max retries exhausted
   312→  status          String   @default("PENDING") // PENDING | CLAIMED | PUBLISHED | FAILED
   313→  // Number of publish attempts (for retry tracking + backoff)
   314→  attempts        Int      @default(0)
   315→  // Last error message (if publishing failed)
   316→  lastError       String?
   317→  // When the event was created (inside the business transaction)
   318→  createdAt       DateTime @default(now())
   319→  // When the publisher successfully delivered the event
   320→  publishedAt     DateTime?
   321→  // P0-24 2b-2: Claim/lease fields for crash-safe publisher
   322→  // When a worker claims this event (sets status=CLAIMED)
   323→  claimedAt       DateTime?
   324→  // When the lease expires (if publisher crashes, another worker can re-claim after this time)
   325→  claimUntil      DateTime?
   326→  // ID of the worker that claimed this event (for debugging + lease management)
   327→  workerId        String?
   328→
   329→  @@index([status, createdAt]) // publisher polls WHERE status='PENDING' ORDER BY createdAt
   330→  @@index([aggregateType, aggregateId]) // lookup by aggregate
   331→}

   334→// P0-24 Sub-Wave 2b — Consumer-side idempotency (ProcessedEvent)
   335→// ----------------------------------------
   336→// When the outbox publisher delivers an event, the consumer checks this

   341→// The dedup record and the protected business effect must be in the same

   349→  consumerId  String   @default("default")
   350→  // When the event was processed
   351→  processedAt DateTime @default(now())
   352→  // Optional: payload hash for verification
   353→  payloadHash String?
   354→
   355→  @@index([consumerId, processedAt])
   356→}

   450→// ----------------------------------------
   451→// P0-04 Wave-5 Sub-Wave 5a — Refund (Razorpay refund lifecycle)
   452→// ----------------------------------------
   453→// A Refund is created when a CAPTURED payment is to be (partially or fully)
   454→// refunded. The refund API call to Razorpay is performed by the outbox publisher
   455→// (OUTSIDE any DB transaction) — the same Wave-4 4c pattern that protects the
   456→// capture flow's TRANSACTION_RETRY_INVARIANT. The refund route only writes the
   457→// Refund + reversal LedgerEntry pair + AuditLog + Outbox event inside a single
   458→// transaction; the publisher then calls refundRazorpayPayment() and on success
   459→// transitions Refund.status REFUND_PENDING → REFUNDED + Payment.status
   460→// CAPTURED → REFUNDED (full refund) atomically.
   461→//
   462→// Lifecycle:
   463→//   REFUND_PENDING (created by refund route, inside txn)
   464→//     → REFUNDED   (publisher confirms Razorpay refund success, inside new txn)
   465→//     → FAILED     (publisher exhausts retries / gateway declined)
   466→//
   467→// Partial refunds: amount < Payment.amount is allowed. Multiple Refunds per
   468→// Payment are supported (1:N). Each Refund produces its own Dr/Cr reversal
   469→// LedgerEntry pair tied to the same paymentId (entries are append-only — no
   470→// updates to existing rows).
   471→model Refund {
   472→  id              String   @id @default(cuid())
   473→  paymentId       String
   474→  payment         Payment  @relation(fields: [paymentId], references: [id])
   475→  // Amount (paise). For a full refund, equals Payment.amount. For a partial
   476→  // refund, is less than Payment.amount. Must be > 0.
   477→  amount          Int
   478→  currency        String   @default("INR")
   479→  // REFUND_PENDING → REFUNDED | FAILED
   480→  status          String   @default("REFUND_PENDING")
   481→  // Razorpay refund ID (rpf_*) — set by the publisher after a successful refund.
   482→  gatewayRefundId String?
   483→  // P0-17: Idempotency-Key linkage (refund double-click dedup). Unique per
   484→  // refund attempt — same key on retry returns the cached Refund.
   485→  idempotencyKey  String?  @unique
   486→  // Failure tracking (set by publisher on retry/decline).
   487→  failureReason   String?
   488→  // Set by publisher when Razorpay confirms the refund.
   489→  refundedAt      DateTime?
   490→  // P0-25: Optimistic-locking version field (race-safe conditional update).
   491→  version         Int      @default(0)
   492→  createdAt       DateTime @default(now())
   493→  updatedAt       DateTime @updatedAt
   494→
   495→  @@index([paymentId, status])
   496→  @@index([status, createdAt])
   497→}

   500→// P0-03 Wave-5 Sub-Wave 5b — Reconciliation (detection-only)
   501→// ----------------------------------------
   502→// Two Class-2 ADDITIVE models that store the audit trail of reconciliation
   503→// runs + their findings. The reconciliation job is READ-ONLY with respect to
   504→// all money-state tables (Payment, Refund, LedgerEntry, Outbox, WebhookEvent,
   505→// IdempotencyKey, AuditLog). Its only writes are to these two tables + to
   506→// ExceptionQueue (via the existing reportInvariantViolation() P0-28 path).
   507→//
   508→// ReconciliationRun: one row per reconciliation cycle (cron / manual / evidence).
   509→// Tracks when it started, completed, what it checked, and what it found.
   510→//
   511→// ReconciliationFinding: one row per detected mismatch (deduped by
   512→// (mismatchClass, entityId) while unresolved — a second run for the same
   513→// mismatch updates lastSeenAt, NOT a new row). High-severity findings are
   514→// routed to ExceptionQueue via reportInvariantViolation() (exceptionId FK).
   515→//
   516→// SAFETY CONTRACT (Orchestrator hard boundary):
   517→//   - Reconciliation NEVER writes to Payment, Refund, LedgerEntry, Outbox,
   518→//     WebhookEvent, IdempotencyKey, or AuditLog.
   519→//   - Reconciliation NEVER makes external Razorpay API calls.
   520→//   - Reconciliation NEVER triggers capture / refund / outbox enqueue.
   521→//   - Reconciliation NEVER performs automatic financial correction.
   522→//   - Its job is: detect → classify → record → report. Repair is a separate
   523→//     authorization boundary (5C or dedicated Orchestrator directive).
   524→model ReconciliationRun {
   525→  id              String   @id @default(cuid())
   526→  // What triggered this run: 'cron' | 'manual' | 'evidence'
   527→  // (renamed from `trigger` — that's a reserved keyword in SQLite/Prisma)
   528→  triggerType     String   @default("cron")
   529→  // Status of the run itself: RUNNING | COMPLETED | FAILED
   530→  status          String   @default("RUNNING")
   531→  startedAt       DateTime @default(now())
   532→  completedAt     DateTime?
   533→  // Summary counts (populated on completion)
   534→  paymentsChecked Int      @default(0)
   535→  refundsChecked  Int      @default(0)
   536→  outboxChecked    Int      @default(0)
   537→  webhooksChecked  Int      @default(0)
   538→  findingsCount    Int      @default(0)
   539→  // The mismatch_count metric value emitted for this run (activates the
   540→  // existing 'reconciliation-mismatch' alert rule in alerting.ts).
   541→  mismatchCount    Int      @default(0)
   542→  // Error tracking (if the run itself failed)
   543→  lastError        String?
   544→  findings         ReconciliationFinding[]
   545→
   546→  @@index([status, startedAt])
   547→}
   548→
   549→model ReconciliationFinding {
   550→  id              String   @id @default(cuid())
   551→  runId           String
   552→  run             ReconciliationRun @relation(fields: [runId], references: [id], onDelete: Cascade)
   553→  // Mismatch class (M1-M17 from WAVE5_5B_GATE_REVIEW.md §2 D4)
   554→  // e.g. "M1_LEDGER_IMBALANCE", "M9_STUCK_CAPTURE_PENDING", "M10_STUCK_REFUND_PENDING"
   555→  mismatchClass   String
   556→  // Severity: CRITICAL | HIGH | MEDIUM
   557→  // CRITICAL/HIGH → routed to ExceptionQueue via reportInvariantViolation()
   558→  severity        String
   559→  // The affected entity type: "Payment" | "Refund" | "Outbox" | "WebhookEvent"
   560→  entityType      String
   561→  // The affected entity id (paymentId / refundId / outbox id / webhookEvent id)
   562→  entityId        String
   563→  // Full state snapshot at detection time (JSON — the evidence for the finding)
   564→  stateSnapshot   String
   565→  // Human-readable description of the mismatch
   566→  description     String
   567→  // Recommended remediation (informational only — NOT auto-applied)
   568→  recommendedRemediation String?
   569→  // ExceptionQueue entry id (set only for CRITICAL/HIGH findings routed via
   570→  // reportInvariantViolation()). NULL for MEDIUM findings (logged + recorded
   571→  // but not frozen).
   572→  exceptionId     String?
   573→  // Detection timestamps (idempotency: a second run for the same mismatch
   574→  // updates lastSeenAt, NOT a new row)
   575→  firstSeenAt     DateTime @default(now())
   576→  lastSeenAt      DateTime @default(now())
   577→  // Resolution tracking (admin action — manual, NOT automatic)
   578→  resolvedAt      DateTime?
   579→  resolvedBy      String?
   580→  resolutionNote  String?
   581→  // 5C: Remediation actions taken for this finding (1:N — a finding may be
   582→  // repaired multiple times if the first repair fails + is retried).
   583→  remediationActions RemediationAction[]
   584→
   585→  // Idempotency: one unresolved finding per (mismatchClass, entityId).
   586→  // A partial unique index on (mismatchClass, entityId) WHERE resolvedAt IS NULL
   587→  // is created in the migration SQL. Prisma's @unique creates a full unique
   588→  // index (including resolvedAt) which allows multiple resolved findings for
   589→  // the same (mismatchClass, entityId) over time (audit trail) while preventing
   590→  // duplicate unresolved findings.
   591→  @@unique([mismatchClass, entityId, resolvedAt])
   592→  @@index([severity, resolvedAt])
   593→  @@index([entityType, entityId])
   594→  @@index([runId])
   595→}

   597→// ----------------------------------------
   598→// P0-03 Wave-5 Sub-Wave 5C — Remediation Action (M16-only, Class-2 additive)
   599→// ----------------------------------------
   600→// Audit trail for remediation/repair actions. Currently ONLY M16 (outbox lag —
   601→// operational, non-financial) remediation is authorized. M3/M9/M10 + CLASS B/D/E
   602→// are NOT authorized — they require separate Orchestrator directives.
   603→//
   604→// SI-6 (Safety Invariant): every repair writes a RemediationAction audit record.
   605→// SI-2 (Repair Idempotency): the unique constraint on (findingId, repairType)
   606→// prevents duplicate repair actions for the same finding + repair type. A second
   607→// attempt to repair the same finding with the same repairType is a no-op (caught
   608→// by the unique constraint → the handler skips).
   609→//
   610→// SAFETY CONTRACT (Orchestrator hard boundary for 5C-M16):
   611→//   - M16 remediation NEVER writes to Payment, Refund, LedgerEntry, Outbox,
   612→//     WebhookEvent, IdempotencyKey, or AuditLog.
   613→//   - M16 remediation NEVER makes external Razorpay API calls.
   614→//   - M16 remediation's only "external" action is an HTTP call to the outbox
   615→//     publisher's /trigger endpoint (operational restart — no financial mutation).
   616→//   - M16 remediation's only DB writes are to this table (RemediationAction) +
   617→//     ReconciliationFinding (resolves the finding).
   618→model RemediationAction {
   619→  id              String   @id @default(cuid())
   620→  // The finding being repaired
   621→  findingId       String
   622→  finding         ReconciliationFinding @relation(fields: [findingId], references: [id], onDelete: Cascade)
   623→  // The repair type (e.g., "M16_PUBLISHER_TRIGGER"). Only M16 is authorized.
   624→  repairType      String
   625→  // The status of the repair action: ATTEMPTED | SUCCEEDED | SKIPPED | FAILED
   626→  status          String   @default("ATTEMPTED")
   627→  // What the repair did (JSON snapshot — e.g., { publisherTriggerCalled: true, lagBefore: 360, lagAfter: 45 })
   628→  actionSnapshot  String
   629→  // When the repair was attempted
   630→  attemptedAt     DateTime @default(now())
   631→  // When the repair completed (set on SUCCEEDED/SKIPPED/FAILED)
   632→  completedAt     DateTime?
   633→  // Error message if the repair FAILED
   634→  error           String?
   635→
   636→  // Idempotency: one repair action per (findingId, repairType). A second attempt
   637→  // to repair the same finding with the same repairType throws P2002 (unique
   638→  // constraint) — the handler catches this and skips (SI-2: Repair Idempotency).
   639→  @@unique([findingId, repairType])
   640→  @@index([status, attemptedAt])
   641→  @@index([repairType])
   642→}

   644→// ----------------------------------------
   645→// P0-06 Wave-6 — Fulfilment (parallel state machine — additive-only)
   646→// ----------------------------------------
   647→// 1:1 to Order. Parallel to Order.status (NEXT_STATUS) — does NOT modify
   648→// the existing Order state machine. Fulfilment has its own NEXT_FULFILMENT_STATUS
   649→// machine: PREPARING → ALMOST_READY → READY_FOR_PICKUP → PICKED_UP.
   650→//
   651→// Lifecycle:
   652→//   PREPARING        (default — created lazily on first access)
   653→//     → ALMOST_READY
   654→//     → READY_FOR_PICKUP
   655→//     → PICKED_UP    (terminal)
   656→//
   657→// Lazy-created: when GET/PATCH /api/orders/[id]/fulfilment is called and no
   658→// Fulfilment row exists yet, it is created with status='PREPARING' +
   659→// pickupOtp copied from Order.pickupOtp.
   660→//
   661→// Optimistic locking: every PATCH uses a conditional UPDATE
   662→// (`WHERE id = X AND version = expectedVersion`). A version mismatch returns
   663→// HTTP 409 (same pattern as Order.status PATCH — P0-25 Case B).
   664→//
   665→// P0-07 (pickup attribution): pickupOtp / pickupVerifiedAt / pickupVerifiedBy
   666→// fields are PRESENT but INACTIVE in this wave. No QR+OTP verification is
   667→// performed on the PICKED_UP transition; no RBAC is enforced on the
   668→// PICKED_UP actor. P0-07 activation is a separate Orchestrator directive.
   669→model Fulfilment {
   670→  id                String   @id @default(cuid())
   671→  // 1:1 to Order — @unique ensures at most one Fulfilment per Order.
   672→  orderId           String   @unique
   673→  order             Order    @relation(fields: [orderId], references: [id])
   674→  // Parallel state machine (NEXT_FULFILMENT_STATUS) — does NOT touch Order.status.
   675→  status            String   @default("PREPARING") // PREPARING | ALMOST_READY | READY_FOR_PICKUP | PICKED_UP
   676→  // JSON array of {status, at} — parallel to Order.statusHistory.
   677→  statusHistory     String   @default("[]")
   678→  // P0-25: Optimistic-locking version field. Incremented on every PATCH.
   679→  version           Int      @default(0)
   680→  // P0-07 future: pickup attribution fields (INACTIVE in this wave).
   681→  pickupOtp         String?  // copied from Order.pickupOtp on lazy-create
   682→  pickupVerifiedAt  DateTime? // set when pickup is verified (P0-07 — not yet enforced)
   683→  pickupVerifiedBy  String?  // actor who verified pickup (P0-07 — not yet enforced)
   684→  createdAt         DateTime @default(now())
   685→  updatedAt         DateTime @updatedAt
   686→
   687→  @@index([status])
   688→}
