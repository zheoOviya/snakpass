     3→// src/components/snak/screens/home-screen.tsx
     4→//
     5→// Home screen — premium consumer landing surface per blueprint §9.
     6→//
     7→// Sections (in render order, mobile-first):
     8→//  1. Campus context bar        — shows selected campus + "Change" affordance
     9→//  2. Search bar                — inline filter that scopes the Popular grid
    10→//  3. Quick Reorder             — horizontal carousel of recently ordered restaurants
    11→//  4. Open Now                  — horizontal carousel of isOpen===true restaurants
    12→//  5. Popular Near You          — 2-col grid of RestaurantCardV2 (top 6 by rating)
    13→//  6. Deals                     — horizontal carousel of restaurants with `deal` label
    14→//  7. Rewards Progress          — RewardProgressRing + "View rewards" CTA
    15→//  8. Friends Ordering Nearby   — SocialFeedCard list (last 3 ordered_from activities)
    16→//  9. Gift a Friend CTA         — violet accent card
    17→// 10. Start Group Order CTA     — rose accent card
    18→// 11. Recently Ordered          — vertical list of last 3 orders
    19→//
    20→// Governance (Task 2B):
    21→//  - Does NOT touch any API route — uses existing GET /api/restaurants,
    22→//    GET /api/orders?role=consumer, GET /api/social/feed, GET /api/rewards/account.
    23→//  - Uses Task 1B components: RestaurantCardV2, RewardProgressRing, EmptyState,
    24→//    SkeletonLoader, SocialFeedCard.
    25→//  - Uses Task 1C stores: useCampus, useSocial, useRewards.
    26→//  - Tapping a restaurant → ui-store.openRestaurant(id) — ConsumerView wires
    27→//    the actual restaurant-detail overlay (Task 2D's screen via dynamic import).
    28→

    66→// ─────────────────────────────────────────────────────────────────────────────
    67→// Local types — additive Wave 2C fields on Restaurant (Task 2C owns types.ts
    68→// extension; we declare a local additive interface here so the file compiles
    69→// even before 2C lands).
    70→// ─────────────────────────────────────────────────────────────────────────────
    71→
    72→type RestaurantWithExtras = Restaurant & {
    73→  /** Wave 2C additive — reward multiplier (default 1). */
    74→  rewardMultiplier?: number
    75→  /** Wave 2C additive — derived open/closed state. */
    76→  isOpen?: boolean
    77→  /** Wave 2C additive — derived "Great value" deal label, if any. */
    78→  deal?: string | null
    79→}
    80→
    81→type RestaurantFetchState = {
    82→  data: RestaurantWithExtras[]
    83→  loading: boolean
    84→  error: string | null
    85→}
    86→
    87→type OrdersFetchState = {
    88→  data: Order[]
    89→  loading: boolean
    90→  error: string | null
    91→}

    98→  hidden: {},
    99→  show: {
   100→    transition: { staggerChildren: 0.03, delayChildren: 0.04 },
   101→  },
   102→}
   103→
   104→const SECTION_ITEM: Variants = {
   105→  hidden: { opacity: 0, y: 8 },
   106→  show: {
   107→    opacity: 1,
   108→    y: 0,
   109→    transition: { duration: 0.28, ease: [0.3, 0, 0, 1] },
   110→  },
   111→}

   113→// ─────────────────────────────────────────────────────────────────────────────
   114→// HomeScreen — public surface
   115→// ─────────────────────────────────────────────────────────────────────────────
   116→
   117→export interface HomeScreenProps {
   118→  /** Called when the user pulls to refresh. Defaults to a no-op. */
   119→  onRefresh?: () => Promise<void> | void
   120→}
   121→
   122→export function HomeScreen({ onRefresh }: HomeScreenProps) {
   123→  const prefersReduced = useReducedMotion()
   124→  const { toast } = useToast()
   125→  const { user } = useAuth()
   126→  const openRestaurant = useUI((s) => s.openRestaurant)
   127→  const setActiveTab = useUI((s) => s.setActiveTab)
   128→
   129→  const selectedCampusId = useCampus((s) => s.selectedCampusId)
   130→  const selectedCampusName = useCampus((s) => s.selectedCampusName)
   131→
   132→  // ── Restaurants ──────────────────────────────────────────────────────────
   133→  // Fetched once on mount, scoped by campus when selected. We do NOT
   134→  // re-fetch on inline search — that's a client-side filter for snappy UX.
   135→  const [restaurants, setRestaurants] = React.useState<RestaurantFetchState>({
   136→    data: [],
   137→    loading: true,
   138→    error: null,
   139→  })
   140→  const [q, setQ] = React.useState('')
   141→
   142→  const fetchRestaurants = React.useCallback(async () => {
   143→    setRestaurants((s) => ({ ...s, loading: true, error: null }))
   144→    try {
   145→      const url = selectedCampusId
   146→        ? `/api/campuses/${encodeURIComponent(selectedCampusId)}/restaurants`
   147→        : '/api/restaurants'
   148→      const res = await fetch(url, { cache: 'no-store' })
   149→      const data = (await res.json().catch(() => ({}))) as {
   150→        restaurants?: RestaurantWithExtras[]
   151→        error?: { message?: string }
   152→      }
   153→      if (!res.ok) {
   154→        throw new Error(data?.error?.message ?? `Failed (${res.status})`)
   155→      }
   156→      setRestaurants({ data: data.restaurants ?? [], loading: false, error: null })
   157→    } catch (e) {
   158→      const msg = (e as Error).message
   159→      setRestaurants({ data: [], loading: false, error: msg })
   160→      toast({
   161→        title: 'Could not load restaurants',
   162→        description: msg,
   163→        variant: 'destructive',
   164→      })
   165→    }
   166→  }, [selectedCampusId, toast])
   167→
   168→  // ── My orders (recent) ────────────────────────────────────────────────────
   169→  const [orders, setOrders] = React.useState<OrdersFetchState>({
   170→    data: [],
   171→    loading: true,
   172→    error: null,
   173→  })
   174→
   175→  const fetchOrders = React.useCallback(async () => {
   176→    setOrders((s) => ({ ...s, loading: true, error: null }))
   177→    try {
   178→      const res = await fetch('/api/orders?role=consumer&limit=10', { cache: 'no-store' })
   179→      const data = (await res.json().catch(() => ({}))) as {
   180→        orders?: Order[]
   181→        error?: { message?: string }
   182→      }
   183→      if (!res.ok) {
   184→        throw new Error(data?.error?.message ?? `Failed (${res.status})`)
   185→      }
   186→      setOrders({ data: data.orders ?? [], loading: false, error: null })
   187→    } catch {
   188→      // Soft-fail — Home should still render even if /api/orders 401s (e.g.,
   189→      // a stale session that's about to refresh). Empty state will surface.
   190→      setOrders({ data: [], loading: false, error: 'unavailable' })
   191→    }
   192→  }, [])
   193→
   194→  // ── Social feed (Friends Ordering Nearby) ────────────────────────────────
   195→  // We use the social-store's `feed` slice — Task 1C's useSocial already
   196→  // fetches /api/social/feed on consumer-view mount. We just read + filter it.
   197→  const socialFeed = useSocial((s) => s.feed)
   198→  const socialConnections = useSocial((s) => s.connections)
   199→  const socialLoading = useSocial((s) => s.isLoading)
   200→  const refreshSocial = useSocial((s) => s.refresh)
   201→
   202→  // ── Rewards ──────────────────────────────────────────────────────────────
   203→  const rewardsAccount = useRewards((s) => s.account)
   204→  const rewardsLoading = useRewards((s) => s.isLoading)
   205→  const refreshRewards = useRewards((s) => s.refresh)
   206→
   207→  // ── Initial load + whenever the campus changes ───────────────────────────
   208→  React.useEffect(() => {
   209→    fetchRestaurants()
   210→    fetchOrders()
   211→    refreshSocial().catch(() => {
   212→      /* best-effort */
   213→    })
   214→    if (user?.userId) {
   215→      refreshRewards(user.userId).catch(() => {
   216→        /* best-effort */
   217→      })
   218→    }
   219→  }, [fetchRestaurants, fetchOrders, refreshSocial, refreshRewards, user?.userId])
   220→
   221→  // ── Pull-to-refresh handler ──────────────────────────────────────────────
   222→  const [refreshing, setRefreshing] = React.useState(false)
   223→  const handleRefresh = React.useCallback(async () => {
   224→    setRefreshing(true)
   225→    try {
   226→      await Promise.all([
   227→        fetchRestaurants(),
   228→        fetchOrders(),
   229→        refreshSocial(),
   230→        user?.userId ? refreshRewards(user.userId) : Promise.resolve(),
   231→      ])
   232→      await onRefresh?.()
   233→    } finally {
   234→      setRefreshing(false)
   235→    }
   236→  }, [
   237→    fetchRestaurants,
   238→    fetchOrders,
   239→    refreshSocial,
   240→    refreshRewards,
   241→    onRefresh,
   242→    user?.userId,
   243→  ])
   244→
   245→  // ── Derived section data ─────────────────────────────────────────────────
   246→  // Restaurants matching the inline search query (client-side filter — snappy).
   247→  const filteredRestaurants = React.useMemo(() => {
   248→    const query = q.trim().toLowerCase()
   249→    if (!query) return restaurants.data
   250→    return restaurants.data.filter(
   251→      (r) =>
   252→        r.name.toLowerCase().includes(query) ||
   253→        r.cuisine.toLowerCase().includes(query) ||
   254→        r.description.toLowerCase().includes(query),
   255→    )
   256→  }, [restaurants.data, q])
   257→
   258→  // Quick Reorder: unique restaurant ids from the user's recent orders, capped
   259→  // at 5. We show their cards as a horizontal carousel so the user can jump
   260→  // straight back into their favourites.
   261→  const quickReorderRestaurants = React.useMemo(() => {
   262→    const seen = new Set<string>()
   263→    const out: RestaurantWithExtras[] = []
   264→    for (const o of orders.data) {
   265→      const rid = o.restaurant.id
   266→      if (seen.has(rid)) continue
   267→      const match = restaurants.data.find((r) => r.id === rid)
   268→      if (match) {
   269→        seen.add(rid)
   270→        out.push(match)
   271→      }
   272→      if (out.length >= 5) break
   273→    }
   274→    return out
   275→  }, [orders.data, restaurants.data])
   276→
   277→  const openNowRestaurants = React.useMemo(
   278→    () => filteredRestaurants.filter((r) => r.isOpen !== false).slice(0, 8),
   279→    [filteredRestaurants],
   280→  )
   281→
   282→  const popularRestaurants = React.useMemo(
   283→    () => filteredRestaurants.slice(0, 6),
   284→    [filteredRestaurants],
   285→  )
   286→
   287→  const dealsRestaurants = React.useMemo(
   288→    () => filteredRestaurants.filter((r) => !!r.deal).slice(0, 8),
   289→    [filteredRestaurants],
   290→  )
   291→
   292→  const friendsOrdering = React.useMemo<SocialActivity[]>(
   293→    () =>
   294→      socialFeed
   295→        .filter((a) => a.verb === 'ordered_from')
   296→        .slice(0, 3),
   297→    [socialFeed],
   298→  )
   299→
   300→  const recentOrders = React.useMemo(() => orders.data.slice(0, 3), [orders.data])
   301→
   302→  const hasFriends = socialConnections.length > 0
   303→  const hasRewardsAccount = !!rewardsAccount
   304→
   305→  // ── Render helpers ────────────────────────────────────────────────────────
   306→  function handleRestaurantTap(r: Restaurant) {
   307→    openRestaurant(r.id)
   308→  }
   309→
   310→  function handleReorderTap(r: Restaurant) {
   311→    openRestaurant(r.id)
   312→  }
   313→
   314→  // ═══════════════════════════════════════════════════════════════════════════
   315→  // RENDER
   316→  // ═══════════════════════════════════════════════════════════════════════════
   317→  return (
   318→    <PullToRefresh onRefresh={handleRefresh} refreshing={refreshing}>
   319→      <motion.div
   320→        variants={SECTION_CONTAINER}
   321→        initial={prefersReduced ? false : 'hidden'}
   322→        animate="show"
   323→        className="mx-auto w-full max-w-6xl space-y-8 px-4 py-5 pb-24"
   324→      >
   325→        {/* ────────────────────────────────────────────────────────────────────
   326→            1. Campus context bar
   327→            ──────────────────────────────────────────────────────────────────── */}
   328→        <motion.section variants={SECTION_ITEM} aria-label="Campus context">
   329→          <div className="flex items-center justify-between gap-3">
   330→            <div className="flex min-w-0 items-center gap-2">
   331→              {selectedCampusName ? (
   332→                <CampusBadge name={selectedCampusName} />
   333→              ) : (
   334→                <span className="inline-flex items-center gap-1 rounded-full bg-secondary px-2.5 py-1 text-xs font-medium text-secondary-foreground">
   335→                  <MapPin className="h-3.5 w-3.5 text-primary" aria-hidden="true" />
   336→                  All campuses
   337→                </span>
   338→              )}
   339→            </div>
   340→            <Button
   341→              type="button"
   342→              size="sm"
   343→              variant="ghost"
   344→              className="h-7 gap-1 px-2 text-xs text-muted-foreground"
   345→              onClick={() => setActiveTab('explore')}
   346→            >
   347→              Change
   348→              <ChevronRight className="h-3.5 w-3.5" aria-hidden="true" />
   349→            </Button>
   350→          </div>
   351→        </motion.section>
   352→
   353→        {/* ────────────────────────────────────────────────────────────────────
   354→            2. Search bar — inline filter
   355→            ──────────────────────────────────────────────────────────────────── */}
   356→        <motion.section variants={SECTION_ITEM} aria-label="Search">
   357→          <div className="relative">
   358→            <Search
   359→              className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   360→              aria-hidden="true"
   361→            />
   362→            <Input
   363→              type="search"
   364→              value={q}
   365→              onChange={(e) => setQ(e.target.value)}
   366→              placeholder="Search restaurants, cuisines…"
   367→              aria-label="Search restaurants and cuisines"
   368→              className="h-11 rounded-2xl border-border/80 bg-card pl-10 pr-10 text-sm shadow-sm"
   369→            />
   370→            {q && (
   371→              <button
   372→                type="button"
   373→                onClick={() => setQ('')}
   374→                aria-label="Clear search"
   375→                className="absolute right-3 top-1/2 -translate-y-1/2 text-xs text-muted-foreground hover:text-foreground"
   376→              >
   377→                Clear
   378→              </button>
   379→            )}
   380→          </div>
   381→        </motion.section>
   382→
   383→        {/* ────────────────────────────────────────────────────────────────────
   384→            3. Quick Reorder — horizontal carousel of recently ordered restaurants
   385→            ──────────────────────────────────────────────────────────────────── */}
   386→        <section aria-labelledby="home-quick-reorder">
   387→          <SectionHeader id="home-quick-reorder" title="Quick Reorder" icon="🔄" />
   388→          {orders.loading ? (
   389→            <HorizontalScroll>
   390→              {Array.from({ length: 3 }).map((_, i) => (
   391→                <div key={i} className="w-64 shrink-0">
   392→                  <RestaurantCardSkeleton />
   393→                </div>
   394→              ))}
   395→            </HorizontalScroll>
   396→          ) : quickReorderRestaurants.length === 0 ? (
   397→            <EmptyState
   398→              variant="no-orders"
   399→              title="No recent orders yet"
   400→              description="Your favourite restaurants will appear here for one-tap reorder."
   401→              className="py-8"
   402→            />
   403→          ) : (
   404→            <HorizontalScroll>
   405→              {quickReorderRestaurants.map((r) => (
   406→                <div key={r.id} className="w-64 shrink-0">
   407→                  <RestaurantCardV2
   408→                    restaurant={r}
   409→                    isOpen={r.isOpen !== false}
   410→                    rewardMultiplier={r.rewardMultiplier}
   411→                    dealLabel={r.deal ?? undefined}
   412→                    onPress={handleReorderTap}
   413→                  />
   414→                </div>
   415→              ))}
   416→            </HorizontalScroll>
   417→          )}
   418→        </section>
   419→
   420→        {/* ────────────────────────────────────────────────────────────────────
   421→            4. Open Now — horizontal carousel
   422→            ──────────────────────────────────────────────────────────────────── */}
   423→        <section aria-labelledby="home-open-now">
   424→          <SectionHeader id="home-open-now" title="Open Now" icon="🟢" />
   425→          {restaurants.loading ? (
   426→            <HorizontalScroll>
   427→              {Array.from({ length: 3 }).map((_, i) => (
   428→                <div key={i} className="w-64 shrink-0">
   429→                  <RestaurantCardSkeleton />
   430→                </div>
   431→              ))}
   432→            </HorizontalScroll>
   433→          ) : openNowRestaurants.length === 0 ? (
   434→            <EmptyState
   435→              variant="no-restaurants"
   436→              title="No restaurants open near you"
   437→              description="Try a different campus or check back in a few minutes."
   438→              className="py-8"
   439→            />
   440→          ) : (
   441→            <HorizontalScroll>
   442→              {openNowRestaurants.map((r) => (
   443→                <div key={r.id} className="w-64 shrink-0">
   444→                  <RestaurantCardV2
   445→                    restaurant={r}
   446→                    isOpen={r.isOpen !== false}
   447→                    rewardMultiplier={r.rewardMultiplier}
   448→                    dealLabel={r.deal ?? undefined}
   449→                    onPress={handleRestaurantTap}
   450→                  />
   451→                </div>
   452→              ))}
   453→            </HorizontalScroll>
   454→          )}
   455→        </section>
   456→
   457→        {/* ────────────────────────────────────────────────────────────────────
   458→            5. Popular Near You — grid of RestaurantCardV2
   459→            ──────────────────────────────────────────────────────────────────── */}
   460→        <section aria-labelledby="home-popular">
   461→          <SectionHeader id="home-popular" title="Popular Near You" icon="🔥" />
   462→          {restaurants.loading ? (
   463→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   464→              {Array.from({ length: 6 }).map((_, i) => (
   465→                <RestaurantCardSkeleton key={i} />
   466→              ))}
   467→            </div>
   468→          ) : popularRestaurants.length === 0 ? (
   469→            <EmptyState
   470→              variant="no-restaurants"
   471→              className="py-8"
   472→              actionLabel="Switch campus"
   473→              onAction={() => setActiveTab('explore')}
   474→            />
   475→          ) : (
   476→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   477→              {popularRestaurants.map((r) => (
   478→                <RestaurantCardV2
   479→                  key={r.id}
   480→                  restaurant={r}
   481→                  isOpen={r.isOpen !== false}
   482→                  rewardMultiplier={r.rewardMultiplier}
   483→                  dealLabel={r.deal ?? undefined}
   484→                  onPress={handleRestaurantTap}
   485→                />
   486→              ))}
   487→            </div>
   488→          )}
   489→        </section>
   490→
   491→        {/* ────────────────────────────────────────────────────────────────────
   492→            6. Deals — horizontal carousel
   493→            ──────────────────────────────────────────────────────────────────── */}
   494→        {dealsRestaurants.length > 0 && (
   495→          <section aria-labelledby="home-deals">
   496→            <SectionHeader id="home-deals" title="Deals" icon="⚡" />
   497→            <HorizontalScroll>
   498→              {dealsRestaurants.map((r) => (
   499→                <div key={r.id} className="w-64 shrink-0">
   500→                  <RestaurantCardV2
   501→                    restaurant={r}
   502→                    isOpen={r.isOpen !== false}
   503→                    rewardMultiplier={r.rewardMultiplier}
   504→                    dealLabel={r.deal ?? undefined}
   505→                    onPress={handleRestaurantTap}
   506→                  />
   507→                </div>
   508→              ))}
   509→            </HorizontalScroll>
   510→          </section>
   511→        )}
   512→
   513→        {/* ────────────────────────────────────────────────────────────────────
   514→            7. Rewards Progress — RewardProgressRing + "View rewards" CTA
   515→            ──────────────────────────────────────────────────────────────────── */}
   516→        <section aria-labelledby="home-rewards">
   517→          <SectionHeader id="home-rewards" title="Rewards Progress" icon="✨" />
   518→          <Card className="overflow-hidden border-gold-200 bg-gradient-to-br from-gold-50 to-amber-50 dark:border-gold-900/40 dark:from-gold-950/30 dark:to-amber-950/20">
   519→            <CardContent className="flex flex-col items-start gap-4 p-5 sm:flex-row sm:items-center">
   520→              {rewardsLoading ? (
   521→                <RewardRingSkeleton />
   522→              ) : hasRewardsAccount ? (
   523→                <RewardProgressRing
   524→                  points={rewardsAccount.pointsBalance}
   525→                  size={96}
   526→                  earnRate="You earn points on every order"
   527→                />
   528→              ) : (
   529→                <div className="flex w-full flex-col items-start gap-3">
   530→                  <div className="flex h-20 w-20 items-center justify-center rounded-full bg-gold-100 dark:bg-gold-950/60">
   531→                    <Sparkles className="h-8 w-8 text-gold-600 dark:text-gold-400" aria-hidden="true" />
   532→                  </div>
   533→                  <p className="text-sm text-muted-foreground">
   534→                    Earn points on every order and unlock tier-based rewards.
   535→                  </p>
   536→                </div>
   537→              )}
   538→              <div className="flex flex-1 flex-col gap-2 sm:items-end">
   539→                <Button
   540→                  type="button"
   541→                  variant="outline"
   542→                  className="border-gold-300 text-gold-800 hover:bg-gold-100 dark:border-gold-800 dark:text-gold-300 dark:hover:bg-gold-950/60"
   543→                  onClick={() => setActiveTab('rewards')}
   544→                >

   760→// ═══════════════════════════════════════════════════════════════════════════
   761→// PullToRefresh — minimal pull-to-refresh wrapper (touch pointer events only).
   762→// We use a simple threshold-based pull on touchstart/touchmove/touchend. The
   763→// page doesn't scroll mid-pull (we only engage when scrollTop === 0). When
   764→// the threshold (70px) is exceeded, onRefresh fires after release.
   765→// ═══════════════════════════════════════════════════════════════════════════
   766→
   767→interface PullToRefreshProps {
   768→  onRefresh: () => Promise<void> | void
   769→  refreshing: boolean
   770→  children: React.ReactNode
   771→}
   772→
   773→const PULL_THRESHOLD = 70
   774→
   775→function PullToRefresh({ onRefresh, refreshing, children }: PullToRefreshProps) {
   776→  const startY = React.useRef<number | null>(null)
   777→  const [pull, setPull] = React.useState(0)
   778→  // `isPulling` tracks whether the user is actively dragging (touchstart fired
   779→  // and touchend hasn't). We use state (not a ref) because the transition
   780→  // style depends on it — refs accessed during render are an eslint violation.
   781→  const [isPulling, setIsPulling] = React.useState(false)
   782→  const containerRef = React.useRef<HTMLDivElement | null>(null)
   783→
   784→  function onTouchStart(e: React.TouchEvent) {
   785→    // Only engage if the user is at the very top of the scroll container.
   786→    const target = containerRef.current
   787→    if (!target || target.scrollTop > 0) {
   788→      startY.current = null
   789→      setIsPulling(false)
   790→      return
   791→    }
   792→    startY.current = e.touches[0]?.clientY ?? null
   793→    setIsPulling(true)
   794→  }
   795→
   796→  function onTouchMove(e: React.TouchEvent) {
   797→    if (startY.current === null || refreshing) return
   798→    const dy = (e.touches[0]?.clientY ?? 0) - startY.current
   799→    if (dy > 0) {
   800→      // Rubber-band: dampen the visual pull so it feels elastic.
   801→      const dampened = Math.min(120, dy * 0.5)
   802→      setPull(dampened)
   803→    }
   804→  }
   805→
   806→  function onTouchEnd() {
   807→    if (startY.current === null) return
   808→    startY.current = null
   809→    setIsPulling(false)
   810→    if (pull >= PULL_THRESHOLD && !refreshing) {
   811→      void onRefresh()
   812→    }
   813→    setPull(0)
   814→  }
   815→
   816→  return (
   817→    <div
   818→      ref={containerRef}
   819→      onTouchStart={onTouchStart}
   820→      onTouchMove={onTouchMove}
   821→      onTouchEnd={onTouchEnd}
   822→      className="relative"
   823→    >
   824→      {/* Pull indicator — only visible while actively pulling or refreshing */}
   825→      {(pull > 4 || refreshing) && (
   826→        <div
   827→          className="pointer-events-none absolute left-1/2 top-2 z-10 -translate-x-1/2"
   828→          aria-hidden="true"
   829→        >
   830→          <RefreshCw
   831→            className={`h-5 w-5 text-muted-foreground ${refreshing ? 'animate-spin' : ''}`}
   832→            style={{ transform: `rotate(${Math.min(180, pull * 2)}deg)` }}
   833→          />
   834→        </div>
   835→      )}
   836→      <div
   837→        style={{
   838→          transform: `translateY(${refreshing ? 24 : pull}px)`,
   839→          transition: isPulling ? 'none' : 'transform 200ms ease',
   840→        }}
   841→      >
   842→        {children}
   843→      </div>
   844→    </div>
   845→  )
   846→}
