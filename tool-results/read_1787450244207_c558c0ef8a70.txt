    11→    11→//      • Outgoing (status === 'PENDING_OUT') — "Pending" label
    12→    12→//  - Current friends list (status === 'ACCEPTED'):
    13→    13→//      avatar + name + phone + "Message" (placeholder toast) + "Unfriend" button
    14→    14→//  - Search results: each row shows "Add friend" button (calls sendRequest).

    19→    19→//    store — we CALL its methods, never modify it).
    20→    20→//  - Search results are fetched locally (the social-store has no `search`
    21→    21→//    method by design — search is a transient, non-cached query).
    22→    22→//
    23→    23→// Governance (Task 6B):
    24→    24→//  - Does NOT touch any API route (Task 6A owns /api/social/**).
    25→    25→//  - Does NOT touch social-store.ts (Task 1C owns).
    26→    26→//  - Does NOT touch consumer-view.tsx (Task 3A owns the tab routing).
    27→    27→//  - Resilient to missing endpoints — surfaces a soft "Couldn't load" toast.

    93→    93→function initials(name: string): string {
    94→    94→  return name
    95→    95→    .split(' ')
    96→    96→    .map((p) => p[0])
    97→    97→    .filter(Boolean)
    98→    98→    .slice(0, 2)
    99→    99→    .join('')
   100→   100→    .toUpperCase()
   101→   101→}

   106→   106→
   107→   107→export function FriendsScreen({ className }: FriendsScreenProps) {
   108→   108→  const prefersReduced = useReducedMotion()
   109→   109→  const { toast } = useToast()
   110→   110→
   111→   111→  // ── social-store hooks (Task 1C) ─────────────────────────────────────────
   112→   112→  const connections = useSocial((s) => s.connections)
   113→   113→  const isLoading = useSocial((s) => s.isLoading)
   114→   114→  const refresh = useSocial((s) => s.refresh)
   115→   115→  const sendRequest = useSocial((s) => s.sendRequest)
   116→   116→  const acceptRequest = useSocial((s) => s.acceptRequest)
   117→   117→  const declineRequest = useSocial((s) => s.declineRequest)
   118→   118→  const unfollow = useSocial((s) => s.unfollow)
   119→   119→
   120→   120→  // ── Derived slices ─────────────────────────────────────────────────────────
   121→   121→  const pendingIn = React.useMemo(
   122→   122→    () => connections.filter((c) => c.status === 'PENDING_IN'),
   123→   123→    [connections],
   124→   124→  )
   125→   125→  const pendingOut = React.useMemo(
   126→   126→    () => connections.filter((c) => c.status === 'PENDING_OUT'),
   127→   127→    [connections],
   128→   128→  )
   129→   129→  const friends = React.useMemo(
   130→   130→    () =>
   131→   131→      connections
   132→   132→        .filter((c) => c.status === 'ACCEPTED')
   133→   133→        // newest accepted first (acceptedAt ?? createdAt fallback)
   134→   134→        .sort((a, b) => {
   135→   135→          const at = (t?: string) => (t ? new Date(t).getTime() : 0)
   136→   136→          return at(b.acceptedAt) - at(a.acceptedAt)
   137→   137→        }),
   138→   138→    [connections],
   139→   139→  )
   140→   140→
   141→   141→  // Set of friend IDs (any status) for client-side exclusion of search results.
   142→   142→  const connectedIds = React.useMemo(
   143→   143→    () => new Set(connections.map((c) => c.friendId)),
   144→   144→    [connections],
   145→   145→  )
   146→   146→
   147→   147→  // ── Search state (local — not in social-store by design) ───────────────────
   148→   148→  const [query, setQuery] = React.useState('')
   149→   149→  const [debounced, setDebounced] = React.useState('')
   150→   150→  const [results, setResults] = React.useState<SocialSearchUser[]>([])
   151→   151→  const [searching, setSearching] = React.useState(false)
   152→   152→  const [searchError, setSearchError] = React.useState<string | null>(null)
   153→   153→  // Track outgoing requests that succeeded locally so we can flip the row's
   154→   154→  // button to "Pending" without waiting for a refresh.
   155→   155→  const [pendingSentIds, setPendingSentIds] = React.useState<Set<string>>(new Set())
   156→   156→
   157→   157→  // 250ms debounce — only triggers a fetch when query length ≥ 2.
   158→   158→  React.useEffect(() => {
   159→   159→    const q = query.trim()
   160→   160→    if (q === debounced) return
   161→   161→    const t = setTimeout(() => setDebounced(q), 250)
   162→   162→    return () => clearTimeout(t)
   163→   163→  }, [query, debounced])
   164→   164→
   165→   165→  // Fetch search results whenever the debounced query changes.
   166→   166→  React.useEffect(() => {
   167→   167→    const q = debounced.trim()
   168→   168→    if (q.length < 2) {
   169→   169→      setResults([])
   170→   170→      setSearchError(null)
   171→   171→      setSearching(false)
   172→   172→      return
   173→   173→    }
   174→   174→    let cancelled = false
   175→   175→    setSearching(true)
   176→   176→    setSearchError(null)
   177→   177→    fetch(`/api/social/search?q=${encodeURIComponent(q)}`, {
   178→   178→      headers: { 'Content-Type': 'application/json' },
   179→   179→    })
   180→   180→      .then(async (res) => {
   181→   181→        if (!res.ok) {
   182→   182→          const body = await res.json().catch(() => ({}))
   183→   183→          throw new Error(body?.error || `Search failed (${res.status})`)
   184→   184→        }
   185→   185→        const data = (await res.json()) as { users?: SocialSearchUser[] }
   186→   186→        if (!cancelled) {
   187→   187→          setResults(data.users ?? [])
   188→   188→        }
   189→   189→      })
   190→   190→      .catch((err) => {
   191→   191→        if (!cancelled) {
   192→   192→          setSearchError(err instanceof Error ? err.message : 'Search failed')
   193→   193→          setResults([])
   194→   194→        }
   195→   195→      })
   196→   196→      .finally(() => {
   197→   197→        if (!cancelled) setSearching(false)
   198→   198→      })
   199→   199→    return () => {
   200→   200→      cancelled = true
   201→   201→    }
   202→   202→  }, [debounced])
   203→   203→
   204→   204→  // ── Action handlers ─────────────────────────────────────────────────────────
   205→   205→  async function handleAccept(conn: SocialConnection) {
   206→   206→    try {
   207→   207→      await acceptRequest(conn.id)
   208→   208→      toast({
   209→   209→        title: 'Friend request accepted',
   210→   210→        description: `${conn.friendName} is now your friend.`,
   211→   211→      })
   212→   212→    } catch (e) {
   213→   213→      toast({
   214→   214→        title: 'Could not accept request',
   215→   215→        description: (e as Error).message,
   216→   216→        variant: 'destructive',
   217→   217→      })
   218→   218→    }
   219→   219→  }
   220→   220→
   221→   221→  async function handleReject(conn: SocialConnection) {
   222→   222→    try {
   223→   223→      await declineRequest(conn.id)
   224→   224→      toast({
   225→   225→        title: 'Request declined',
   226→   226→        description: `${conn.friendName}'s request was dismissed.`,
   227→   227→      })
   228→   228→    } catch (e) {
   229→   229→      toast({
   230→   230→        title: 'Could not decline request',
   231→   231→        description: (e as Error).message,
   232→   232→        variant: 'destructive',
   233→   233→      })
   234→   234→    }
   235→   235→  }
   236→   236→
   237→   237→  async function handleUnfriend(conn: SocialConnection) {
   238→   238→    // Optimistic confirmation — friend removal is destructive.
   239→   239→    const ok = window.confirm(
   240→   240→      `Unfriend ${conn.friendName}? They won't be notified, but you'll need to send a new request to reconnect.`,
   241→   241→    )
   242→   242→    if (!ok) return
   243→   243→    try {
   244→   244→      await unfollow(conn.friendId)
   245→   245→      toast({
   246→   246→        title: 'Unfriended',
   247→   247→        description: `${conn.friendName} is no longer your friend.`,
   248→   248→      })
   249→   249→    } catch (e) {
   250→   250→      toast({
   251→   251→        title: 'Could not unfriend',
   252→   252→        description: (e as Error).message,
   253→   253→        variant: 'destructive',
   254→   254→      })
   255→   255→    }
   256→   256→  }
   257→   257→
   258→   258→  async function handleSendMessage(conn: SocialConnection) {
   259→   259→    // Placeholder — messaging is on the Wave 8+ roadmap (blueprint §22).
   260→   260→    toast({
   261→   261→      title: 'Messaging coming soon',
   262→   262→      description: `You'll be able to message ${conn.friendName} in a future release.`,
   263→   263→    })
   264→   264→  }
   265→   265→
   266→   266→  async function handleAddFriend(user: SocialSearchUser) {
   267→   267→    try {
   268→   268→      await sendRequest(user.id)
   269→   269→      setPendingSentIds((s) => new Set(s).add(user.id))
   270→   270→      toast({
   271→   271→        title: 'Friend request sent',
   272→   272→        description: `Waiting for ${user.name} to accept.`,
   273→   273→      })
   274→   274→    } catch (e) {
   275→   275→      toast({
   276→   276→        title: 'Could not send request',
   277→   277→        description: (e as Error).message,
   278→   278→        variant: 'destructive',
   279→   279→      })
   280→   280→    }
   281→   281→  }
   282→   282→
   283→   283→  function handleRefresh() {
   284→   284→    refresh().catch(() => {
   285→   285→      /* best-effort — store surfaces its own error */
   286→   286→    })
   287→   287→  }
   288→   288→
   289→   289→  const hasSearchQuery = debounced.trim().length >= 2
   290→   290→  const filteredResults = React.useMemo(
   291→   291→    () => results.filter((u) => !connectedIds.has(u.id) && !pendingSentIds.has(u.id)),
   292→   292→    [results, connectedIds, pendingSentIds],
   293→   293→  )
   294→   294→
   295→   295→  // ── Loading skeleton ─────────────────────────────────────────────────────────
   296→   296→  if (isLoading && connections.length === 0) {
   297→   297→    return (
   298→   298→      <div className="space-y-3" role="status" aria-label="Loading friends">
   299→   299→        <span className="sr-only">Loading your friends…</span>
   300→   300→        {Array.from({ length: 4 }).map((_, i) => (
   301→   301→          <SocialFeedSkeleton key={i} />
   302→   302→        ))}
   303→   303→      </div>
   304→   304→    )
   305→   305→  }
   306→   306→
   307→   307→  return (
   308→   308→    <div className={cn('space-y-5', className)}>
   309→   309→      {/* ── Search bar ─────────────────────────────────────────────────────── */}
   310→   310→      <div className="relative">
   311→   311→        <Search
   312→   312→          className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   313→   313→          aria-hidden="true"
   314→   314→        />
   315→   315→        <Input
   316→   316→          type="search"
   317→   317→          value={query}
   318→   318→          onChange={(e) => setQuery(e.target.value)}
   319→   319→          placeholder="Search by name or phone…"
   320→   320→          aria-label="Search for friends"
   321→   321→          className="h-11 rounded-xl pl-9 pr-9"
   322→   322→        />
   323→   323→        {query && (
   324→   324→          <button
   325→   325→            type="button"
   326→   326→            onClick={() => setQuery('')}
   327→   327→            aria-label="Clear search"
   328→   328→            className="snak-focus-ring absolute right-2.5 top-1/2 -translate-y-1/2 rounded-full p-1 text-muted-foreground transition hover:bg-muted"
   329→   329→          >
   330→   330→            <X className="h-4 w-4" aria-hidden="true" />
   331→   331→          </button>
   332→   332→        )}
   333→   333→      </div>
   334→   334→
   335→   335→      {/* ── SEARCH RESULTS mode ─────────────────────────────────────────────── */}
   336→   336→      {hasSearchQuery && (
   337→   337→        <section aria-label="Search results" className="space-y-2">
   338→   338→          <header className="flex items-center justify-between gap-2">
   339→   339→            <h2 className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   340→   340→              {searching
   341→   341→                ? 'Searching…'
   342→   342→                : `${filteredResults.length} ${filteredResults.length === 1 ? 'result' : 'results'}`}
   343→   343→            </h2>
   344→   344→          </header>
   345→   345→
   346→   346→          {searching ? (
   347→   347→            <div className="space-y-2">
   348→   348→              {Array.from({ length: 3 }).map((_, i) => (
   349→   349→                <SocialFeedSkeleton key={i} />
   350→   350→              ))}
   351→   351→            </div>
   352→   352→          ) : searchError ? (
   353→   353→            <Card>
   354→   354→              <CardContent className="flex items-center gap-2 p-4 text-sm text-muted-foreground">
   355→   355→                <AlertCircle className="h-4 w-4 text-danger-600" aria-hidden="true" />
   356→   356→                <span>{searchError}</span>
   357→   357→              </CardContent>
   358→   358→            </Card>
   359→   359→          ) : filteredResults.length === 0 ? (
   360→   360→            <EmptyState
   361→   361→              variant="no-friends"
   362→   362→              title="No users found"
   363→   363→              description={
   364→   364→                results.length === 0
   365→   365→                  ? `No one matching "${debounced}". Try a different name or phone.`
   366→   366→                  : 'Everyone matching your search is already your friend or has a pending request.'
   367→   367→              }
   368→   368→              className="py-8"
   369→   369→            />
   370→   370→          ) : (
   371→   371→            <motion.ul
   372→   372→              variants={LIST_CONTAINER}
   373→   373→              initial={prefersReduced ? false : 'hidden'}
   374→   374→              animate="show"
   375→   375→              className="space-y-2"
   376→   376→            >
   377→   377→              <AnimatePresence>
   378→   378→                {filteredResults.map((u) => (
   379→   379→                  <motion.li key={u.id} variants={LIST_ITEM} exit={{ opacity: 0 }}>
   380→   380→                    <SearchResultRow
   381→   381→                      user={u}
   382→   382→                      pending={pendingSentIds.has(u.id)}
   383→   383→                      onAdd={() => handleAddFriend(u)}
   384→   384→                    />
   385→   385→                  </motion.li>
   386→   386→                ))}
   387→   387→              </AnimatePresence>
   388→   388→            </motion.ul>
   389→   389→          )}
   390→   390→        </section>
   391→   391→      )}
   392→   392→
   393→   393→      {/* ── DEFAULT mode (no active search) ────────────────────────────────── */}
   394→   394→      {!hasSearchQuery && (
   395→   395→        <motion.div
   396→   396→          variants={LIST_CONTAINER}
   397→   397→          initial={prefersReduced ? false : 'hidden'}
   398→   398→          animate="show"
   399→   399→          className="space-y-5"
   400→   400→        >
   401→   401→          {/* Pending incoming requests */}
   402→   402→          <section aria-labelledby="friends-pending-in">
   403→   403→            <header className="mb-2 flex items-center justify-between gap-2">
   404→   404→              <h2
   405→   405→                id="friends-pending-in"
   406→   406→                className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   407→   407→              >
   408→   408→                <span className="h-1.5 w-1.5 rounded-full bg-violet-500" aria-hidden="true" />
   409→   409→                Incoming Requests
   410→   410→                {pendingIn.length > 0 && (
   411→   411→                  <span className="rounded-full bg-violet-100 px-1.5 py-0.5 text-[10px] font-bold text-violet-700 dark:bg-violet-950/60 dark:text-violet-300">
   412→   412→                    {pendingIn.length}
   413→   413→                  </span>
   414→   414→                )}
   415→   415→              </h2>
   416→   416→              <button
   417→   417→                type="button"
   418→   418→                onClick={handleRefresh}
   419→   419→                className="text-xs font-medium text-muted-foreground transition hover:text-foreground"
   420→   420→                aria-label="Refresh friends list"
   421→   421→              >
   422→   422→                Refresh
   423→   423→              </button>
   424→   424→            </header>
   425→   425→
   426→   426→            {pendingIn.length === 0 ? (
   427→   427→              <p className="rounded-lg bg-muted/40 px-3 py-3 text-xs text-muted-foreground">
   428→   428→                No pending requests.
   429→   429→              </p>
   430→   430→            ) : (
   431→   431→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   432→   432→                {pendingIn.map((c) => (
   433→   433→                  <motion.li key={c.id} variants={LIST_ITEM}>
   434→   434→                    <PendingRequestRow
   435→   435→                      conn={c}
   436→   436→                      direction="in"
   437→   437→                      onAccept={() => handleAccept(c)}
   438→   438→                      onReject={() => handleReject(c)}
   439→   439→                    />
   440→   440→                  </motion.li>
   441→   441→                ))}
   442→   442→              </motion.ul>
   443→   443→            )}
   444→   444→          </section>
   445→   445→
   446→   446→          {/* Pending outgoing requests */}
   447→   447→          {pendingOut.length > 0 && (
   448→   448→            <section aria-labelledby="friends-pending-out">
   449→   449→              <h2
   450→   450→                id="friends-pending-out"
   451→   451→                className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   452→   452→              >
   453→   453→                <Clock className="h-3 w-3" aria-hidden="true" />
   454→   454→                Sent Requests
   455→   455→                <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-bold text-muted-foreground">
   456→   456→                  {pendingOut.length}
   457→   457→                </span>
   458→   458→              </h2>
   459→   459→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   460→   460→                {pendingOut.map((c) => (
   461→   461→                  <motion.li key={c.id} variants={LIST_ITEM}>
   462→   462→                    <PendingRequestRow conn={c} direction="out" />
   463→   463→                  </motion.li>
   464→   464→                ))}
   465→   465→              </motion.ul>
   466→   466→            </section>
   467→   467→          )}
   468→   468→
   469→   469→          {/* Current friends list */}
   470→   470→          <section aria-labelledby="friends-list">
   471→   471→            <h2
   472→   472→              id="friends-list"
   473→   473→              className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   474→   474→            >

   525→   525→  async function run(action: 'accept' | 'reject') {
   526→   526→    setBusy(action)
   527→   527→    try {
   528→   528→      if (action === 'accept') await onAccept?.()
   529→   529→      else await onReject?.()
   530→   530→    } finally {
   531→   531→      setBusy(null)
   532→   532→    }
   533→   533→  }
   534→   534→
   535→   535→  return (
   536→   536→    <Card className="overflow-hidden">
   537→   537→      <CardContent className="flex items-center gap-3 p-3">

   541→   541→            {initials(conn.friendName) || '?'}

   548→   548→          </p>
   549→   549→        </div>
   550→   550→        {direction === 'in' ? (
   551→   551→          <div className="flex shrink-0 items-center gap-1.5">
   552→   552→            <Button
   553→   553→              type="button"
   554→   554→              size="sm"
   555→   555→              onClick={() => run('accept')}
   556→   556→              disabled={busy !== null}
   557→   557→              className="h-8 bg-violet-600 px-3 text-xs hover:bg-violet-700"
   558→   558→              aria-label={`Accept ${conn.friendName}'s friend request`}
   559→   559→            >
   560→   560→              {busy === 'accept' ? (
   561→   561→                <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />

   569→   569→              size="sm"
   570→   570→              variant="outline"
   571→   571→              onClick={() => run('reject')}
   572→   572→              disabled={busy !== null}

   574→   574→              aria-label={`Decline ${conn.friendName}'s friend request`}
   575→   575→            >
   576→   576→              {busy === 'reject' ? (
   577→   577→                <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
