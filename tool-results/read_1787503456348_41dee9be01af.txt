     5→// FriendsScreen — Friends sub-tab content for the Social screen (Wave 6 Task 6B).
     6→//
     7→// Per blueprint §18 SOCIAL GRAPH + DESIGN_SYSTEM.md §5.2.7 (Friends list):
     8→//  - Search bar (debounced 250ms) → GET /api/social/search?q=
     9→//  - Pending requests section:
    10→//      • Incoming (status === 'PENDING_RECEIVED') — Accept / Reject buttons
    11→//      • Outgoing (status === 'PENDING_SENT') — "Pending" label
    12→//  - Current friends list (status === 'ACCEPTED'):
    13→//      avatar + name + phone + "Message" (placeholder toast) + "Unfriend" button
    14→//  - Search results: each row shows "Add friend" button (calls sendRequest).
    15→//    Already-connected users are excluded client-side.
    16→//
    17→// State sourcing:
    18→//  - Reads `connections` + mutation methods from `useSocial` (Task 1C owns the
    19→//    store — we CALL its methods, never modify it).
    20→//  - Search results are fetched locally (the social-store has no `search`
    21→//    method by design — search is a transient, non-cached query).
    22→//
    23→// Governance (Task 6B):
    24→//  - Does NOT touch any API route (Task 6A owns /api/social/**).
    25→//  - Does NOT touch social-store.ts (Task 1C owns).
    26→//  - Does NOT touch consumer-view.tsx (Task 3A owns the tab routing).
    27→//  - Resilient to missing endpoints — surfaces a soft "Couldn't load" toast.

    93→function initials(name: string): string {
    94→  return name
    95→    .split(' ')
    96→    .map((p) => p[0])
    97→    .filter(Boolean)
    98→    .slice(0, 2)
    99→    .join('')
   100→    .toUpperCase()
   101→}

   107→export function FriendsScreen({ className }: FriendsScreenProps) {
   108→  const prefersReduced = useReducedMotion()
   109→  const { toast } = useToast()
   110→
   111→  // ── social-store hooks (Task 1C) ─────────────────────────────────────────
   112→  const connections = useSocial((s) => s.connections)
   113→  const isLoading = useSocial((s) => s.isLoading)
   114→  const refresh = useSocial((s) => s.refresh)
   115→  const sendRequest = useSocial((s) => s.sendRequest)
   116→  const acceptRequest = useSocial((s) => s.acceptRequest)
   117→  const declineRequest = useSocial((s) => s.declineRequest)
   118→  const unfollow = useSocial((s) => s.unfollow)
   119→
   120→  // ── Derived slices ─────────────────────────────────────────────────────────
   121→  const pendingIn = React.useMemo(
   122→    // S1 Reconstruction: server returns 'PENDING_RECEIVED' (they sent to me)
   123→    () => connections.filter((c) => c.status === 'PENDING_RECEIVED'),
   124→    [connections],
   125→  )
   126→  const pendingOut = React.useMemo(
   127→    // S1 Reconstruction: server returns 'PENDING_SENT' (I sent to them)
   128→    () => connections.filter((c) => c.status === 'PENDING_SENT'),
   129→    [connections],
   130→  )
   131→  const friends = React.useMemo(
   132→    () =>
   133→      connections
   134→        .filter((c) => c.status === 'ACCEPTED')
   135→        // newest accepted first (acceptedAt ?? createdAt fallback)
   136→        .sort((a, b) => {
   137→          const at = (t?: string | null) => (t ? new Date(t).getTime() : 0)
   138→          return at(b.acceptedAt) - at(a.acceptedAt)
   139→        }),
   140→    [connections],
   141→  )
   142→
   143→  // Set of friend IDs (any status) for client-side exclusion of search results.
   144→  const connectedIds = React.useMemo(
   145→    () => new Set(connections.map((c) => c.userId)),

   149→  // ── Search state (local — not in social-store by design) ───────────────────
   150→  const [query, setQuery] = React.useState('')
   151→  const [debounced, setDebounced] = React.useState('')
   152→  const [results, setResults] = React.useState<SocialSearchUser[]>([])
   153→  const [searching, setSearching] = React.useState(false)
   154→  const [searchError, setSearchError] = React.useState<string | null>(null)
   155→  // Track outgoing requests that succeeded locally so we can flip the row's
   156→  // button to "Pending" without waiting for a refresh.
   157→  const [pendingSentIds, setPendingSentIds] = React.useState<Set<string>>(new Set())
   158→
   159→  // 250ms debounce — only triggers a fetch when query length ≥ 2.
   160→  React.useEffect(() => {
   161→    const q = query.trim()
   162→    if (q === debounced) return
   163→    const t = setTimeout(() => setDebounced(q), 250)
   164→    return () => clearTimeout(t)
   165→  }, [query, debounced])
   166→
   167→  // Fetch search results whenever the debounced query changes.
   168→  React.useEffect(() => {
   169→    const q = debounced.trim()
   170→    if (q.length < 2) {
   171→      setResults([])
   172→      setSearchError(null)
   173→      setSearching(false)
   174→      return
   175→    }
   176→    let cancelled = false
   177→    setSearching(true)
   178→    setSearchError(null)
   179→    fetch(`/api/social/search?q=${encodeURIComponent(q)}`, {
   180→      headers: { 'Content-Type': 'application/json' },
   181→    })
   182→      .then(async (res) => {
   183→        if (!res.ok) {
   184→          const body = await res.json().catch(() => ({}))
   185→          throw new Error(body?.error || `Search failed (${res.status})`)
   186→        }
   187→        const data = (await res.json()) as { users?: SocialSearchUser[] }
   188→        if (!cancelled) {
   189→          setResults(data.users ?? [])
   190→        }
   191→      })
   192→      .catch((err) => {
   193→        if (!cancelled) {
   194→          setSearchError(err instanceof Error ? err.message : 'Search failed')
   195→          setResults([])
   196→        }
   197→      })
   198→      .finally(() => {
   199→        if (!cancelled) setSearching(false)
   200→      })
   201→    return () => {
   202→      cancelled = true
   203→    }
   204→  }, [debounced])

   270→      })

   272→      toast({ title: 'Blocked', description: `${conn.name} has been blocked.` })

   275→      toast({ title: 'Could not block', description: (e as Error).message, variant: 'destructive' })

   551→// Row components
   552→// ═══════════════════════════════════════════════════════════════════════════
   553→
   554→interface PendingRequestRowProps {
   555→  conn: SocialConnection
   556→  direction: 'in' | 'out'
   557→  onAccept?: () => void
   558→  onReject?: () => void
   559→}
   560→
   561→function PendingRequestRow({ conn, direction, onAccept, onReject }: PendingRequestRowProps) {
   562→  const [busy, setBusy] = React.useState<'accept' | 'reject' | null>(null)
   563→
   564→  async function run(action: 'accept' | 'reject') {
   565→    setBusy(action)
   566→    try {
   567→      if (action === 'accept') await onAccept?.()
   568→      else await onReject?.()
   569→    } finally {
