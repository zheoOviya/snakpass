     5→// FriendsScreen — Friends sub-tab content for the Social screen (Wave 6 Task 6B).
     6→//
     7→// Per blueprint §18 SOCIAL GRAPH + DESIGN_SYSTEM.md §5.2.7 (Friends list):
     8→//  - Search bar (debounced 250ms) → GET /api/social/search?q=
     9→//  - Pending requests section:
    10→//      • Incoming (status === 'PENDING_IN') — Accept / Reject buttons
    11→//      • Outgoing (status === 'PENDING_OUT') — "Pending" label
    12→//  - Current friends list (status === 'ACCEPTED'):
    13→//      avatar + name + phone + "Message" (placeholder toast) + "Unfriend" button
    14→//  - Search results: each row shows "Add friend" button (calls sendRequest).
    15→//    Already-connected users are excluded client-side.
    16→//
    17→// State sourcing:
    18→//  - Reads `connections` + mutation methods from `useSocial` (Task 1C owns the
    19→//    store — we CALL its methods, never modify it).
    20→//  - Search results are fetched locally (the social-store has no `search`
    21→//    method by design — search is a transient, non-cached query).
    22→//
    23→// Governance (Task 6B):
    24→//  - Does NOT touch any API route (Task 6A owns /api/social/**).
    25→//  - Does NOT touch social-store.ts (Task 1C owns).
    26→//  - Does NOT touch consumer-view.tsx (Task 3A owns the tab routing).
    27→//  - Resilient to missing endpoints — surfaces a soft "Couldn't load" toast.
    28→

    93→function initials(name: string): string {
    94→  return name
    95→    .split(' ')
    96→    .map((p) => p[0])
    97→    .filter(Boolean)
    98→    .slice(0, 2)
    99→    .join('')
   100→    .toUpperCase()
   101→}

   103→// ─────────────────────────────────────────────────────────────────────────────
   104→// FriendsScreen
   105→// ─────────────────────────────────────────────────────────────────────────────
   106→
   107→export function FriendsScreen({ className }: FriendsScreenProps) {
   108→  const prefersReduced = useReducedMotion()
   109→  const { toast } = useToast()
   110→
   111→  // ── social-store hooks (Task 1C) ─────────────────────────────────────────
   112→  const connections = useSocial((s) => s.connections)
   113→  const isLoading = useSocial((s) => s.isLoading)
   114→  const refresh = useSocial((s) => s.refresh)
   115→  const sendRequest = useSocial((s) => s.sendRequest)
   116→  const acceptRequest = useSocial((s) => s.acceptRequest)
   117→  const declineRequest = useSocial((s) => s.declineRequest)
   118→  const unfollow = useSocial((s) => s.unfollow)
   119→
   120→  // ── Derived slices ─────────────────────────────────────────────────────────
   121→  const pendingIn = React.useMemo(
   122→    () => connections.filter((c) => c.status === 'PENDING_IN'),
   123→    [connections],
   124→  )
   125→  const pendingOut = React.useMemo(
   126→    () => connections.filter((c) => c.status === 'PENDING_OUT'),
   127→    [connections],
   128→  )
   129→  const friends = React.useMemo(
   130→    () =>
   131→      connections
   132→        .filter((c) => c.status === 'ACCEPTED')
   133→        // newest accepted first (acceptedAt ?? createdAt fallback)
   134→        .sort((a, b) => {
   135→          const at = (t?: string) => (t ? new Date(t).getTime() : 0)
   136→          return at(b.acceptedAt) - at(a.acceptedAt)
   137→        }),
   138→    [connections],
   139→  )
   140→
   141→  // Set of friend IDs (any status) for client-side exclusion of search results.
   142→  const connectedIds = React.useMemo(
   143→    () => new Set(connections.map((c) => c.friendId)),
   144→    [connections],
   145→  )
   146→
   147→  // ── Search state (local — not in social-store by design) ───────────────────
   148→  const [query, setQuery] = React.useState('')
   149→  const [debounced, setDebounced] = React.useState('')
   150→  const [results, setResults] = React.useState<SocialSearchUser[]>([])
   151→  const [searching, setSearching] = React.useState(false)
   152→  const [searchError, setSearchError] = React.useState<string | null>(null)
   153→  // Track outgoing requests that succeeded locally so we can flip the row's
   154→  // button to "Pending" without waiting for a refresh.
   155→  const [pendingSentIds, setPendingSentIds] = React.useState<Set<string>>(new Set())
   156→
   157→  // 250ms debounce — only triggers a fetch when query length ≥ 2.
   158→  React.useEffect(() => {
   159→    const q = query.trim()
   160→    if (q === debounced) return
   161→    const t = setTimeout(() => setDebounced(q), 250)
   162→    return () => clearTimeout(t)
   163→  }, [query, debounced])
   164→
   165→  // Fetch search results whenever the debounced query changes.
   166→  React.useEffect(() => {
   167→    const q = debounced.trim()
   168→    if (q.length < 2) {
   169→      setResults([])
   170→      setSearchError(null)
   171→      setSearching(false)
   172→      return
   173→    }
   174→    let cancelled = false
   175→    setSearching(true)
   176→    setSearchError(null)
   177→    fetch(`/api/social/search?q=${encodeURIComponent(q)}`, {
   178→      headers: { 'Content-Type': 'application/json' },
   179→    })
   180→      .then(async (res) => {
   181→        if (!res.ok) {
   182→          const body = await res.json().catch(() => ({}))
   183→          throw new Error(body?.error || `Search failed (${res.status})`)
   184→        }
   185→        const data = (await res.json()) as { users?: SocialSearchUser[] }
   186→        if (!cancelled) {
   187→          setResults(data.users ?? [])
   188→        }
   189→      })
   190→      .catch((err) => {
   191→        if (!cancelled) {
   192→          setSearchError(err instanceof Error ? err.message : 'Search failed')
   193→          setResults([])
   194→        }
   195→      })
   196→      .finally(() => {
   197→        if (!cancelled) setSearching(false)
   198→      })
   199→    return () => {
   200→      cancelled = true
   201→    }
   202→  }, [debounced])
   203→
   204→  // ── Action handlers ─────────────────────────────────────────────────────────
   205→  async function handleAccept(conn: SocialConnection) {
   206→    try {
   207→      await acceptRequest(conn.id)
   208→      toast({
   209→        title: 'Friend request accepted',
   210→        description: `${conn.friendName} is now your friend.`,
   211→      })
   212→    } catch (e) {
   213→      toast({
   214→        title: 'Could not accept request',
   215→        description: (e as Error).message,
   216→        variant: 'destructive',
   217→      })
   218→    }
   219→  }
   220→
   221→  async function handleReject(conn: SocialConnection) {
   222→    try {
   223→      await declineRequest(conn.id)
   224→      toast({
   225→        title: 'Request declined',
   226→        description: `${conn.friendName}'s request was dismissed.`,
   227→      })
   228→    } catch (e) {
   229→      toast({
   230→        title: 'Could not decline request',
   231→        description: (e as Error).message,
   232→        variant: 'destructive',
   233→      })
   234→    }
   235→  }
   236→
   237→  async function handleUnfriend(conn: SocialConnection) {
   238→    // Optimistic confirmation — friend removal is destructive.
   239→    const ok = window.confirm(
   240→      `Unfriend ${conn.friendName}? They won't be notified, but you'll need to send a new request to reconnect.`,
   241→    )
   242→    if (!ok) return
   243→    try {
   244→      await unfollow(conn.friendId)
   245→      toast({
   246→        title: 'Unfriended',
   247→        description: `${conn.friendName} is no longer your friend.`,
   248→      })
   249→    } catch (e) {
   250→      toast({
   251→        title: 'Could not unfriend',
   252→        description: (e as Error).message,
   253→        variant: 'destructive',
   254→      })
   255→    }
   256→  }
   257→
   258→  async function handleSendMessage(conn: SocialConnection) {
   259→    // Placeholder — messaging is on the Wave 8+ roadmap (blueprint §22).
   260→    toast({
   261→      title: 'Messaging coming soon',
   262→      description: `You'll be able to message ${conn.friendName} in a future release.`,
   263→    })
   264→  }
   265→
   266→  async function handleAddFriend(user: SocialSearchUser) {
   267→    try {
   268→      await sendRequest(user.id)
   269→      setPendingSentIds((s) => new Set(s).add(user.id))
   270→      toast({
   271→        title: 'Friend request sent',
   272→        description: `Waiting for ${user.name} to accept.`,
   273→      })
   274→    } catch (e) {
   275→      toast({
   276→        title: 'Could not send request',
   277→        description: (e as Error).message,
   278→        variant: 'destructive',
   279→      })
   280→    }
   281→  }
   282→
   283→  function handleRefresh() {
   284→    refresh().catch(() => {
   285→      /* best-effort — store surfaces its own error */
   286→    })
   287→  }
   288→
   289→  const hasSearchQuery = debounced.trim().length >= 2
   290→  const filteredResults = React.useMemo(
   291→    () => results.filter((u) => !connectedIds.has(u.id) && !pendingSentIds.has(u.id)),
   292→    [results, connectedIds, pendingSentIds],
   293→  )
   294→
   295→  // ── Loading skeleton ─────────────────────────────────────────────────────────
   296→  if (isLoading && connections.length === 0) {
   297→    return (
   298→      <div className="space-y-3" role="status" aria-label="Loading friends">
   299→        <span className="sr-only">Loading your friends…</span>
   300→        {Array.from({ length: 4 }).map((_, i) => (
   301→          <SocialFeedSkeleton key={i} />
   302→        ))}
   303→      </div>
   304→    )
   305→  }
   306→
   307→  return (
   308→    <div className={cn('space-y-5', className)}>
   309→      {/* ── Search bar ─────────────────────────────────────────────────────── */}
   310→      <div className="relative">
   311→        <Search
   312→          className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   313→          aria-hidden="true"
   314→        />
   315→        <Input
   316→          type="search"
   317→          value={query}
   318→          onChange={(e) => setQuery(e.target.value)}
   319→          placeholder="Search by name or phone…"
   320→          aria-label="Search for friends"
   321→          className="h-11 rounded-xl pl-9 pr-9"
   322→        />
   323→        {query && (
   324→          <button
   325→            type="button"
   326→            onClick={() => setQuery('')}
   327→            aria-label="Clear search"
   328→            className="snak-focus-ring absolute right-2.5 top-1/2 -translate-y-1/2 rounded-full p-1 text-muted-foreground transition hover:bg-muted"
   329→          >
   330→            <X className="h-4 w-4" aria-hidden="true" />
   331→          </button>
   332→        )}
   333→      </div>
   334→
   335→      {/* ── SEARCH RESULTS mode ─────────────────────────────────────────────── */}
   336→      {hasSearchQuery && (
   337→        <section aria-label="Search results" className="space-y-2">
   338→          <header className="flex items-center justify-between gap-2">
   339→            <h2 className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   340→              {searching
   341→                ? 'Searching…'
   342→                : `${filteredResults.length} ${filteredResults.length === 1 ? 'result' : 'results'}`}
   343→            </h2>
   344→          </header>
   345→
   346→          {searching ? (
   347→            <div className="space-y-2">
   348→              {Array.from({ length: 3 }).map((_, i) => (
   349→                <SocialFeedSkeleton key={i} />
   350→              ))}
   351→            </div>
   352→          ) : searchError ? (
   353→            <Card>
   354→              <CardContent className="flex items-center gap-2 p-4 text-sm text-muted-foreground">
   355→                <AlertCircle className="h-4 w-4 text-danger-600" aria-hidden="true" />
   356→                <span>{searchError}</span>
   357→              </CardContent>
   358→            </Card>
   359→          ) : filteredResults.length === 0 ? (
   360→            <EmptyState
   361→              variant="no-friends"
   362→              title="No users found"
   363→              description={
   364→                results.length === 0
   365→                  ? `No one matching "${debounced}". Try a different name or phone.`
   366→                  : 'Everyone matching your search is already your friend or has a pending request.'
   367→              }
   368→              className="py-8"
   369→            />
   370→          ) : (
   371→            <motion.ul
   372→              variants={LIST_CONTAINER}
   373→              initial={prefersReduced ? false : 'hidden'}
   374→              animate="show"
   375→              className="space-y-2"
   376→            >
   377→              <AnimatePresence>
   378→                {filteredResults.map((u) => (
   379→                  <motion.li key={u.id} variants={LIST_ITEM} exit={{ opacity: 0 }}>
   380→                    <SearchResultRow
   381→                      user={u}
   382→                      pending={pendingSentIds.has(u.id)}
   383→                      onAdd={() => handleAddFriend(u)}
   384→                    />
   385→                  </motion.li>
   386→                ))}
   387→              </AnimatePresence>
   388→            </motion.ul>
   389→          )}
   390→        </section>
   391→      )}
   392→
   393→      {/* ── DEFAULT mode (no active search) ────────────────────────────────── */}
   394→      {!hasSearchQuery && (
   395→        <motion.div
   396→          variants={LIST_CONTAINER}
   397→          initial={prefersReduced ? false : 'hidden'}
   398→          animate="show"
   399→          className="space-y-5"
   400→        >
   401→          {/* Pending incoming requests */}
   402→          <section aria-labelledby="friends-pending-in">
   403→            <header className="mb-2 flex items-center justify-between gap-2">
   404→              <h2
   405→                id="friends-pending-in"
   406→                className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   407→              >
   408→                <span className="h-1.5 w-1.5 rounded-full bg-violet-500" aria-hidden="true" />
   409→                Incoming Requests
   410→                {pendingIn.length > 0 && (
   411→                  <span className="rounded-full bg-violet-100 px-1.5 py-0.5 text-[10px] font-bold text-violet-700 dark:bg-violet-950/60 dark:text-violet-300">
   412→                    {pendingIn.length}
   413→                  </span>
   414→                )}
   415→              </h2>
   416→              <button
   417→                type="button"
   418→                onClick={handleRefresh}
   419→                className="text-xs font-medium text-muted-foreground transition hover:text-foreground"
   420→                aria-label="Refresh friends list"
   421→              >
   422→                Refresh
   423→              </button>
   424→            </header>
   425→
   426→            {pendingIn.length === 0 ? (
   427→              <p className="rounded-lg bg-muted/40 px-3 py-3 text-xs text-muted-foreground">
   428→                No pending requests.
   429→              </p>
   430→            ) : (
   431→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   432→                {pendingIn.map((c) => (
   433→                  <motion.li key={c.id} variants={LIST_ITEM}>
   434→                    <PendingRequestRow
   435→                      conn={c}
   436→                      direction="in"
   437→                      onAccept={() => handleAccept(c)}
   438→                      onReject={() => handleReject(c)}
   439→                    />
   440→                  </motion.li>
   441→                ))}
   442→              </motion.ul>
   443→            )}
   444→          </section>
   445→
   446→          {/* Pending outgoing requests */}
   447→          {pendingOut.length > 0 && (
   448→            <section aria-labelledby="friends-pending-out">
   449→              <h2
   450→                id="friends-pending-out"
   451→                className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   452→              >
   453→                <Clock className="h-3 w-3" aria-hidden="true" />
   454→                Sent Requests
   455→                <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-bold text-muted-foreground">
   456→                  {pendingOut.length}
   457→                </span>
   458→              </h2>
   459→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   460→                {pendingOut.map((c) => (
   461→                  <motion.li key={c.id} variants={LIST_ITEM}>
   462→                    <PendingRequestRow conn={c} direction="out" />
   463→                  </motion.li>
   464→                ))}
   465→              </motion.ul>
   466→            </section>
   467→          )}
   468→
   469→          {/* Current friends list */}
   470→          <section aria-labelledby="friends-list">
   471→            <h2
   472→              id="friends-list"
   473→              className="mb-2 flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground"
   474→            >
   475→              <span className="h-1.5 w-1.5 rounded-full bg-teal-500" aria-hidden="true" />
   476→              Your Friends
   477→              {friends.length > 0 && (
   478→                <span className="rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-bold text-muted-foreground">
   479→                  {friends.length}
   480→                </span>
   481→              )}
   482→            </h2>
   483→
   484→            {friends.length === 0 ? (
   485→              <EmptyState
   486→                variant="no-friends"
   487→                title="No friends yet"
   488→                description="Search by name or phone above to find people you know on SnakZap."
   489→                className="py-6"
   490→              />
   491→            ) : (
   492→              <motion.ul variants={LIST_CONTAINER} className="space-y-2">
   493→                {friends.map((c) => (
   494→                  <motion.li key={c.id} variants={LIST_ITEM}>
   495→                    <FriendRow
   496→                      conn={c}
   497→                      onMessage={() => handleSendMessage(c)}
   498→                      onUnfriend={() => handleUnfriend(c)}
   499→                    />
   500→                  </motion.li>
   501→                ))}
   502→              </motion.ul>
   503→            )}
   504→          </section>
   505→        </motion.div>
   506→      )}
   507→    </div>
   508→  )
   509→}

   514→
   515→interface PendingRequestRowProps {
   516→  conn: SocialConnection
   517→  direction: 'in' | 'out'
   518→  onAccept?: () => void
   519→  onReject?: () => void
   520→}
   521→
   522→function PendingRequestRow({ conn, direction, onAccept, onReject }: PendingRequestRowProps) {
   523→  const [busy, setBusy] = React.useState<'accept' | 'reject' | null>(null)
   524→
   525→  async function run(action: 'accept' | 'reject') {
   526→    setBusy(action)
   527→    try {
   528→      if (action === 'accept') await onAccept?.()
   529→      else await onReject?.()
   530→    } finally {
   531→      setBusy(null)
   532→    }
   533→  }
   534→
   535→  return (
   536→    <Card className="overflow-hidden">
   537→      <CardContent className="flex items-center gap-3 p-3">
   538→        <Avatar className="h-10 w-10 ring-2 ring-violet-300 ring-offset-2 ring-offset-background">
   539→          {conn.friendAvatarUrl && <AvatarImage src={conn.friendAvatarUrl} alt="" />}
   540→          <AvatarFallback className="bg-gradient-to-br from-violet-400 to-violet-600 text-xs font-bold text-white">
   541→            {initials(conn.friendName) || '?'}
   542→          </AvatarFallback>
   543→        </Avatar>
   544→        <div className="min-w-0 flex-1">
   545→          <p className="truncate text-sm font-semibold text-foreground">{conn.friendName}</p>
   546→          <p className="truncate text-xs text-muted-foreground">
   547→            {conn.friendCampusName ?? 'SnakZap user'}
   548→          </p>
   549→        </div>
   550→        {direction === 'in' ? (
   551→          <div className="flex shrink-0 items-center gap-1.5">
   552→            <Button
   553→              type="button"
   554→              size="sm"
   555→              onClick={() => run('accept')}
   556→              disabled={busy !== null}
   557→              className="h-8 bg-violet-600 px-3 text-xs hover:bg-violet-700"
   558→              aria-label={`Accept ${conn.friendName}'s friend request`}
   559→            >
   560→              {busy === 'accept' ? (
   561→                <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   562→              ) : (
   563→                <Check className="h-3.5 w-3.5" aria-hidden="true" />
   564→              )}
   565→              Accept
   566→            </Button>
   567→            <Button
   568→              type="button"
   569→              size="sm"
   570→              variant="outline"
   571→              onClick={() => run('reject')}
   572→              disabled={busy !== null}
   573→              className="h-8 px-3 text-xs"
   574→              aria-label={`Decline ${conn.friendName}'s friend request`}
