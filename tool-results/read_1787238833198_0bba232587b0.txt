     1→// src/lib/gift-service.ts
     2→//
     3→// Wave 6 Task 6C — Gifting backend (ghost order pattern)
     4→// ----------------------------------------------------------------------------
     5→// Server-side transactional helpers for the Gift lifecycle.
     6→//
     7→// Gift flow (blueprint §19 FOOD GIFTING):
     8→//
     9→//   Select food → select friend → optional note → pay → friend notification →
    10→//   friend redeems
    11→//
    12→// Lifecycle states:
    13→//   CREATED → PAID → AVAILABLE → REDEEMED | EXPIRED | CANCELLED | REFUNDED
    14→//
    15→// Ghost order pattern (plan Decision #3):
    16→//   - Sender creates a Gift (status=CREATED)
    17→//   - Sender creates a "ghost Order" via direct `tx.order.create` (NOT calling
    18→//     /api/orders POST — additive only — preserves order route governance).
    19→//     The ghost order's `note` encodes `GIFT:${giftId}:for:${recipientId}` so
    20→//     the frontend can filter it out of "My Orders" via `note.startsWith('GIFT:')`.
    21→//   - Sender pays the ghost order via inline demo Payment (mirrors
    22→//     /api/payments POST logic — NOT calling the route — additive only).
    23→//   - On success: Gift.status → PAID + AVAILABLE, paymentId + orderId set,
    24→//     expiresAt = now + 30 days, Notification to recipient.
    25→//   - Recipient redeems: creates a NEW zero-amount Order with
    26→//     `note='GIFT_FROM:${senderId}:${giftId}'`, `totalAmount=0`,
    27→//     `userId=recipientId`. Gift.status → REDEEMED + recipientOrderId.
    28→//
    29→// Fraud controls (blueprint §19):
    30→//   - recipient binding (recipientId immutable post-create)
    31→//   - 30-day expiry (lazy enforcement via expireGifts())
    32→//   - redemption audit (AuditLog GIFT_REDEEMED + Outbox event)
    33→//   - no double redemption (Gift.status transition is single-use)
    34→//   - payment/refund separation (sender pays + sender cancels → refund)
    35→//
    36→// Governance boundaries:
    37→//   - Does NOT call /api/orders POST (uses tx.order.create directly)
    38→//   - Does NOT call /api/payments POST (creates Payment record inline —
    39→//     mirrors the route's logic but inside the same transaction)
    40→//   - Does NOT call /api/payments/refund POST (creates Refund record inline)
    41→//   - Does NOT modify Order / Payment / Refund / LedgerEntry models
    42→//   - Does NOT modify prisma/schema.prisma (Gift model already exists from 1A)
    43→//
    44→// All functions take a `tx: Prisma.TransactionClient` parameter — callers wrap
    45→// them in `withTransaction()`. Each function is idempotent for retry safety
    46→// (re-execution on a P2034 retry produces the same effect; the route's
    47→// idempotency cache check at the top of the transaction prevents duplicates).
    48→// ----------------------------------------------------------------------------
    49→

   183→// ---------------------------------------------------------------------------
   184→// Internal helpers
   185→// ---------------------------------------------------------------------------
   186→
   187→/**
   188→ * Generate a unique redemption code (8-char hex). Used as the single-use code
   189→ * the recipient enters to redeem a gift. Stored on Gift.redemptionCode (@unique).
   190→ */
   191→function generateRedemptionCode(): string {
   192→  return randomBytes(4).toString('hex').toUpperCase()
   193→}
   194→
   195→/**
   196→ * Generate a 6-digit pickup OTP for the recipient's zero-amount order.
   197→ * Mirrors the /api/orders POST route's OTP generation pattern.
   198→ */
   199→function generatePickupOtp(): string {
   200→  return String(Math.floor(100000 + Math.random() * 900000))
   201→}
   202→
   203→/**
   204→ * Build the JSON statusHistory array entry (mirrors /api/orders POST).
   205→ */
   206→function buildStatusHistory(status: string, atIso: string): string {
   207→  return JSON.stringify([{ status, at: atIso }])
   208→}
   209→
   210→// ---------------------------------------------------------------------------
   211→// createGift — full ghost-order flow (create gift + ghost order + demo payment)
   212→// ---------------------------------------------------------------------------
   213→
   214→/**
   215→ * Create a Gift + ghost Order + demo Payment atomically (inside the caller's
   216→ * transaction). The flow:
   217→ *
   218→ *   1. Load + validate the menu item (exists, available, not deleted)
   219→ *   2. Validate the recipient exists + sender ≠ recipient
   220→ *   3. Create Gift (status=CREATED, expiresAt=now+30d placeholder so the
   221→ *      NOT NULL constraint is satisfied; updated to AVAILABLE later)
   222→ *   4. Create ghost Order (userId=senderId, status=CONFIRMED, totalAmount=item.price,
   223→ *      note=`GIFT:${giftId}:for:${recipientId}`)
   224→ *   5. Create Payment (status=CAPTURED in demo mode since realPayments is OFF —
   225→ *      skip CAPTURE_PENDING publisher step for gifts to keep MVP simple)
   226→ *   6. Create 2 LedgerEntries (DEBIT GATEWAY_RECEIVABLE + CREDIT CONSUMER_REVENUE)
   227→ *   7. Update ghost Order status=PAID
   228→ *   8. Update Gift status=PAID+AVAILABLE + paymentId + orderId + paidAt + availableAt
   229→ *      + expiresAt (30 days from now)
   230→ *   9. Create AuditLog: GIFT_CREATED + PAYMENT_CAPTURED (gift)
   231→ *   10. Enqueue Outbox events: GIFT_CREATED + PAYMENT_CAPTURED (gift)
   232→ *   11. Create Notification to recipient: GIFT_RECEIVED
   233→ *
   234→ * Idempotency: the caller (route handler) checks the idempotency cache at
   235→ * the top of the transaction; this function is only invoked on a cache miss.
   236→ *
   237→ * Throws AppError on validation failures (NOT_FOUND, VALIDATION_ERROR,
   238→ * CONFLICT, INTERNAL_ERROR).
   239→ */
   240→export async function createGift(
   241→  tx: Prisma.TransactionClient,
   242→  input: CreateGiftInput,
   243→): Promise<CreateGiftResult> {
   244→  const traceId = input.traceId ?? ''
   245→  const now = new Date()
   246→  const nowIso = now.toISOString()
   247→
   248→  // -------------------------------------------------------------------------
   249→  // 1. Load + validate menu item
   250→  // -------------------------------------------------------------------------
   251→  const menuItem = await tx.menuItem.findUnique({
   252→    where: { id: input.menuItemId },
   253→    select: {
   254→      id: true,
   255→      restaurantId: true,
   256→      name: true,
   257→      price: true,
   258→      isAvailable: true,
   259→      deletedAt: true,
   260→      restaurant: { select: { id: true, name: true, isActive: true, isSuspended: true } },
   261→    },
   262→  })
   263→  if (!menuItem || menuItem.deletedAt !== null) {
   264→    throw new AppError('NOT_FOUND', `Menu item ${input.menuItemId} not found`, 404, {
   265→      menuItemId: input.menuItemId,
   266→    })
   267→  }
   268→  if (!menuItem.isAvailable) {
   269→    throw new AppError('VALIDATION_ERROR', `${menuItem.name} is no longer available`, 400, {
   270→      menuItemId: input.menuItemId,
   271→      isAvailable: menuItem.isAvailable,
   272→    })
   273→  }
   274→  if (!menuItem.restaurant.isActive || menuItem.restaurant.isSuspended) {
   275→    throw new AppError(
   276→      'VALIDATION_ERROR',
   277→      `Restaurant '${menuItem.restaurant.name}' is currently unavailable`,
   278→      400,
   279→      { restaurantId: menuItem.restaurantId },
   280→    )
   281→  }
   282→
   283→  // -------------------------------------------------------------------------
   284→  // 2. Validate recipient + sender ≠ recipient
   285→  // -------------------------------------------------------------------------
   286→  if (input.senderId === input.recipientId) {
   287→    throw new AppError('VALIDATION_ERROR', 'Cannot send a gift to yourself', 400, {
   288→      senderId: input.senderId,
   289→      recipientId: input.recipientId,
   290→    })
   291→  }
   292→  const recipient = await tx.user.findUnique({
   293→    where: { id: input.recipientId },
   294→    select: { id: true, name: true, phone: true },
   295→  })
   296→  if (!recipient) {
   297→    throw new AppError('NOT_FOUND', `Recipient ${input.recipientId} not found`, 404, {
   298→      recipientId: input.recipientId,
   299→    })
   300→  }
   301→
   302→  // -------------------------------------------------------------------------
   303→  // 3. Create Gift (status=CREATED)
   304→  // -------------------------------------------------------------------------
   305→  // expiresAt is NOT NULL per schema — set placeholder to now+30d. It will be
   306→  // re-set to now+30d when status → AVAILABLE below (effectively the same value,
   307→  // but explicit so the lifecycle is self-documenting).
   308→  const initialExpiresAt = new Date(now.getTime() + GIFT_EXPIRY_MS)
   309→  const redemptionCode = generateRedemptionCode()
   310→  const gift = await tx.gift.create({
   311→    data: {
   312→      senderId: input.senderId,
   313→      recipientId: input.recipientId,
   314→      menuItemId: input.menuItemId,
   315→      menuItemName: menuItem.name,
   316→      menuItemPrice: menuItem.price,
   317→      message: input.message ?? null,
   318→      status: 'CREATED',
   319→      redemptionCode,
   320→      expiresAt: initialExpiresAt,
   321→    },
   322→  })
   323→
   324→  // -------------------------------------------------------------------------
   325→  // 4. Create ghost Order (sender pays for the gift item)
   326→  // -------------------------------------------------------------------------
   327→  // Mirrors /api/orders POST pattern but direct (NOT calling the route):
   328→  //   - userId = senderId
   329→  //   - restaurantId = menuItem.restaurantId
   330→  //   - status = 'CONFIRMED'
   331→  //   - totalAmount = menuItem.price (paise)
   332→  //   - pickupOtp = '000000' (sender never picks up — ghost order)
   333→  //   - note = `GIFT:${gift.id}:for:${recipientId}` (encodes gift linkage;
   334→  //     frontend filters via note.startsWith('GIFT:'))
   335→  //   - itemsCount = 1
   336→  //   - orderItems: 1 row snapshotting the gifted menu item
   337→  const ghostOrderNote = `GIFT:${gift.id}:for:${input.recipientId}`
   338→  const ghostOrder = await tx.order.create({
   339→    data: {
   340→      userId: input.senderId,
   341→      restaurantId: menuItem.restaurantId,
   342→      status: 'CONFIRMED',
   343→      totalAmount: menuItem.price,
   344→      pickupOtp: '000000',
   345→      isCatering: false,
   346→      headcount: null,
   347→      itemsCount: 1,
   348→      note: ghostOrderNote,
   349→      statusHistory: buildStatusHistory('CONFIRMED', nowIso),
   350→      orderItems: {
   351→        create: [
   352→          {
   353→            menuItemId: menuItem.id,
   354→            name: menuItem.name,
   355→            price: menuItem.price,
   356→            quantity: 1,
   357→            subtotal: menuItem.price,
   358→          },
   359→        ],
   360→      },
   361→    },
   362→    include: { orderItems: true },
   363→  })
   364→
   365→  // -------------------------------------------------------------------------
   366→  // 5. Create Payment (demo mode: CAPTURED directly — skip CAPTURE_PENDING)
   367→  // -------------------------------------------------------------------------
   368→  // When realPayments is OFF (default), we synthesize the payment inline:
   369→  //   - status = 'CAPTURED' (no publisher step needed)
   370→  //   - gatewayPaymentId = `pay_demo_<ts>`
   371→  //   - gatewaySignature = `sig_demo_<ts>`
   372→  //   - capturedAt = now
   373→  //   - LedgerEntries created immediately
   374→  //
   375→  // When realPayments is ON (future), this route should be modified to:
   376→  //   - status = 'CAPTURE_PENDING'
   377→  //   - Enqueue PAYMENT_CAPTURE_REQUESTED for the publisher to call
   378→  //     captureRazorpayPayment() and transition to CAPTURED
   379→  // For now (MVP, realPayments OFF), we go straight to CAPTURED.
   380→  const isDemoMode = !isFeatureEnabled('realPayments')
   381→  const gatewayPaymentId = isDemoMode
   382→    ? `pay_demo_${Date.now()}_${randomUUID().slice(0, 8)}`
   383→    : null
   384→  const gatewaySignature = isDemoMode
   385→    ? `sig_demo_${Date.now()}_${randomUUID().slice(0, 8)}`
   386→    : null
   387→  const payment = await tx.payment.create({
   388→    data: {
   389→      orderId: ghostOrder.id,
   390→      userId: input.senderId,
   391→      gatewayOrderId: `order_demo_gift_${gift.id}`,
   392→      gatewayPaymentId,
   393→      gatewaySignature,
   394→      amount: menuItem.price,
   395→      currency: 'INR',
   396→      status: 'CAPTURED',
   397→      capturedAt: now,
   398→    },
   399→  })
   400→
   401→  // -------------------------------------------------------------------------
   402→  // 6. Create double-entry LedgerEntries (Dr gateway receivable + Cr revenue)
   403→  // -------------------------------------------------------------------------
   404→  // Mirrors /api/payments POST route's ledger logic — keeps the ledger
   405→  // balanced (I-06 invariant: every credit has a matching debit).
   406→  await tx.ledgerEntry.create({
   407→    data: {
   408→      paymentId: payment.id,
   409→      entryType: 'DEBIT',
   410→      accountType: 'GATEWAY_RECEIVABLE',
   411→      amount: menuItem.price,
   412→      traceId,
   413→    },
   414→  })
   415→  await tx.ledgerEntry.create({
   416→    data: {
   417→      paymentId: payment.id,
   418→      entryType: 'CREDIT',
   419→      accountType: 'CONSUMER_REVENUE',
   420→      amount: menuItem.price,
   421→      traceId,
   422→    },
   423→  })
   424→
   425→  // -------------------------------------------------------------------------
   426→  // 7. Update ghost Order status → PAID
   427→  // -------------------------------------------------------------------------
   428→  // Append the new status to statusHistory (preserve CONFIRMED entry).
   429→  const updatedStatusHistory = JSON.stringify([
   430→    { status: 'CONFIRMED', at: nowIso },
   431→    { status: 'PAID', at: nowIso },
   432→  ])
   433→  await tx.order.update({
   434→    where: { id: ghostOrder.id },
   435→    data: {
   436→      status: 'PAID',
   437→      statusHistory: updatedStatusHistory,
   438→    },
   439→  })
   440→
   441→  // -------------------------------------------------------------------------
   442→  // 8. Update Gift status → PAID + AVAILABLE + paymentId + orderId + expiry
   443→  // -------------------------------------------------------------------------
   444→  // Single transition to AVAILABLE (PAID is implicit — the gift is paid AND
   445→  // available for the recipient to redeem). We track both timestamps for
   446→  // audit clarity.
   447→  const expiresAt = new Date(now.getTime() + GIFT_EXPIRY_MS)
   448→  const updatedGift = await tx.gift.update({
   449→    where: { id: gift.id },
   450→    data: {
   451→      status: 'AVAILABLE',
   452→      paymentId: payment.id,
   453→      recipientOrderId: null,
   454→      expiresAt,
   455→      paidAt: now,
   456→      availableAt: now,
   457→    },
   458→  })
   459→
   460→  // -------------------------------------------------------------------------
   461→  // 9. Audit logs
   462→  // -------------------------------------------------------------------------
   463→  await tx.auditLog.create({
   464→    data: {
   465→      actorId: input.senderId,
   466→      actorRole: input.senderRole,
   467→      action: 'GIFT_CREATED',
   468→      metadata: JSON.stringify({
   469→        giftId: gift.id,
   470→        senderId: input.senderId,
   471→        recipientId: input.recipientId,
   472→        menuItemId: menuItem.id,
   473→        menuItemName: menuItem.name,
   474→        menuItemPrice: menuItem.price,
   475→        ghostOrderId: ghostOrder.id,
   476→        paymentId: payment.id,
   477→        message: input.message ?? null,
   478→        expiresAt: expiresAt.toISOString(),
   479→      }),
   480→    },
   481→  })
   482→  await tx.auditLog.create({
   483→    data: {
   484→      actorId: input.senderId,
   485→      actorRole: input.senderRole,
   486→      action: 'PAYMENT_CAPTURED',
   487→      metadata: JSON.stringify({
   488→        giftId: gift.id,
   489→        orderId: ghostOrder.id,
   490→        paymentId: payment.id,
   491→        amount: menuItem.price,
   492→        currency: 'INR',
   493→        source: 'gift-ghost-order',
   494→        demoMode: isDemoMode,
   495→      }),
   496→    },
   497→  })
   498→
   499→  // -------------------------------------------------------------------------
   500→  // 10. Outbox events (committed atomically with the business mutation)
   501→  // -------------------------------------------------------------------------
   502→  await enqueueOutboxEvent(tx, {
   503→    eventType: 'GIFT_CREATED',
   504→    aggregateType: 'Gift',
   505→    aggregateId: gift.id,
   506→    payload: {
   507→      giftId: gift.id,
   508→      senderId: input.senderId,
   509→      recipientId: input.recipientId,
   510→      menuItemId: menuItem.id,
   511→      menuItemName: menuItem.name,
   512→      ghostOrderId: ghostOrder.id,
   513→      paymentId: payment.id,
   514→      status: 'AVAILABLE',
   515→      expiresAt: expiresAt.toISOString(),
   516→    },
   517→  })
   518→  await enqueueOutboxEvent(tx, {
   519→    eventType: 'PAYMENT_CAPTURED',
   520→    aggregateType: 'Payment',
   521→    aggregateId: payment.id,
   522→    payload: {
   523→      paymentId: payment.id,
   524→      orderId: ghostOrder.id,
   525→      giftId: gift.id,
   526→      amount: menuItem.price,
   527→      currency: 'INR',
   528→      demoMode: isDemoMode,
   529→      source: 'gift-ghost-order',
   530→    },
   531→  })
   532→
   533→  // -------------------------------------------------------------------------
   534→  // 11. Notification to recipient (GIFT_RECEIVED)
   535→  // -------------------------------------------------------------------------
   536→  // The notification is created INSIDE the transaction so it's atomic with
   537→  // the gift creation. Real-time delivery is via the outbox publisher (Task
   538→  // 2b — emits socket event 'gift:received'); for MVP the row is created +
   539→  // the UI polls /api/notifications or receives via Socket.io.
   540→  await tx.notification.create({
   541→    data: {
   542→      userId: input.recipientId,
   543→      type: 'GIFT_RECEIVED',
   544→      title: 'You received a gift! 🎁',
   545→      body: `A ${menuItem.name} was gifted to you. Redeem before ${expiresAt.toLocaleDateString()}.`,
   546→      data: JSON.stringify({
   547→        giftId: gift.id,
   548→        senderId: input.senderId,
   549→        menuItemId: menuItem.id,
   550→        menuItemName: menuItem.name,
   551→        restaurantId: menuItem.restaurantId,
   552→        restaurantName: menuItem.restaurant.name,
   553→        redemptionCode,
   554→        expiresAt: expiresAt.toISOString(),
   555→        deepLink: `/gifts/${gift.id}`,
   556→      }),
   557→      readAt: null,
   558→    },
   559→  })
   560→
   561→  return {
   562→    gift: {
   563→      id: updatedGift.id,
   564→      status: updatedGift.status,
   565→      redemptionCode: updatedGift.redemptionCode,
   566→      expiresAt: updatedGift.expiresAt,
   567→      menuItemId: updatedGift.menuItemId,
   568→      menuItemName: updatedGift.menuItemName,
   569→      menuItemPrice: updatedGift.menuItemPrice,
   570→      message: updatedGift.message,
   571→      senderId: updatedGift.senderId,
   572→      recipientId: updatedGift.recipientId,
   573→      paymentId: updatedGift.paymentId,
   574→      recipientOrderId: updatedGift.recipientOrderId,
   575→      paidAt: updatedGift.paidAt,
   576→      availableAt: updatedGift.availableAt,
   577→      createdAt: updatedGift.createdAt,
   578→      updatedAt: updatedGift.updatedAt,
   579→    },
   580→    ghostOrder: {
   581→      id: ghostOrder.id,
   582→      status: 'PAID',
   583→      totalAmount: menuItem.price,
   584→      pickupOtp: ghostOrder.pickupOtp,
   585→      note: ghostOrder.note,
   586→    },
   587→    payment: {
   588→      id: payment.id,
   589→      status: payment.status,
   590→      amount: payment.amount,
   591→      currency: payment.currency,
   592→      gatewayPaymentId: payment.gatewayPaymentId,
   593→    },
   594→  }
   595→}
   596→
   597→// ---------------------------------------------------------------------------
   598→// redeemGift — recipient creates a zero-amount Order + transitions gift
   599→// ---------------------------------------------------------------------------
   600→
   601→/**
   602→ * Recipient redeems an AVAILABLE gift. Creates a NEW zero-amount Order for the
   603→ * recipient (userId=recipientId, totalAmount=0, note=`GIFT_FROM:${senderId}:${giftId}`)
   604→ * + transitions Gift.status → REDEEMED + recipientOrderId + redeemedAt.
   605→ *
   606→ * Authorization: caller must verify `gift.recipientId === recipientId` BEFORE
   607→ * calling this function. (The route does this check.)
   608→ *
   609→ * Idempotent: if the gift is already REDEEMED, returns the existing
   610→ * recipientOrderId without creating a duplicate.
   611→ *
   612→ * Throws AppError on:
   613→ *   - NOT_FOUND: gift doesn't exist
   614→ *   - CONFLICT: gift is not in AVAILABLE status (already redeemed, expired, cancelled)
   615→ *   - INTERNAL_ERROR: order creation failed
   616→ */
   617→export async function redeemGift(
   618→  tx: Prisma.TransactionClient,
   619→  input: RedeemGiftInput,
   620→): Promise<RedeemGiftResult> {
   621→  const traceId = input.traceId ?? ''
   622→  const now = new Date()
   623→  const nowIso = now.toISOString()
   624→
   625→  // -------------------------------------------------------------------------
   626→  // Load the gift (with sender info for the notification)
   627→  // -------------------------------------------------------------------------
   628→  const gift = await tx.gift.findUnique({
   629→    where: { id: input.giftId },
   630→    select: {
   631→      id: true,
   632→      senderId: true,
   633→      recipientId: true,
   634→      status: true,
   635→      menuItemId: true,
   636→      menuItemName: true,
   637→      menuItemPrice: true,
   638→      redemptionCode: true,
   639→      expiresAt: true,
   640→      recipientOrderId: true,
   641→      redeemedAt: true,
   642→    },
   643→  })
   644→  if (!gift) {
   645→    throw new AppError('NOT_FOUND', `Gift ${input.giftId} not found`, 404, {
   646→      giftId: input.giftId,
   647→    })
   648→  }
   649→
   650→  // -------------------------------------------------------------------------
   651→  // Idempotent: if already redeemed, return the existing order without
   652→  // creating a duplicate.
   653→  // -------------------------------------------------------------------------
   654→  if (gift.status === 'REDEEMED') {
   655→    if (!gift.recipientOrderId) {
   656→      // Inconsistent state — should never happen (REDEEMED implies order created).
   657→      throw new AppError(
   658→        'INTERNAL_ERROR',
   659→        `Gift ${gift.id} is REDEEMED but has no recipientOrderId`,
   660→        500,
   661→        { giftId: gift.id },
   662→      )
   663→    }
   664→    const existingOrder = await tx.order.findUnique({
   665→      where: { id: gift.recipientOrderId },
   666→      select: { id: true, status: true, totalAmount: true, pickupOtp: true, note: true },
   667→    })
   668→    if (!existingOrder) {
   669→      throw new AppError(
   670→        'INTERNAL_ERROR',
   671→        `Gift ${gift.id} references missing recipientOrder ${gift.recipientOrderId}`,
   672→        500,
   673→        { giftId: gift.id, recipientOrderId: gift.recipientOrderId },
   674→      )
   675→    }
   676→    return {
   677→      gift: {
   678→        id: gift.id,
   679→        status: gift.status,
   680→        recipientOrderId: gift.recipientOrderId,
   681→        redeemedAt: gift.redeemedAt,
   682→      },
   683→      order: {
   684→        id: existingOrder.id,
   685→        status: existingOrder.status,
   686→        totalAmount: existingOrder.totalAmount,
   687→        pickupOtp: existingOrder.pickupOtp,
   688→        note: existingOrder.note,
   689→      },
   690→    }
   691→  }
   692→
   693→  // -------------------------------------------------------------------------
   694→  // Validate status — must be AVAILABLE to redeem
   695→  // -------------------------------------------------------------------------
   696→  if (gift.status !== 'AVAILABLE') {
   697→    throw new AppError(
   698→      'CONFLICT',
   699→      `Gift ${gift.id} is in status ${gift.status} — only AVAILABLE gifts can be redeemed`,
   700→      409,
   701→      { giftId: gift.id, currentStatus: gift.status, requiredStatus: 'AVAILABLE' },
   702→    )
   703→  }
   704→
   705→  // -------------------------------------------------------------------------
   706→  // Validate expiry — lazy enforcement (defensive)
   707→  // -------------------------------------------------------------------------
   708→  if (gift.expiresAt.getTime() < now.getTime()) {
   709→    // Transition to EXPIRED + trigger refund (best-effort; the expireGifts
   710→    // job handles the refund properly).
   711→    await tx.gift.update({
   712→      where: { id: gift.id },
   713→      data: { status: 'EXPIRED' },
   714→    })
   715→    throw new AppError(
   716→      'CONFLICT',
   717→      `Gift ${gift.id} has expired (expiresAt ${gift.expiresAt.toISOString()})`,
   718→      409,
   719→      { giftId: gift.id, expiresAt: gift.expiresAt.toISOString() },
   720→    )
   721→  }
   722→
   723→  // -------------------------------------------------------------------------
   724→  // Load the menu item (for restaurantId + name — needed to create the order)
   725→  // -------------------------------------------------------------------------
   726→  const menuItem = await tx.menuItem.findUnique({
   727→    where: { id: gift.menuItemId },
   728→    select: {
   729→      id: true,
   730→      restaurantId: true,
   731→      name: true,
   732→      deletedAt: true,
   733→      restaurant: { select: { id: true, name: true } },
   734→    },
   735→  })
   736→  if (!menuItem) {
   737→    throw new AppError(
   738→      'INTERNAL_ERROR',
   739→      `Gift ${gift.id} references missing menuItem ${gift.menuItemId}`,
   740→      500,
   741→      { giftId: gift.id, menuItemId: gift.menuItemId },
   742→    )
   743→  }
   744→
   745→  // -------------------------------------------------------------------------
   746→  // Create the recipient's zero-amount Order
   747→  // -------------------------------------------------------------------------
   748→  // This is the recipient's "free pickup" order. The restaurant sees a
   749→  // normal order come in but with totalAmount=0 (the item was already paid
   750→  // for by the sender). The recipient picks up using the OTP below.
   751→  const pickupOtp = generatePickupOtp()
   752→  const orderNote = `GIFT_FROM:${gift.senderId}:${gift.id}`
   753→  const order = await tx.order.create({
   754→    data: {
   755→      userId: input.recipientId,
   756→      restaurantId: menuItem.restaurantId,
   757→      status: 'CONFIRMED',
   758→      totalAmount: 0,
   759→      pickupOtp,
   760→      isCatering: false,
   761→      headcount: null,
   762→      itemsCount: 1,
   763→      note: orderNote,
   764→      statusHistory: buildStatusHistory('CONFIRMED', nowIso),
   765→      orderItems: {
   766→        create: [
   767→          {
   768→            menuItemId: menuItem.id,
   769→            name: menuItem.name,
   770→            price: 0,
   771→            quantity: 1,
   772→            subtotal: 0,
   773→          },
   774→        ],
   775→      },
   776→    },
   777→    include: { orderItems: true },
   778→  })
   779→
   780→  // -------------------------------------------------------------------------
   781→  // Transition Gift.status → REDEEMED + recipientOrderId + redeemedAt
   782→  // -------------------------------------------------------------------------
   783→  // Uses optimistic-lock conditional update (WHERE status='AVAILABLE'). If a
   784→  // concurrent transaction already redeemed/cancelled/expired the gift, the
   785→  // update affects 0 rows → the new order is orphaned BUT the transaction
   786→  // rolls back (withTransaction atomicity — no orphan).
   787→  const transitionResult = await tx.gift.updateMany({
   788→    where: { id: gift.id, status: 'AVAILABLE' },
   789→    data: {
   790→      status: 'REDEEMED',
   791→      recipientOrderId: order.id,
   792→      redeemedAt: now,
   793→    },
   794→  })
   795→  if (transitionResult.count === 0) {
   796→    // Concurrent transition — throw to abort the transaction.
   797→    throw new AppError(
   798→      'CONFLICT',
   799→      `Gift ${gift.id} was modified by a concurrent transaction — please retry`,
   800→      409,
   801→      { giftId: gift.id },
   802→    )
   803→  }
   804→
   805→  // -------------------------------------------------------------------------
   806→  // Audit log
   807→  // -------------------------------------------------------------------------
   808→  await tx.auditLog.create({
   809→    data: {
   810→      actorId: input.recipientId,
   811→      actorRole: input.recipientRole,
   812→      action: 'GIFT_REDEEMED',
   813→      metadata: JSON.stringify({
   814→        giftId: gift.id,
   815→        senderId: gift.senderId,
   816→        recipientId: input.recipientId,
   817→        menuItemId: menuItem.id,
   818→        menuItemName: menuItem.name,
   819→        recipientOrderId: order.id,
   820→        redemptionCode: gift.redemptionCode,
   821→      }),
   822→    },
   823→  })
   824→  await tx.auditLog.create({
   825→    data: {
   826→      actorId: input.recipientId,
   827→      actorRole: input.recipientRole,
   828→      action: 'ORDER_CREATED',
   829→      metadata: JSON.stringify({
   830→        orderId: order.id,
   831→        giftId: gift.id,
   832→        total: 0,
   833→        restaurantId: menuItem.restaurantId,
   834→        source: 'gift-redemption',
   835→      }),
   836→    },
   837→  })
   838→
   839→  // -------------------------------------------------------------------------
   840→  // Outbox events
   841→  // -------------------------------------------------------------------------
   842→  await enqueueOutboxEvent(tx, {
   843→    eventType: 'GIFT_REDEEMED',
   844→    aggregateType: 'Gift',
   845→    aggregateId: gift.id,
   846→    payload: {
   847→      giftId: gift.id,
   848→      senderId: gift.senderId,
   849→      recipientId: input.recipientId,
   850→      recipientOrderId: order.id,
   851→      menuItemId: menuItem.id,
   852→      menuItemName: menuItem.name,
   853→    },
   854→  })
   855→  await enqueueOutboxEvent(tx, {
   856→    eventType: 'ORDER_CREATED',
   857→    aggregateType: 'Order',
   858→    aggregateId: order.id,
   859→    payload: {
   860→      orderId: order.id,
   861→      giftId: gift.id,
   862→      restaurantId: menuItem.restaurantId,
   863→      status: order.status,
   864→      totalAmount: order.totalAmount,
   865→      updatedAt: nowIso,
   866→      pickupOtp: order.pickupOtp,
   867→      source: 'gift-redemption',
   868→    },
   869→  })
   870→
   871→  // -------------------------------------------------------------------------
   872→  // Notification to sender (GIFT_REDEEMED)
   873→  // -------------------------------------------------------------------------
   874→  await tx.notification.create({
   875→    data: {
   876→      userId: gift.senderId,
   877→      type: 'GIFT_REDEEMED',
   878→      title: 'Your gift was redeemed! 🎁',
   879→      body: `${gift.menuItemName} was redeemed by the recipient.`,
   880→      data: JSON.stringify({
   881→        giftId: gift.id,
   882→        recipientId: input.recipientId,
   883→        recipientOrderId: order.id,
   884→        menuItemName: gift.menuItemName,
   885→        deepLink: `/orders/${order.id}`,
   886→      }),
   887→      readAt: null,
   888→    },
   889→  })
   890→
   891→  // -------------------------------------------------------------------------
   892→  // Social activity (optional — only if the recipient's privacy allows)
   893→  // -------------------------------------------------------------------------
   894→  // We record a REDEEMED activity for the recipient's friends feed (blueprint
   895→  // §18 verbs). The metadata NEVER includes payment amount (fraud control).
   896→  // We guard with try/catch so a failure here doesn't roll back the txn
   897→  // (best-effort — the gift is already REDEEMED).
   898→  try {
   899→    await tx.socialActivity.create({
   900→      data: {
   901→        actorId: input.recipientId,
   902→        verb: 'REDEEMED',
   903→        objectType: 'Gift',
   904→        objectId: gift.id,
   905→        metadata: JSON.stringify({
   906→          menuItemName: gift.menuItemName,
   907→          restaurantName: menuItem.restaurant.name,
   908→          // NOTE: never include price/payment amount per blueprint §18.
   909→        }),
   910→        visibility: 'FRIENDS',
   911→      },
   912→    })
   913→  } catch {
   914→    // best-effort — don't fail the redemption over social activity write failure
   915→  }
   916→
   917→  return {
   918→    gift: {
   919→      id: gift.id,
   920→      status: 'REDEEMED',
   921→      recipientOrderId: order.id,
   922→      redeemedAt: now,
   923→    },
   924→    order: {
   925→      id: order.id,
   926→      status: order.status,
   927→      totalAmount: order.totalAmount,
   928→      pickupOtp: order.pickupOtp,
   929→      note: order.note,
   930→    },
   931→  }
   932→}
   933→
   934→// ---------------------------------------------------------------------------
   935→// cancelGift — sender cancels + (if paid) triggers refund inline
   936→// ---------------------------------------------------------------------------
   937→
   938→/**
   939→ * Sender cancels a gift. If the gift was PAID (ghost order has a Payment),
   940→ * triggers a refund inline (mirrors /api/payments/refund POST logic but
   941→ * direct — additive only). Updates Gift.status → CANCELLED + cancelledAt
   942→ * + refundedAt (if refund processed).
   943→ *
   944→ * Authorization: caller must verify `gift.senderId === senderId` BEFORE
   945→ * calling this function. (The route does this check.)
   946→ *
   947→ * Allowed source statuses: CREATED, PAID, AVAILABLE. Not allowed: REDEEMED,
   948→ * EXPIRED, CANCELLED, REFUNDED → 409.
   949→ *
   950→ * Idempotent: if already CANCELLED, returns the existing gift state.
   951→ */
   952→export async function cancelGift(
   953→  tx: Prisma.TransactionClient,
   954→  input: CancelGiftInput,
   955→): Promise<CancelGiftResult> {
   956→  const traceId = input.traceId ?? ''
   957→  const now = new Date()
   958→  const nowIso = now.toISOString()
   959→
   960→  // -------------------------------------------------------------------------
   961→  // Load the gift + payment + ghost order
   962→  // -------------------------------------------------------------------------
   963→  const gift = await tx.gift.findUnique({
   964→    where: { id: input.giftId },
   965→    select: {
   966→      id: true,
   967→      senderId: true,
   968→      recipientId: true,
   969→      status: true,
   970→      menuItemId: true,
   971→      menuItemName: true,
   972→      menuItemPrice: true,
   973→      paymentId: true,
   974→      recipientOrderId: true,
   975→      expiresAt: true,
   976→      cancelledAt: true,
   977→      refundedAt: true,
   978→    },
   979→  })
   980→  if (!gift) {
   981→    throw new AppError('NOT_FOUND', `Gift ${input.giftId} not found`, 404, {
   982→      giftId: input.giftId,
   983→    })
   984→  }
   985→
   986→  // -------------------------------------------------------------------------
   987→  // Idempotent: if already cancelled, return the existing state.
   988→  // -------------------------------------------------------------------------
   989→  if (gift.status === 'CANCELLED' || gift.status === 'REFUNDED') {
   990→    return {
   991→      gift: {
   992→        id: gift.id,
   993→        status: gift.status,
   994→        cancelledAt: gift.cancelledAt,
   995→        refundedAt: gift.refundedAt,
   996→      },
   997→      refund: {
   998→        id: null,
   999→        status: gift.status === 'REFUNDED' ? 'REFUNDED' : null,
  1000→        amount: null,
  1001→      },
  1002→    }
  1003→  }
  1004→
  1005→  // -------------------------------------------------------------------------
  1006→  // Validate source status
  1007→  // -------------------------------------------------------------------------
  1008→  const allowedSourceStatuses = ['CREATED', 'PAID', 'AVAILABLE']
  1009→  if (!allowedSourceStatuses.includes(gift.status)) {
  1010→    throw new AppError(
  1011→      'CONFLICT',
  1012→      `Gift ${gift.id} is in status ${gift.status} — can only cancel gifts in CREATED, PAID, or AVAILABLE`,
  1013→      409,
  1014→      { giftId: gift.id, currentStatus: gift.status, allowedStatuses: allowedSourceStatuses },
  1015→    )
  1016→  }
  1017→
  1018→  // -------------------------------------------------------------------------
  1019→  // Load the recipient (for the notification body)
  1020→  // -------------------------------------------------------------------------
  1021→  const recipient = await tx.user.findUnique({
  1022→    where: { id: gift.recipientId },
  1023→    select: { id: true, name: true },
  1024→  })
  1025→  const recipientName = recipient?.name ?? 'the recipient'
  1026→
  1027→  // -------------------------------------------------------------------------
  1028→  // If the gift was PAID (has a paymentId), trigger the refund inline.
  1029→  // -------------------------------------------------------------------------
  1030→  // Mirrors /api/payments/refund POST logic — direct (NOT calling the route):
  1031→  //   - Load the Payment (must be CAPTURED)
  1032→  //   - Create Refund record (status=REFUNDED in demo mode since realPayments OFF)
  1033→  //   - Create reversal LedgerEntries (DEBIT CONSUMER_REVENUE + CREDIT GATEWAY_RECEIVABLE)
  1034→  //   - Update Payment.status → REFUNDED (full refund)
  1035→  //   - Update ghost Order.status → CANCELLED
  1036→  //   - Audit log: PAYMENT_REFUNDED
  1037→  //   - Outbox: PAYMENT_REFUNDED
  1038→  //   - Update Gift.refundedAt
  1039→  let refundId: string | null = null
  1040→  let refundStatus: string | null = null
  1041→  let refundAmount: number | null = null
  1042→  let paymentWasRefunded = false
  1043→
  1044→  if (gift.paymentId) {
  1045→    const payment = await tx.payment.findUnique({
  1046→      where: { id: gift.paymentId },
  1047→      select: {
  1048→        id: true,
  1049→        orderId: true,
  1050→        userId: true,
  1051→        amount: true,
  1052→        currency: true,
  1053→        status: true,
  1054→        frozen: true,
  1055→        version: true,
  1056→      },
  1057→    })
  1058→    if (!payment) {
  1059→      throw new AppError(
  1060→        'INTERNAL_ERROR',
  1061→        `Gift ${gift.id} references missing payment ${gift.paymentId}`,
  1062→        500,
  1063→        { giftId: gift.id, paymentId: gift.paymentId },
  1064→      )
  1065→    }
  1066→    if (payment.frozen) {
  1067→      throw new AppError(
  1068→        'CONFLICT',
  1069→        `Payment ${payment.id} is frozen — refund blocked (unfreeze first)`,
  1070→        409,
  1071→        { giftId: gift.id, paymentId: payment.id, frozen: true },
  1072→      )
  1073→    }
  1074→    if (payment.status !== 'CAPTURED') {
  1075→      // Payment is in CAPTURE_PENDING / FAILED / REFUNDED — can't refund.
  1076→      // For CREATED-status gifts (not yet paid), skip refund entirely.
  1077→      if (gift.status === 'CREATED') {
  1078→        // No refund needed — gift wasn't paid yet.
  1079→      } else {
  1080→        throw new AppError(
  1081→          'CONFLICT',
  1082→          `Payment ${payment.id} status is ${payment.status} — only CAPTURED payments can be refunded`,
  1083→          409,
  1084→          { giftId: gift.id, paymentId: payment.id, paymentStatus: payment.status },
  1085→        )
  1086→      }
  1087→    } else {
  1088→      // -----------------------------------------------------------------
  1089→      // Create the Refund record (REFUNDED in demo mode — inline)
  1090→      // -----------------------------------------------------------------
  1091→      const isDemoMode = !isFeatureEnabled('realPayments')
  1092→      const refund = await tx.refund.create({
  1093→        data: {
  1094→          paymentId: payment.id,
  1095→          amount: payment.amount,
  1096→          currency: payment.currency,
  1097→          // Demo mode: refund succeeds immediately (REFUNDED). Real mode:
  1098→          // REFUND_PENDING — the publisher calls refundRazorpayPayment() +
  1099→          // transitions to REFUNDED on gateway confirmation.
  1100→          status: isDemoMode ? 'REFUNDED' : 'REFUND_PENDING',
  1101→          // In demo mode, synthesize a Razorpay-style refund ID for audit clarity.
  1102→          // In real mode, the publisher sets this after the gateway call.
  1103→          gatewayRefundId: isDemoMode
  1104→            ? `rpf_demo_${Date.now()}_${randomUUID().slice(0, 8)}`
  1105→            : null,
  1106→          // In demo mode, the refund is already "confirmed" by the gateway.
  1107→          refundedAt: isDemoMode ? now : null,
  1108→        },
  1109→      })
  1110→      refundId = refund.id
  1111→      refundStatus = refund.status
  1112→      refundAmount = refund.amount
  1113→
  1114→      // -----------------------------------------------------------------
  1115→      // Reversal LedgerEntries (Dr CONSUMER_REVENUE + Cr GATEWAY_RECEIVABLE)
  1116→      // -----------------------------------------------------------------
  1117→      await tx.ledgerEntry.create({
  1118→        data: {
  1119→          paymentId: payment.id,
  1120→          entryType: 'DEBIT',
  1121→          accountType: 'CONSUMER_REVENUE',
  1122→          amount: payment.amount,
  1123→          traceId,
  1124→        },
  1125→      })
  1126→      await tx.ledgerEntry.create({
  1127→        data: {
  1128→          paymentId: payment.id,
  1129→          entryType: 'CREDIT',
  1130→          accountType: 'GATEWAY_RECEIVABLE',
  1131→          amount: payment.amount,
  1132→          traceId,
  1133→        },
  1134→      })
  1135→
  1136→      // -----------------------------------------------------------------
  1137→      // Update Payment.status → REFUNDED (full refund)
  1138→      // -----------------------------------------------------------------
  1139→      // Optimistic-lock: WHERE status='CAPTURED' + version=X. If a concurrent
  1140→      // transaction already refunded it, the update affects 0 rows → throw.
  1141→      const paymentUpdate = await tx.payment.updateMany({
  1142→        where: { id: payment.id, status: 'CAPTURED', version: payment.version },
  1143→        data: {
  1144→          status: 'REFUNDED',
  1145→          version: { increment: 1 },
  1146→        },
  1147→      })
  1148→      if (paymentUpdate.count === 0) {
  1149→        throw new AppError(
  1150→          'CONFLICT',
  1151→          `Payment ${payment.id} was modified by a concurrent transaction — please retry`,
  1152→          409,
  1153→          { giftId: gift.id, paymentId: payment.id },
  1154→        )
  1155→      }
  1156→
  1157→      // -----------------------------------------------------------------
  1158→      // Update ghost Order.status → CANCELLED
  1159→      // -----------------------------------------------------------------
  1160→      const orderHistory = JSON.stringify([
  1161→        { status: 'CONFIRMED', at: nowIso },
  1162→        { status: 'PAID', at: nowIso },
  1163→        { status: 'CANCELLED', at: nowIso },
  1164→      ])
  1165→      await tx.order.update({
  1166→        where: { id: payment.orderId },
  1167→        data: {
  1168→          status: 'CANCELLED',
  1169→          statusHistory: orderHistory,
  1170→        },
  1171→      })
  1172→
  1173→      // -----------------------------------------------------------------
  1174→      // Audit + Outbox
  1175→      // -----------------------------------------------------------------
  1176→      await tx.auditLog.create({
  1177→        data: {
  1178→          actorId: input.senderId,
  1179→          actorRole: input.senderRole,
  1180→          action: 'PAYMENT_REFUNDED',
  1181→          metadata: JSON.stringify({
  1182→            paymentId: payment.id,
  1183→            orderId: payment.orderId,
  1184→            giftId: gift.id,
  1185→            refundId: refund.id,
  1186→            amount: payment.amount,
  1187→            fullRefund: true,
  1188→            source: 'gift-cancel',
  1189→            demoMode: isDemoMode,
  1190→          }),
  1191→        },
  1192→      })
  1193→      await enqueueOutboxEvent(tx, {
  1194→        eventType: 'PAYMENT_REFUNDED',
  1195→        aggregateType: 'Refund',
  1196→        aggregateId: refund.id,
  1197→        payload: {
  1198→          refundId: refund.id,
  1199→          paymentId: payment.id,
  1200→          orderId: payment.orderId,
  1201→          giftId: gift.id,
  1202→          amount: payment.amount,
  1203→          currency: payment.currency,
  1204→          fullRefund: true,
  1205→          source: 'gift-cancel',
  1206→          demoMode: isDemoMode,
  1207→        },
  1208→      })
  1209→
  1210→      paymentWasRefunded = true
  1211→    }
  1212→  }
  1213→
  1214→  // -------------------------------------------------------------------------
  1215→  // Transition Gift.status → CANCELLED (+ refundedAt if refund was processed)
  1216→  // -------------------------------------------------------------------------
  1217→  // Conditional update — only proceeds if status is still in the allowed set.
  1218→  // (Concurrent redeem/cancel would have transitioned it already.)
  1219→  const allowedStatusesForTransition = paymentWasRefunded
  1220→    ? ['AVAILABLE', 'PAID']
  1221→    : allowedSourceStatuses
  1222→  const giftTransition = await tx.gift.updateMany({
  1223→    where: {
  1224→      id: gift.id,
  1225→      status: { in: allowedStatusesForTransition },
  1226→    },
  1227→    data: {
  1228→      status: 'CANCELLED',
  1229→      cancelledAt: now,
  1230→      refundedAt: paymentWasRefunded ? now : null,
  1231→    },
  1232→  })
  1233→  if (giftTransition.count === 0) {
  1234→    throw new AppError(
  1235→      'CONFLICT',
  1236→      `Gift ${gift.id} was modified by a concurrent transaction — please retry`,
  1237→      409,
  1238→      { giftId: gift.id },
  1239→    )
  1240→  }
  1241→
  1242→  // -------------------------------------------------------------------------
  1243→  // Audit + Outbox (GIFT_CANCELLED)
  1244→  // -------------------------------------------------------------------------
  1245→  await tx.auditLog.create({
  1246→    data: {
  1247→      actorId: input.senderId,
  1248→      actorRole: input.senderRole,
  1249→      action: 'GIFT_CANCELLED',
  1250→      metadata: JSON.stringify({
  1251→        giftId: gift.id,
  1252→        senderId: input.senderId,
  1253→        recipientId: gift.recipientId,
  1254→        previousStatus: gift.status,
  1255→        paymentId: gift.paymentId,
  1256→        refundId,
  1257→        refundAmount,
  1258→        paymentWasRefunded,
  1259→      }),
  1260→    },
  1261→  })
  1262→  await enqueueOutboxEvent(tx, {
  1263→    eventType: 'GIFT_CANCELLED',
  1264→    aggregateType: 'Gift',
  1265→    aggregateId: gift.id,
  1266→    payload: {
  1267→      giftId: gift.id,
  1268→      senderId: input.senderId,
  1269→      recipientId: gift.recipientId,
  1270→      previousStatus: gift.status,
  1271→      refundId,
  1272→      paymentWasRefunded,
  1273→    },
  1274→  })
  1275→
  1276→  // -------------------------------------------------------------------------
  1277→  // Notification to recipient
  1278→  // -------------------------------------------------------------------------
  1279→  await tx.notification.create({
  1280→    data: {
  1281→      userId: gift.recipientId,
  1282→      type: 'SYSTEM',
  1283→      title: 'A gift was cancelled',
  1284→      body: `A gift from the sender was cancelled${paymentWasRefunded ? ' and refunded' : ''}.`,
  1285→      data: JSON.stringify({
  1286→        giftId: gift.id,
  1287→        senderId: input.senderId,
  1288→        menuItemName: gift.menuItemName,
  1289→        refundProcessed: paymentWasRefunded,
  1290→        deepLink: `/gifts/${gift.id}`,
  1291→      }),
  1292→      readAt: null,
  1293→    },
  1294→  })
  1295→
  1296→  return {
  1297→    gift: {
  1298→      id: gift.id,
  1299→      status: 'CANCELLED',
  1300→      cancelledAt: now,
  1301→      refundedAt: paymentWasRefunded ? now : null,
  1302→    },
  1303→    refund: {
  1304→      id: refundId,
  1305→      status: refundStatus,
  1306→      amount: refundAmount,
  1307→    },
  1308→  }
  1309→}
  1310→
  1311→// ---------------------------------------------------------------------------
  1312→// expireGifts — background job (placeholder — lazy enforcement + cron)
  1313→// ---------------------------------------------------------------------------
  1314→
  1315→/**
  1316→ * Background job: transitions AVAILABLE gifts whose `expiresAt < now` to
  1317→ * EXPIRED + triggers refund for any paid gift.
  1318→ *
  1319→ * Idempotent: only transitions AVAILABLE → EXPIRED (already-EXPIRED gifts
  1320→ * are skipped). Refund is only created once per gift (idempotent via the
  1321→ * `paymentWasRefunded` flag check — if the gift was already refunded via
  1322→ * cancelGift, this job does nothing).
  1323→ *
  1324→ * Usage:
  1325→ *   - Called from a cron job (Wave 8+).
  1326→ *   - Called lazily from GET /api/gifts/[id] when an expired gift is read.
  1327→ *   - Safe to call repeatedly (idempotent).
  1328→ *
  1329→ * NOTE: This is a PLACEHOLDER for the cron job — the route handlers (GET
  1330→ * /api/gifts, GET /api/gifts/[id]) also do lazy expiry checks inline. The
  1331→ * cron job is the primary mechanism; lazy checks are a defensive backstop.
  1332→ */
  1333→export async function expireGifts(
  1334→  tx: Prisma.TransactionClient,
  1335→  options?: { now?: Date; traceId?: string },
  1336→): Promise<ExpireGiftsResult> {
  1337→  const now = options?.now ?? new Date()
  1338→  const traceId = options?.traceId ?? ''
  1339→  const expiredGiftIds: string[] = []
  1340→  let refundCount = 0
  1341→
  1342→  // Find all AVAILABLE gifts whose expiresAt < now.
  1343→  const expiredGifts = await tx.gift.findMany({
  1344→    where: {
  1345→      status: 'AVAILABLE',
  1346→      expiresAt: { lt: now },
  1347→    },
  1348→    select: {
  1349→      id: true,
  1350→      senderId: true,
  1351→      recipientId: true,
  1352→      paymentId: true,
  1353→      menuItemName: true,
  1354→    },
  1355→  })
  1356→
  1357→  for (const gift of expiredGifts) {
  1358→    // Transition AVAILABLE → EXPIRED (conditional update — idempotent).
  1359→    const transition = await tx.gift.updateMany({
  1360→      where: { id: gift.id, status: 'AVAILABLE' },
  1361→      data: { status: 'EXPIRED' },
  1362→    })
  1363→    if (transition.count === 0) {
  1364→      // Concurrent transaction already transitioned — skip.
  1365→      continue
  1366→    }
  1367→    expiredGiftIds.push(gift.id)
  1368→
  1369→    // Audit log
  1370→    await tx.auditLog.create({
  1371→      data: {
  1372→        actorId: null,
  1373→        actorRole: 'SYSTEM',
  1374→        action: 'GIFT_EXPIRED',
  1375→        metadata: JSON.stringify({
  1376→          giftId: gift.id,
  1377→          senderId: gift.senderId,
  1378→          recipientId: gift.recipientId,
  1379→          paymentId: gift.paymentId,
  1380→          source: 'expireGifts-job',
  1381→        }),
  1382→      },
  1383→    })
  1384→
  1385→    // Outbox event
  1386→    await enqueueOutboxEvent(tx, {
  1387→      eventType: 'GIFT_EXPIRED',
  1388→      aggregateType: 'Gift',
  1389→      aggregateId: gift.id,
  1390→      payload: {
  1391→        giftId: gift.id,
  1392→        senderId: gift.senderId,
  1393→        recipientId: gift.recipientId,
  1394→        paymentId: gift.paymentId,
  1395→      },
  1396→    })
  1397→
  1398→    // Trigger refund for paid gifts (if a Payment exists + is CAPTURED).
  1399→    // Mirrors cancelGift's refund flow — direct (NOT calling /api/payments/refund).
  1400→    if (gift.paymentId) {
  1401→      const payment = await tx.payment.findUnique({
  1402→        where: { id: gift.paymentId },
  1403→        select: { id: true, orderId: true, amount: true, currency: true, status: true, frozen: true, version: true },
  1404→      })
  1405→      if (payment && payment.status === 'CAPTURED' && !payment.frozen) {
  1406→        const isDemoMode = !isFeatureEnabled('realPayments')
  1407→        const refund = await tx.refund.create({
  1408→          data: {
  1409→            paymentId: payment.id,
  1410→            amount: payment.amount,
  1411→            currency: payment.currency,
  1412→            status: isDemoMode ? 'REFUNDED' : 'REFUND_PENDING',
  1413→            gatewayRefundId: isDemoMode
  1414→              ? `rpf_demo_${Date.now()}_${randomUUID().slice(0, 8)}`
  1415→              : null,
  1416→            refundedAt: isDemoMode ? now : null,
  1417→          },
  1418→        })
  1419→        await tx.ledgerEntry.create({
  1420→          data: {
  1421→            paymentId: payment.id,
  1422→            entryType: 'DEBIT',
  1423→            accountType: 'CONSUMER_REVENUE',
  1424→            amount: payment.amount,
  1425→            traceId,
  1426→          },
  1427→        })
  1428→        await tx.ledgerEntry.create({
  1429→          data: {
  1430→            paymentId: payment.id,
  1431→            entryType: 'CREDIT',
  1432→            accountType: 'GATEWAY_RECEIVABLE',
  1433→            amount: payment.amount,
  1434→            traceId,
  1435→          },
  1436→        })
  1437→        const paymentUpdate = await tx.payment.updateMany({
  1438→          where: { id: payment.id, status: 'CAPTURED', version: payment.version },
  1439→          data: { status: 'REFUNDED', version: { increment: 1 } },
  1440→        })
  1441→        if (paymentUpdate.count > 0) {
  1442→          const nowIso = now.toISOString()
  1443→          await tx.order.update({
  1444→            where: { id: payment.orderId },
  1445→            data: {
  1446→              status: 'CANCELLED',
  1447→              statusHistory: JSON.stringify([
  1448→                { status: 'CONFIRMED', at: nowIso },
  1449→                { status: 'PAID', at: nowIso },
  1450→                { status: 'CANCELLED', at: nowIso },
  1451→              ]),
  1452→            },
  1453→          })
  1454→          await tx.auditLog.create({
  1455→            data: {
  1456→              actorId: null,
  1457→              actorRole: 'SYSTEM',
  1458→              action: 'PAYMENT_REFUNDED',
  1459→              metadata: JSON.stringify({
  1460→                paymentId: payment.id,
  1461→                orderId: payment.orderId,
  1462→                giftId: gift.id,
  1463→                refundId: refund.id,
  1464→                amount: payment.amount,
  1465→                source: 'gift-expire',
  1466→                demoMode: isDemoMode,
  1467→              }),
  1468→            },
  1469→          })
  1470→          await enqueueOutboxEvent(tx, {
  1471→            eventType: 'PAYMENT_REFUNDED',
  1472→            aggregateType: 'Refund',
  1473→            aggregateId: refund.id,
  1474→            payload: {
  1475→              refundId: refund.id,
  1476→              paymentId: payment.id,
  1477→              orderId: payment.orderId,
  1478→              giftId: gift.id,
  1479→              amount: payment.amount,
  1480→              currency: payment.currency,
  1481→              source: 'gift-expire',
  1482→              demoMode: isDemoMode,
  1483→            },
  1484→          })
  1485→          refundCount++
  1486→        }
  1487→      }
  1488→    }
  1489→
  1490→    // Notification to sender (gift expired)
  1491→    await tx.notification.create({
  1492→      data: {
  1493→        userId: gift.senderId,
  1494→        type: 'SYSTEM',
  1495→        title: 'A gift you sent has expired',
  1496→        body: `${gift.menuItemName} was not redeemed within 30 days. A refund has been processed.`,
  1497→        data: JSON.stringify({
  1498→          giftId: gift.id,
  1499→          menuItemName: gift.menuItemName,
  1500→          refundProcessed: !!gift.paymentId,
  1501→          deepLink: `/gifts/${gift.id}`,
  1502→        }),
  1503→        readAt: null,
  1504→      },
  1505→    })
  1506→  }
  1507→
  1508→  return {
  1509→    expiredCount: expiredGiftIds.length,
  1510→    refundCount,
  1511→    expiredGiftIds,
  1512→  }
  1513→}
