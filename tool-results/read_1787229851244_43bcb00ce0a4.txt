    52→// ════════════════════════════════════════════════════════════════════════════
    53→//  TYPES — additive shapes that 2C may return via /api/restaurants/[id] +
    54→//  /api/restaurants/[id]/menu. All fields optional so the screen degrades
    55→//  gracefully when the API hasn't been extended yet.

    58→/**
    59→ * Restaurant shape extended with the additive fields Task 2C is wiring into
    60→ * GET /api/restaurants/[id]. Every new field is optional — the screen renders
    61→ * a sensible baseline with just the Wave 1 Restaurant shape.
    62→ */
    63→export interface RestaurantDetail extends Restaurant {
    64→  /** Distance (km) from user's current location, if known. */
    65→  distanceKm?: number
    66→  /** Reward multiplier for this restaurant (e.g., 2 = "2× pts"). Default 1. */
    67→  rewardMultiplier?: number
    68→  /** Active deals — Task 2C shape: `{ title, description }`. Falls back to
    69→   *  legacy `string[]` shape if a future/older backend returns labels only. */
    70→  deals?: Array<string | { title: string; description?: string }>
    71→  /** Top-N items tagged as "popular" by the backend (Task 2C returns full
    72→   *  MenuItem-shaped objects). Falls back to ID strings if backend changes. */
    73→  popularItems?: Array<MenuItem | string>
    74→  /** Campuses linked via RestaurantCampus junction (Task 2C shape:
    75→   *  `{ id, name, isPrimary? }`). Falls back to plain strings. */
    76→  campuses?: Array<string | { id: string; name: string; isPrimary?: boolean }>
    77→  /** Open/closed state. If absent, we treat the restaurant as open. */
    78→  isOpen?: boolean
    79→}
    80→
    81→/**
    82→ * Menu payload from GET /api/restaurants/[id]/menu. The legacy response returns
    83→ * a flat `items[]` array; the Wave 1 grouped object is preserved here too so
    84→ * the screen renders identically whether the backend returns `grouped` (Task
    85→ * 2C additively includes it) or only `items` (we group locally).
    86→ */
    87→interface MenuResponse {
    88→  items?: MenuItem[]
    89→  grouped?: Record<string, MenuItem[]>
    90→  /** Optional flat list of popular MenuItems (Task 2C additively includes). */
    91→  popularItems?: MenuItem[]
    92→  /** Optional per-restaurant reward multiplier override. */
    93→  rewardMultiplier?: number
    94→}
    95→
    96→interface RestaurantResponse {
    97→  restaurant?: RestaurantDetail
    98→  error?: unknown
    99→}

   107→  onCheckout: () => void
   108→}

   113→
   114→/** Distance (px) the user must drag down at the top of the scroll container
   115→ *  to trigger a pull-to-refresh. Below this we just snap back. */
   116→const PULL_REFRESH_THRESHOLD = 70
   117→
   118→/** Height of the sticky categories tab bar (px) — used as scroll-mt offset
   119→ *  so anchored sections don't hide their header behind the bar. */
   120→const STICKY_TAB_BAR_HEIGHT = 56
   121→
   122→/** Maximum number of popular items to render in the carousel. */
   123→const MAX_POPULAR_ITEMS = 3

   125→// ════════════════════════════════════════════════════════════════════════════
   126→//  RestaurantDetailScreen — main export
   127→// ════════════════════════════════════════════════════════════════════════════
   128→
   129→export function RestaurantDetailScreen({
   130→  restaurantId,
   131→  onBack,
   132→  onCheckout,
   133→}: RestaurantDetailScreenProps) {
   134→  const prefersReduced = useReducedMotion()
   135→  const { toast } = useToast()
   136→  const cart = useCart()
   137→
   138→  // ─── State ────────────────────────────────────────────────────────────────
   139→  const [restaurant, setRestaurant] = React.useState<RestaurantDetail | null>(null)
   140→  const [menuItems, setMenuItems] = React.useState<MenuItem[]>([])
   141→  const [loading, setLoading] = React.useState(true)
   142→  const [error, setError] = React.useState<string | null>(null)
   143→  const [refreshing, setRefreshing] = React.useState(false)
   144→
   145→  // ─── Confirm-switch dialog state ─────────────────────────────────────────
   146→  const [pendingAdd, setPendingAdd] = React.useState<MenuItem | null>(null)
   147→
   148→  // ─── Active category for sticky tab highlight ────────────────────────────
   149→  const [activeCategory, setActiveCategory] = React.useState<string>('')
   150→
   151→  // ─── Pull-to-refresh state ────────────────────────────────────────────────
   152→  const [pullDistance, setPullDistance] = React.useState(0)
   153→  const touchStartY = React.useRef<number | null>(null)
   154→  const isPulling = React.useRef(false)
   155→
   156→  // ─── Refs for anchored scroll-to-category ─────────────────────────────────
   157→  const sectionRefs = React.useRef<Map<string, HTMLElement>>(new Map())
   158→  const scrollContainerRef = React.useRef<HTMLDivElement | null>(null)
   159→
   160→  // ─── Derived data ──────────────────────────────────────────────────────────
   161→  const groupedMenu = React.useMemo(() => groupByCategory(menuItems), [menuItems])
   162→  const categories = React.useMemo(() => Object.keys(groupedMenu), [groupedMenu])
   163→
   164→  const rewardMultiplier =
   165→    restaurant?.rewardMultiplier && restaurant.rewardMultiplier > 1
   166→      ? restaurant.rewardMultiplier
   167→      : undefined
   168→
   169→  // Resolve "popular items" — the Task 2C API returns popularItems as full
   170→  // MenuItem objects on the RESTAURANT payload (top-3 by category/name). The
   171→  // legacy menu endpoint doesn't include them, so we read from
   172→  // `restaurant.popularItems`. Strings in the array (older shape) are matched
   173→  // against menuItems by ID.
   174→  const popularResolved = React.useMemo(() => {
   175→    if (!restaurant?.popularItems?.length) return []
   176→    const out: MenuItem[] = []
   177→    for (const p of restaurant.popularItems) {
   178→      if (typeof p === 'string') {
   179→        const matched = menuItems.find((m) => m.id === p)
   180→        if (matched) out.push(matched)
   181→      } else {
   182→        // Object-shape popular item — verify it's still on the menu so the
   183→        // cart integration has a real MenuItem to add.
   184→        const matched = menuItems.find((m) => m.id === p.id)
   185→        out.push(matched ?? p)
   186→      }
   187→      if (out.length >= MAX_POPULAR_ITEMS) break
   188→    }
   189→    return out
   190→  }, [restaurant?.popularItems, menuItems])
   191→
   192→  // ─── Data fetching ────────────────────────────────────────────────────────
   193→  const loadAll = React.useCallback(
   194→    async (opts?: { silent?: boolean }) => {
   195→      const silent = opts?.silent ?? false
   196→      if (!silent) {
   197→        setLoading(true)
   198→        setError(null)
   199→      } else {
   200→        setRefreshing(true)
   201→      }
   202→      try {
   203→        const [rRes, mRes] = await Promise.all([
   204→          fetch(`/api/restaurants/${restaurantId}`, { cache: 'no-store' }),
   205→          fetch(`/api/restaurants/${restaurantId}/menu`, { cache: 'no-store' }),
   206→        ])
   207→        if (!rRes.ok) {
   208→          throw new Error(`Couldn't load this restaurant (${rRes.status})`)
   209→        }
   210→        const rJson = (await rRes.json().catch(() => ({}))) as RestaurantResponse
   211→        const mJson = (await mRes.json().catch(() => ({}))) as MenuResponse
   212→
   213→        if (!rJson.restaurant) {
   214→          throw new Error('This restaurant could not be found.')
   215→        }
   216→        setRestaurant(rJson.restaurant as RestaurantDetail)
   217→
   218→        // The Task 2C menu endpoint returns a flat `items[]` array — we group
   219→        // it locally to preserve category order (alphabetical by category,
   220→        // then by name within each category, matching the SQL ORDER BY).
   221→        const items = mJson.items ?? []
   222→        setMenuItems(items)
   223→
   224→        // Seed the active category as the first one alphabetically.
   225→        const firstCat = Object.keys(groupByCategory(items))[0]
   226→        if (firstCat) setActiveCategory(firstCat)
   227→      } catch (e) {
   228→        const msg = e instanceof Error ? e.message : 'Something went wrong.'
   229→        setError(msg)
   230→        if (silent) {
   231→          // Surface silent failures via toast so the user knows the refresh
   232→          // didn't work even though their previous data is still visible.
   233→          toast({
   234→            title: 'Refresh failed',
   235→            description: msg,
   236→            variant: 'destructive',
   237→          })
   238→        }
   239→      } finally {
   240→        if (!silent) setLoading(false)
   241→        else setRefreshing(false)
   242→        setPullDistance(0)
   243→        isPulling.current = false
   244→      }
   245→    },
   246→    [restaurantId, toast],
   247→  )
   248→
   249→  // Initial load + reload when restaurantId changes.
   250→  React.useEffect(() => {
   251→    loadAll()
   252→  }, [loadAll])
   253→
   254→  // ─── Cart helpers ──────────────────────────────────────────────────────────
   255→  const handleAdd = React.useCallback(
   256→    (item: MenuItem) => {
   257→      if (!restaurant) return
   258→      // Cart already has items from a DIFFERENT restaurant → confirm first.
   259→      if (
   260→        cart.restaurantId &&
   261→        cart.restaurantId !== restaurant.id &&
   262→        cart.count() > 0
   263→      ) {
   264→        setPendingAdd(item)
   265→        return
   266→      }
   267→      cart.add(item, restaurant.id, restaurant.name)
   268→      toast({
   269→        title: 'Added to cart',
   270→        description: `${item.name} · ${inr(item.price)}`,
   271→      })
   272→    },
   273→    [restaurant, cart, toast],
   274→  )
   275→
   276→  const handleIncrement = React.useCallback(
   277→    (item: MenuItem) => {
   278→      if (!restaurant) return
   279→      // Switch-restaurant safety — same guard as handleAdd. In practice this
   280→      // branch rarely fires (you can't increment an item not in the cart), but
   281→      // we keep it for robustness.
   282→      if (
   283→        cart.restaurantId &&
   284→        cart.restaurantId !== restaurant.id &&
   285→        cart.count() > 0
   286→      ) {
   287→        setPendingAdd(item)
   288→        return
   289→      }
   290→      cart.add(item, restaurant.id, restaurant.name)
   291→    },
   292→    [restaurant, cart],
   293→  )
   294→
   295→  const handleDecrement = React.useCallback(
   296→    (item: MenuItem) => {
   297→      cart.decrement(item.id)
   298→    },
   299→    [cart],
   300→  )
   301→
   302→  // Confirmed switch — clear cart and add the pending item to the new
   303→  // restaurant's cart.
   304→  const confirmSwitchAndAdd = React.useCallback(() => {
   305→    if (!pendingAdd || !restaurant) return
   306→    cart.clear()
   307→    cart.add(pendingAdd, restaurant.id, restaurant.name)
   308→    toast({
   309→      title: 'Started a new order',
   310→      description: `${pendingAdd.name} added from ${restaurant.name}.`,
   311→    })
   312→    setPendingAdd(null)
   313→  }, [pendingAdd, restaurant, cart, toast])
   314→
   315→  const cancelSwitch = React.useCallback(() => {
   316→    setPendingAdd(null)
   317→  }, [])
   318→
   319→  // ─── Anchor-scroll to a category ──────────────────────────────────────────
   320→  const scrollToCategory = React.useCallback((category: string) => {
   321→    const el = sectionRefs.current.get(category)
   322→    if (!el) return
   323→    setActiveCategory(category)
   324→    el.scrollIntoView({ behavior: 'smooth', block: 'start' })
   325→  }, [])
   326→
   327→  // ─── Active-category tracking via IntersectionObserver ───────────────────
   328→  React.useEffect(() => {
   329→    if (categories.length === 0 || typeof IntersectionObserver === 'undefined') return
   330→    const observer = new IntersectionObserver(
   331→      (entries) => {
   332→        // Find the topmost intersecting section whose top is closest to the
   333→        // sticky tab bar's bottom edge — that's the section currently "in view".
   334→        const visible = entries
   335→          .filter((e) => e.isIntersecting)
   336→          .sort((a, b) => a.boundingClientRect.top - b.boundingClientRect.top)
   337→        if (visible[0]) {
   338→          const id = visible[0].target.getAttribute('data-category')
   339→          if (id) setActiveCategory(id)
   340→        }
   341→      },
   342→      {
   343→        // Trigger when a section's top edge enters the band just below the
   344→        // sticky tab bar (so the highlight updates as the user scrolls).
   345→        rootMargin: `-${STICKY_TAB_BAR_HEIGHT + 16}px 0px -60% 0px`,
   346→        threshold: 0,
   347→      },
   348→    )
   349→    sectionRefs.current.forEach((el) => observer.observe(el))
   350→    return () => observer.disconnect()
   351→  }, [categories])
   352→
   353→  // ─── Pull-to-refresh touch handlers ────────────────────────────────────────
   354→  const onTouchStart = React.useCallback((e: React.TouchEvent) => {
   355→    const scroller = scrollContainerRef.current
   356→    if (!scroller) return
   357→    if (scroller.scrollTop <= 0) {
   358→      touchStartY.current = e.touches[0]?.clientY ?? null
   359→      isPulling.current = true
   360→    } else {
   361→      touchStartY.current = null
   362→      isPulling.current = false
   363→    }
   364→  }, [])
   365→
   366→  const onTouchMove = React.useCallback(
   367→    (e: React.TouchEvent) => {
   368→      if (!isPulling.current || touchStartY.current == null) return
   369→      const scroller = scrollContainerRef.current
   370→      if (!scroller || scroller.scrollTop > 0) {
   371→        isPulling.current = false
   372→        setPullDistance(0)
   373→        return
   374→      }
   375→      const dy = (e.touches[0]?.clientY ?? 0) - touchStartY.current
   376→      // Only positive deltas (pulling DOWN) count. Apply a rubber-band factor
   377→      // so the indicator doesn't feel like it tracks 1:1 with the finger.
   378→      if (dy > 0) {
   379→        const eased = Math.min(PULL_REFRESH_THRESHOLD * 1.5, dy * 0.5)
   380→        setPullDistance(eased)
   381→      } else {
   382→        setPullDistance(0)
   383→      }
   384→    },
   385→    [],
   386→  )
   387→
   388→  const onTouchEnd = React.useCallback(() => {
   389→    if (!isPulling.current) return
   390→    if (pullDistance >= PULL_REFRESH_THRESHOLD) {
   391→      loadAll({ silent: true })
   392→    } else {
   393→      setPullDistance(0)
   394→    }
   395→    isPulling.current = false
   396→    touchStartY.current = null
   397→  }, [pullDistance, loadAll])
   398→
   399→  // ─── Derived display values ────────────────────────────────────────────────
   400→  const cartTotal = cart.total()
   401→  const cartCount = cart.count()
   402→  const showCartBar = cartCount > 0 && !!restaurant && cart.restaurantId === restaurant.id
   403→
   404→  // ════════════════════════════════════════════════════════════════════════════
   405→  //  RENDER
   406→  // ════════════════════════════════════════════════════════════════════════════
   407→
   408→  // ─── Error state ────────────────────────────────────────────────────────────
   409→  if (error && !restaurant) {
   410→    return (
   411→      <div className="mx-auto max-w-2xl px-4 py-6">
   412→        <Button variant="ghost" size="sm" className="mb-4" onClick={onBack}>
   413→          <ArrowLeft className="mr-1 h-4 w-4" /> Back to restaurants
   414→        </Button>
   415→        <Card className="border-destructive/30 bg-destructive/5">
   416→          <CardContent className="flex flex-col items-center gap-4 py-8 text-center">
   417→            <div className="flex h-14 w-14 items-center justify-center rounded-full bg-destructive/10 text-destructive">
   418→              <AlertTriangle className="h-7 w-7" />
   419→            </div>
   420→            <div>
   421→              <p className="text-base font-semibold text-foreground">Couldn&apos;t load this restaurant</p>
   422→              <p className="mt-1 text-sm text-muted-foreground">{error}</p>
   423→            </div>
   424→            <Button onClick={() => loadAll()}>
   425→              <RefreshCw className="mr-1 h-4 w-4" /> Try again
   426→            </Button>
   427→          </CardContent>
   428→        </Card>
   429→      </div>
   430→    )
   431→  }
   432→
   433→  // ─── Loading state (skeletons + shimmer hero) ──────────────────────────────
   434→  if (loading && !restaurant) {
   435→    return (
   436→      <div className="mx-auto max-w-2xl px-4 py-6">
   437→        <RestaurantDetailSkeleton />
   438→      </div>
   439→    )
   440→  }
   441→
   442→  // ─── Restaurant not found (after fetch) ────────────────────────────────────
   443→  if (!restaurant) {
   444→    return (
   445→      <div className="mx-auto max-w-2xl px-4 py-6">
   446→        <Button variant="ghost" size="sm" className="mb-4" onClick={onBack}>
   447→          <ArrowLeft className="mr-1 h-4 w-4" /> Back to restaurants
   448→        </Button>
   449→        <EmptyState
   450→          variant="no-restaurants"
   451→          title="Restaurant not found"
   452→          description="This restaurant may have been removed or is no longer accepting orders."
   453→          actionLabel="Back to restaurants"
   454→          onAction={onBack}
   455→        />
   456→      </div>
   457→    )
   458→  }
   459→
   460→  const isOpen = restaurant.isOpen ?? true
   461→  // Normalise deals — Task 2C returns `{ title, description }` objects;
   462→  // legacy shape is a bare string. We render both as DealBadge cards.
   463→  const deals = (restaurant.deals ?? []).map((d, idx) => ({
   464→    id: idx,
   465→    label: typeof d === 'string' ? d : d.title,
   466→    description: typeof d === 'string' ? undefined : d.description,
   467→  }))
   468→  // Normalise campuses — Task 2C returns `{ id, name, isPrimary }` objects;
   469→  // legacy shape is a bare string. We render both as CampusBadge chips.
   470→  const campusNames = (restaurant.campuses ?? []).map((c) =>
   471→    typeof c === 'string' ? c : c.name,
   472→  )
   473→  const showRewardBanner = !!rewardMultiplier && rewardMultiplier > 1
   474→
   475→  // Pickup estimate — use restaurant.prepTimeMins as the baseline.
   476→  const pickupMinutes = restaurant.prepTimeMins
   477→
   478→  return (
   479→    <div
   480→      ref={scrollContainerRef}
   481→      onTouchStart={onTouchStart}
   482→      onTouchMove={onTouchMove}
   483→      onTouchEnd={onTouchEnd}
   484→      className="relative mx-auto max-w-2xl pb-40"
   485→    >
   486→      {/* ══════════════════════════════════════════════════════════════════════
   487→          HERO HEADER
   488→         ══════════════════════════════════════════════════════════════════════ */}
   489→      <section
   490→        className="relative -mx-4 -mt-6 h-56 w-[calc(100%+2rem)] overflow-hidden sm:h-64"
   491→        aria-label={`${restaurant.name} header`}
   492→      >
   493→        {/* Hero image with cuisine gradient fallback */}
   494→        {restaurant.image ? (
   495→          <img
   496→            src={restaurant.image}
   497→            alt={restaurant.name}
   498→            className="absolute inset-0 h-full w-full object-cover"
   499→            loading="eager"
   500→          />
   501→        ) : (
   502→          <div
   503→            className={cn(
   504→              'flex h-full w-full items-center justify-center bg-gradient-to-br',
   505→              cuisineGradient(restaurant.cuisine),
   506→            )}
   507→            aria-hidden="true"
   508→          >
   509→            <CuisineIcon cuisine={restaurant.cuisine} className="text-6xl opacity-90" />
   510→          </div>
   511→        )}
   512→
   513→        {/* Gradient scrim for legibility of overlay text */}
   514→        <div className="absolute inset-0 bg-gradient-to-b from-black/40 via-transparent to-black/70" />
   515→
   516→        {/* Floating top-left back button */}
   517→        <button
   518→          type="button"
   519→          aria-label="Back to restaurants"
   520→          onClick={onBack}
   521→          className="snak-focus-ring absolute left-3 top-3 z-20 flex h-10 w-10 items-center justify-center rounded-full bg-background/80 text-foreground shadow-md backdrop-blur-md transition hover:bg-background"
   522→        >
   523→          <ArrowLeft className="h-5 w-5" />
   524→        </button>
   525→
   526→        {/* Floating top-right heart/share — placeholders */}
   527→        <div className="absolute right-3 top-3 z-20 flex gap-2">
   528→          <button
   529→            type="button"
   530→            aria-label="Save restaurant to favourites"
   531→            onClick={() =>
   532→              toast({ title: 'Saved!', description: 'Favourites are coming soon.' })
   533→            }
   534→            className="snak-focus-ring flex h-10 w-10 items-center justify-center rounded-full bg-background/80 text-foreground shadow-md backdrop-blur-md transition hover:bg-background"
   535→          >
   536→            <Heart className="h-5 w-5" />
   537→          </button>
   538→          <button
   539→            type="button"
   540→            aria-label="Share this restaurant"
   541→            onClick={() => {
   542→              if (typeof navigator !== 'undefined' && navigator.share) {
   543→                navigator
   544→                  .share({ title: restaurant.name, text: `Order from ${restaurant.name} on SnakZap` })

   991→  onDecrement,

  1049→        <div className="mt-auto flex items-center justify-between gap-2 pt-2">
