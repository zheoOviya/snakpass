   137→export function CheckoutView({ onBack, onSuccess }: CheckoutViewProps) {
   138→  const cart = useCart()
   139→  const { toast } = useToast()
   140→  const prefersReduced = useReducedMotion()
   141→
   142→  // ---- Form state ----------------------------------------------------------
   143→  const [pickupName, setPickupName] = useState('')
   144→  const [pickupPhone, setPickupPhone] = useState('')
   145→  const [note, setNote] = useState('')
   146→  const [phase, setPhase] = useState<Phase>('idle')
   147→  const [prefillLoaded, setPrefillLoaded] = useState(false)
   148→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('razorpay')
   149→  const [restaurantCuisine, setRestaurantCuisine] = useState<string | null>(null)
   150→
   151→  // ---- Prefill pickup name + phone from the active session -----------------
   152→  useEffect(() => {
   153→    let cancelled = false
   154→    fetch('/api/auth/me')
   155→      .then((r) => r.json())
   156→      .then((d) => {
   157→        if (cancelled || !d?.user) return
   158→        if (d.user.phone) setPickupPhone(d.user.phone)
   159→        if (d.user.name) setPickupName(d.user.name)
   160→      })
   161→      .catch(() => {
   162→        /* session fetch best-effort — user may be unauthenticated */
   163→      })
   164→      .finally(() => !cancelled && setPrefillLoaded(true))
   165→    return () => {
   166→      cancelled = true
   167→    }
   168→  }, [])
   169→
   170→  // ---- Best-effort fetch of restaurant cuisine for the banner --------------
   171→  // cart-store only persists restaurantName (not cuisine). We do a single
   172→  // GET /api/restaurants/[id] on mount to surface the cuisine tag. Failure
   173→  // is silent — the banner falls back to just the name.
   174→  useEffect(() => {
   175→    if (!cart.restaurantId) return
   176→    let cancelled = false
   177→    fetch(`/api/restaurants/${cart.restaurantId}`)
   178→      .then((r) => (r.ok ? r.json() : null))
   179→      .then((d) => {
   180→        if (cancelled || !d?.cuisine) return
   181→        setRestaurantCuisine(d.cuisine)
   182→      })
   183→      .catch(() => {
   184→        /* cuisine is decorative — silent fallback */
   185→      })
   186→    return () => {
   187→      cancelled = true
   188→    }
   189→  }, [cart.restaurantId])
   190→
   191→  // ---- Derived pricing -----------------------------------------------------
   192→  const pricing = useMemo(() => cart.pricing(), [cart])
   193→  const cartCount = cart.count()
   194→  const itemsCount = cart.lines.reduce((s, l) => s + l.quantity, 0)
   195→
   196→  // 10-digit Indian mobile (strip non-digits before validating).
   197→  const phoneDigits = pickupPhone.replace(/\D/g, '').slice(-PHONE_DIGITS)
   198→  const phoneValid = phoneDigits.length === PHONE_DIGITS
   199→  const nameValid = pickupName.trim().length >= 2
   200→  const canPay =
   201→    phase === 'idle' &&
   202→    cartCount > 0 &&
   203→    !!cart.restaurantId &&
   204→    phoneValid &&
   205→    nameValid
   206→
   207→  // ---- Build PricingBreakdown rows (blueprint §4 P4) -----------------------
   208→  // Food subtotal + tax + platform fee − discount − reward + tip = total.
   209→  const pricingRows: PricingRow[] = useMemo(() => {
   210→    const rows: PricingRow[] = []
   211→    rows.push({
   212→      key: 'subtotal',
   213→      label: 'Food subtotal',
   214→      amountPaise: pricing.subtotal,
   215→      kind: 'add',
   216→    })
   217→    rows.push({
   218→      key: 'tax',
   219→      label: 'GST (5%)',
   220→      amountPaise: pricing.tax,
   221→      kind: 'add',
   222→    })
   223→    // Platform fee only rendered when non-zero (SnakZap low-fee model = 0 in MVP).
   224→    if (pricing.platformFee > 0) {
   225→      rows.push({
   226→        key: 'platform-fee',
   227→        label: 'Platform fee',
   228→        amountPaise: pricing.platformFee,
   229→        kind: 'add',
   230→      })
   231→    }
   232→    if (pricing.discount > 0) {
   233→      rows.push({
   234→        key: 'discount',
   235→        label: 'Discount',
   236→        amountPaise: -pricing.discount,
   237→        kind: 'sub',
   238→        hint: cart.couponCode ? `Coupon ${cart.couponCode} applied` : undefined,
   239→      })
   240→    }
   241→    if (pricing.rewardDiscount > 0) {
   242→      const pts = cart.rewardPointsToRedeem
   243→      rows.push({
   244→        key: 'reward-discount',
   245→        label: 'Reward discount',
   246→        amountPaise: -pricing.rewardDiscount,
   247→        kind: 'sub',
   248→        hint: `${pts} pts = ${inr(pricing.rewardDiscount)} off`,
   249→      })
   250→    }
   251→    if (pricing.tip > 0) {
   252→      rows.push({
   253→        key: 'tip',
   254→        label: 'Tip',
   255→        amountPaise: pricing.tip,
   256→        kind: 'add',
   257→      })
   258→    }
   259→    rows.push({
   260→      key: 'total',
   261→      label: 'Total payable',
   262→      amountPaise: pricing.total,
   263→      kind: 'total',
   264→    })
   265→    return rows
   266→  }, [pricing, cart.couponCode, cart.rewardPointsToRedeem])
   267→
   268→  // ---- Pickup time display (from cart.pickupTime) --------------------------
   269→  const pickupTimeLabel = useMemo(() => {
   270→    if (!cart.pickupTime) return 'ASAP (in ~15 min)'
   271→    // cart.pickupTime is an ISO time string or null for ASAP. We surface a
   272→    // friendlier label — the actual scheduling UI lives in the Cart screen
   273→    // (Task 3A).
   274→    try {
   275→      const d = new Date(cart.pickupTime)
   276→      if (!isNaN(d.getTime())) {
   277→        return `Scheduled · ${d.toLocaleString('en-IN', {
   278→          hour: '2-digit',
   279→          minute: '2-digit',
   280→          day: 'numeric',
   281→          month: 'short',
   282→        })}`
   283→      }
   284→    } catch {
   285→      /* fall through to ASAP */
   286→    }
   287→    return 'ASAP (in ~15 min)'
   288→  }, [cart.pickupTime])
   289→
   290→  // ---- Two-phase pay handler (PRESERVED VERBATIM) --------------------------
   291→  const handlePay = async () => {
   292→    if (!cart.restaurantId || cart.lines.length === 0) {
   293→      toast({
   294→        title: 'Your cart is empty',
   295→        description: 'Add an item before checking out.',
   296→        variant: 'destructive',
   297→      })
   298→      return
   299→    }
   300→    if (!nameValid) {
   301→      toast({
   302→        title: 'Pickup name required',
   303→        description: 'Please tell us who is picking up the order.',
   304→        variant: 'destructive',
   305→      })
   306→      return
   307→    }
   308→    if (!phoneValid) {
   309→      toast({
   310→        title: 'Valid phone required',
   311→        description: 'Enter a 10-digit Indian mobile number.',
   312→        variant: 'destructive',
   313→      })
   314→      return
   315→    }
   316→
   317→    // Combine pickup details + free-form note into a single note string.
   318→    // The order schema has one `note` field (max 500 chars), so we pack the
   319→    // pickup contact details + instructions together.
   320→    const pickupBlock = `Pickup: ${pickupName.trim()} · ${phoneDigits}`
   321→    const methodBlock = `Payment: ${paymentMethod.toUpperCase()}`
   322→    const noteBlock = note.trim() ? `Note: ${note.trim()}` : null
   323→    const composedNote = [pickupBlock, methodBlock, noteBlock]
   324→      .filter(Boolean)
   325→      .join('\n')
   326→      .slice(0, MAX_NOTE)
   327→
   328→    setPhase('placing')
   329→    let createdOrder: Order | null = null
   330→
   331→    try {
   332→      // === Step A: create the order (status CONFIRMED, returns pickupOtp) ===
   333→      const orderRes = await csrfFetch('/api/orders', {
   334→        method: 'POST',
   335→        headers: { 'content-type': 'application/json' },
   336→        body: JSON.stringify({
   337→          restaurantId: cart.restaurantId,
   338→          items: cart.lines.map((l) => ({
   339→            menuItemId: l.menuItemId,
   340→            name: l.name,
   341→            price: l.price,
   342→            quantity: l.quantity,
   343→          })),
   344→          note: composedNote,
   345→        }),
   346→        idempotencyKey: crypto.randomUUID(),
   347→      })
   348→      const orderData = await orderRes.json().catch(() => ({}))
   349→      if (!orderRes.ok) {
   350→        throw new Error(
   351→          orderData?.error?.message ??
   352→            (typeof orderData?.error === 'string' ? orderData.error : null) ??
   353→            'Could not place order. Please try again.',
   354→        )
   355→      }
   356→      createdOrder = orderData.order as Order
   357→
   358→      // === Step B: capture payment (DEMO mode — realPayments flag is OFF) ===
   359→      // In demo mode the backend accepts any non-empty razorpaySignature, so we
   360→      // synthesise a deterministic demo payment id + signature. No real money
   361→      // moves — this is purely to drive the order into the PAID state.
   362→      setPhase('paying')
   363→      const payRes = await csrfFetch('/api/payments', {
   364→        method: 'POST',
   365→        headers: { 'content-type': 'application/json' },
   366→        body: JSON.stringify({
   367→          orderId: createdOrder.id,
   368→          razorpayPaymentId: `pay_demo_${Date.now()}`,
   369→          razorpaySignature: `sig_demo_${Date.now()}`,
   370→        }),
   371→        idempotencyKey: crypto.randomUUID(),
   372→      })
   373→      const payData = await payRes.json().catch(() => ({}))
   374→      if (!payRes.ok) {
   375→        // Order was created but payment failed — surface the error but still
   376→        // transition to tracking so the user can see the CONFIRMED order and
   377→        // (in a future iteration) retry payment from "My Orders".
   378→        cart.clear()
   379→        toast({
   380→          title: 'Payment failed — order placed but payment pending',
   381→          description:
   382→            payData?.error?.message ??
   383→              (typeof payData?.error === 'string' ? payData.error : null) ??
   384→              'Your order is confirmed — please retry payment from My Orders.',
   385→          variant: 'destructive',
   386→        })
   387→        onSuccess(createdOrder)
   388→        return
   389→      }
   390→
   391→      // The payments route updates order.status to 'PAID' inside its
   392→      // transaction, but the response only returns the Payment object — so
   393→      // re-fetch the order to surface the fresh PAID status + statusHistory.
   394→      let finalOrder: Order = createdOrder
   395→      try {
   396→        const r = await fetch(`/api/orders/${createdOrder.id}`)
   397→        const d = await r.json()
   398→        if (d?.order) finalOrder = d.order as Order
   399→      } catch {
   400→        /* best-effort refresh — fall back to createdOrder */
   401→      }
   402→
   403→      cart.clear()
   404→      toast({
   405→        title: 'Payment confirmed! 💳',
   406→        description: `Pickup OTP ${finalOrder.pickupOtp} · ${inr(finalOrder.totalAmount)}`,
   407→      })
   408→      onSuccess(finalOrder)
   409→    } catch (e) {
   410→      // Order creation (or pre-order validation) failed — keep the cart so
   411→      // the user can adjust and try again.
   412→      toast({
   413→        title: createdOrder ? 'Payment failed' : 'Checkout failed',
   414→        description: (e as Error).message ?? 'Please try again.',
   415→        variant: 'destructive',
   416→      })
   417→    } finally {
   418→      setPhase('idle')
   419→    }
   420→  }
   421→
   422→  // ---- Empty-cart safety ---------------------------------------------------
   423→  // Shouldn't normally happen (cart bar is hidden when empty), but a direct
   424→  // route hit would land here.
   425→  if (cartCount === 0) {
   426→    return (
   427→      <div className="mx-auto max-w-2xl px-4 py-6">
   428→        <Button variant="ghost" size="sm" className="mb-3" onClick={onBack}>
   429→          <ArrowLeft className="mr-1 h-4 w-4" /> Back to menu
   430→        </Button>
   431→        <div className="rounded-xl border border-dashed py-16 text-center text-muted-foreground">
   432→          <Store className="mx-auto mb-2 h-8 w-8" />
   433→          Your cart is empty. Add some items before checking out.
   434→        </div>
   435→      </div>
   436→    )
   437→  }
   438→
   439→  const processing = phase !== 'idle'
   440→  const phaseLabel =
   441→    phase === 'placing'
   442→      ? 'Placing order…'
   443→      : phase === 'paying'
   444→        ? 'Processing payment…'
   445→        : null
   446→
   447→  // Stagger index helper — increments per section.
   448→  let sectionIndex = 0
   449→  const nextSection = () => sectionIndex++
   450→
   451→  return (
   452→    <div className="mx-auto max-w-2xl px-4 py-6 pb-40">
   453→      {/* ─────────────────────────────────────────────────────────────────
   454→          1. Restaurant banner
   455→         ───────────────────────────────────────────────────────────────── */}
   456→      <motion.section
   457→        custom={nextSection()}
   458→        initial={prefersReduced ? false : 'hidden'}
   459→        animate="visible"
   460→        variants={sectionEnter}
   461→        className="mb-4"
   462→      >
   463→        <Card className="overflow-hidden border-teal-200 bg-gradient-to-br from-teal-50 via-white to-white dark:border-teal-900 dark:from-teal-950/30 dark:via-card dark:to-card">
   464→          <CardContent className="flex items-center gap-3 p-4">
   465→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-xl bg-teal-500/15 text-teal-700 dark:text-teal-300">
   466→              <Store className="h-6 w-6" />
   467→            </div>
   468→            <div className="min-w-0 flex-1">
   469→              <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
   470→                Pickup from
   471→              </p>
   472→              <p className="truncate font-bold leading-tight">
   473→                {cart.restaurantName ?? 'Restaurant'}
   474→              </p>
   475→              {restaurantCuisine && (
   476→                <p className="mt-0.5 flex items-center gap-1 text-xs text-muted-foreground">
   477→                  <CuisineIcon cuisine={restaurantCuisine} />
   478→                  <span>{restaurantCuisine}</span>
   479→                </p>
   480→              )}
   481→            </div>
   482→            <Button
   483→              variant="ghost"
   484→              size="sm"
   485→              onClick={onBack}
   486→              disabled={processing}
   487→              className="text-teal-700 hover:bg-teal-500/10 hover:text-teal-800 dark:text-teal-300 dark:hover:bg-teal-500/10"
   488→            >
   489→              <ArrowLeft className="mr-1 h-4 w-4" /> Cart
   490→            </Button>
   491→          </CardContent>
   492→        </Card>
   493→      </motion.section>
   494→
   495→      {/* ─────────────────────────────────────────────────────────────────
   496→          2. Order summary
   497→         ───────────────────────────────────────────────────────────────── */}
   498→      <motion.section
   499→        custom={nextSection()}
   500→        initial={prefersReduced ? false : 'hidden'}
   501→        animate="visible"
   502→        variants={sectionEnter}
   503→        className="mb-4"
   504→      >
   505→        <Card>
   506→          <CardContent className="p-4">
   507→            <div className="mb-3 flex items-center justify-between">
   508→              <h3 className="text-sm font-semibold">Order summary</h3>
   509→              <Badge variant="secondary" className="text-xs">
   510→                {itemsCount} item{itemsCount === 1 ? '' : 's'}
   511→              </Badge>
   512→            </div>
   513→            <div className="space-y-2.5">
   514→              {cart.lines.map((l) => {
   515→                const linePaise = l.price * l.quantity
   516→                const pts = pointsEarnedFor(linePaise / 100)
   517→                return (
   518→                  <motion.div
   519→                    key={l.menuItemId}
   520→                    layout={!prefersReduced}
   521→                    initial={prefersReduced ? false : { opacity: 0, x: -4 }}
   522→                    animate={{ opacity: 1, x: 0 }}
   523→                    transition={{ duration: 0.2, ease: [0.3, 0, 0, 1] as const }}
   524→                    className="flex items-center gap-3"
   525→                  >
   526→                    <VegBadge veg={l.isVeg} />
   527→                    <div className="min-w-0 flex-1">
   528→                      <p className="truncate text-sm font-medium">{l.name}</p>
   529→                      <div className="mt-0.5 flex flex-wrap items-center gap-2">
   530→                        <p className="text-xs text-muted-foreground">
   531→                          {inr(l.price)} × {l.quantity}
   532→                        </p>
   533→                        {pts > 0 && (
   534→                          <RewardBadge className="!px-1.5 !py-0 !text-[10px]">
   535→                            +{pts}
   536→                          </RewardBadge>
   537→                        )}
   538→                      </div>
   539→                    </div>
   540→                    <p className="text-sm font-semibold tabular-nums">
   541→                      {inr(linePaise)}
   542→                    </p>
   543→                  </motion.div>
   544→                )
   545→              })}
   546→            </div>
   547→            <Separator className="my-3" />
   548→            <div className="flex items-center justify-between text-sm">
   549→              <span className="text-muted-foreground">Subtotal</span>
   550→              <span className="font-medium tabular-nums">
   551→                {inr(pricing.subtotal)}
   552→              </span>
   553→            </div>
   554→          </CardContent>
   555→        </Card>
   556→      </motion.section>
   557→
